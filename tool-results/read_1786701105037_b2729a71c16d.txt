    54→/**
    55→ * High-resolution timestamp for precise event tracking
    56→ */
    57→class DebugTimestamp {
    58→public:
    59→    static DebugTimestamp now() {
    60→        DebugTimestamp ts;
    61→        ts.wall_clock = std::chrono::system_clock::now();
    62→        ts.steady_clock = std::chrono::steady_clock::now();
    63→        ts.frame_number = 0;  // Set by caller if in emulation context
    64→        return ts;
    65→    }
    66→    
    67→    static DebugTimestamp atFrame(uint64_t frame) {
    68→        DebugTimestamp ts = now();
    69→        ts.frame_number = frame;
    70→        return ts;
    71→    }
    72→    
    73→    /**
    74→     * ISO 8601 formatted timestamp
    75→     */
    76→    std::string toISO8601() const {
    77→        auto time_t = std::chrono::system_clock::to_time_t(wall_clock);
    78→        std::stringstream ss;
    79→        ss << std::put_time(std::gmtime(&time_t), "%Y-%m-%dT%H:%M:%SZ");
    80→        return ss.str();
    81→    }
    82→    
    83→    /**
    84→     * File-friendly timestamp (no colons)
    85→     */
    86→    std::string toFileFriendly() const {
    87→        auto time_t = std::chrono::system_clock::to_time_t(wall_clock);
    88→        std::stringstream ss;
    89→        ss << std::put_time(std::gmtime(&time_t), "%Y%m%d_%H%M%S");
    90→        
    91→        // Add milliseconds for uniqueness
    92→        auto ms = std::chrono::duration_cast<std::chrono::milliseconds>(
    93→            wall_clock.time_since_epoch()) % 1000;
    94→        ss << "_" << ms.count();
    95→        
    96→        return ss.str();
    97→    }
    98→    
    99→    /**
   100→     * Microseconds since epoch (for sorting/comparison)
   101→     */
   102→    uint64_t toMicroseconds() const {
   103→        return std::chrono::duration_cast<std::chrono::microseconds>(
   104→            wall_clock.time_since_epoch()).count();
   105→    }
   106→    
   107→    // Time sources
   108→    std::chrono::system_clock::time_point wall_clock;
   109→    std::chrono::steady_clock::time_point steady_clock;
   110→    uint64_t frame_number{0};
   111→    
   112→    // Comparison operators
   113→    inline bool operator<(const DebugTimestamp& other) const {
   114→        return wall_clock < other.wall_clock;
   115→    }
   116→    
   117→    inline bool operator>(const DebugTimestamp& other) const {
   118→        return wall_clock > other.wall_clock;
   119→    }
   120→    
   121→    inline bool operator==(const DebugTimestamp& other) const {
   122→        return wall_clock == other.wall_clock;
   123→    }
   124→};

   283→/**
   284→ * Precise source attribution for any event
   285→ */
   286→struct SourceLocation {
   287→    std::string file_name;
   288→    int line_number{0};
   289→    std::string function_name;
   290→    std::string module_name;  // Library/module
   291→    
   292→    // For HLE calls
   293→    std::string hle_function_name;
   294→    int hle_function_id{-1};
   295→    
   296→    // For guest code
   297→    uint64_t guest_address{0};
   298→    std::string symbol_name;
   299→    
   300→    bool isValid() const {
   301→        return !file_name.empty() || !function_name.empty() || 
   302→               !hle_function_name.empty() || guest_address != 0;
   303→    }
   304→    
   305→    std::string toString() const {
   306→        std::stringstream ss;
   307→        
   308→        if (!hle_function_name.empty()) {
   309→            ss << hle_function_name;
   310→            if (!file_name.empty()) {
   311→                ss << " at " << file_name;
   312→                if (line_number > 0) {
   313→                    ss << ":" << line_number;
   314→                }
   315→            }
   316→        } else if (!function_name.empty()) {
   317→            ss << function_name << "()";
   318→            if (!file_name.empty()) {
   319→                ss << " at " << file_name << ":" << line_number;
   320→            }
   321→        } else if (guest_address != 0) {
   322→            ss << "0x" << std::hex << guest_address << std::dec;
   323→            if (!symbol_name.empty()) {
   324→                ss << " (" << symbol_name << ")";
   325→            }
   326→        } else {
   327→            ss << "unknown location";
   328→        }
   329→        
   330→        return ss.str();
   331→    }
   332→    
   333→    /**
   334→     * Create SourceLocation from current position (use macros)
   335→     */
   336→    static SourceLocation here(const std::string& file, int line, 
   337→                               const std::string& func) {
   338→        return {file, line, func, {}, {}, -1, 0, {}};
   339→    }
   340→};
   341→

   344→#define DEBUG_HLE(func_name, func_id) ::prosper_debug::SourceLocation{__FILE__, __LINE__, __func__, {}, (func_name), (func_id), 0, {}}
   345→#define DEBUG_GUEST(addr, sym) ::prosper_debug::SourceLocation{__FILE__, __LINE__, __func__, {}, {}, -1, (addr), (sym)}

   445→// ============================================================================
   446→// Core Evidence Structure
   447→// ============================================================================
   448→
   449→/**
   450→ * Single piece of evidence with full provenance
   451→ * 
   452→ * Every evidence item MUST answer:
   453→ * - WHEN did it happen? (timestamp)
   454→ * - WHO caused it? (source_location + subsystem)
   455→ * - WHAT happened? (type + data)
   456→ * - Why is this notable? (severity)
   457→ */
   458→struct Evidence {
   459→    // Unique identifier
   460→    std::string id;
   461→    
   462→    // When
   463→    DebugTimestamp timestamp;
   464→    
   465→    // Who
   466→    Subsystem subsystem{Subsystem::Unknown};
   467→    SourceLocation source;
   468→    std::string thread_id;
   469→    
   470→    // What
   471→    EvidenceType type{EvidenceType::LogMessage};
   472→    std::string description;
   473→    
   474→    // Data payload (JSON-serializable string)
   475→    std::string data_json;
   476→    
   477→    // Significance
   478→    Severity severity{Severity::Info};
   479→    
   480→    // Verification
   481→    bool verified{false};
   482→    std::string verification_method;
   483→    
   484→    // Relationships
   485→    std::string parent_id;       // Causal parent
   486→    std::vector<std::string> related_ids;
   487→    std::vector<std::string> refutes_ids;  // This evidence refutes these hypotheses
   488→    
   489→    // Metadata
   490→    std::map<std::string, std::string> tags;
   491→    std::map<std::string, std::string> metadata;
   492→    
   493→    // Generation
   494→    static std::string generateId() {
   495→        static uint64_t counter = 0;
   496→        auto now = std::chrono::high_resolution_clock::now().time_since_epoch().count();
   497→        return "evd_" + std::to_string(now) + "_" + std::to_string(++counter);
   498→    }
   499→    
   500→    Evidence() {
   501→        id = generateId();
   502→        timestamp = DebugTimestamp::now();
   503→    }
   504→    
   505→    /**
   506→     * Convert to map for JSON serialization
   507→     */
   508→    std::map<std::string, std::string> toMap() const {
   509→        return {
   510→            {"id", id},
   511→            {"timestamp", timestamp.toISO8601()},
   512→            {"frame", std::to_string(timestamp.frame_number)},
   513→            {"subsystem", subsystemToString(subsystem)},
   514→            {"source", source.toString()},
   515→            {"thread_id", thread_id},
   516→            {"type", evidenceTypeToString(type)},
   517→            {"description", description},
   518→            {"data", data_json},
   519→            {"severity", severityToString(severity)},
   520→            {"verified", verified ? "true" : "false"},
   521→            {"parent_id", parent_id}
   522→        };
   523→    }
   524→};

   527→// Evidence Collection
   528→// ============================================================================
   529→
   530→class EvidenceCollection {
   531→public:
   532→    void add(const Evidence& evidence) {
   533→        m_evidences[evidence.id] = evidence;
   534→        
   535→        // Update indexes
   536→        m_by_type[static_cast<size_t>(evidence.type)].push_back(evidence.id);
   537→        m_by_subsystem[static_cast<size_t>(evidence.subsystem)].push_back(evidence.id);
   538→        
   539→        if (evidence.severity >= Severity::Error) {
   540→            m_errors.push_back(evidence.id);
   541→        }
   542→    }
   543→    
   544→    bool remove(const std::string& id) {
   545→        auto it = m_evidences.find(id);
   546→        if (it == m_evidences.end()) return false;
   547→        
   548→        // Update indexes (slow but correct)
   549→        Evidence evd = it->second;
   550→        auto& type_vec = m_by_type[static_cast<size_t>(evd.type)];
   551→        type_vec.erase(std::remove(type_vec.begin(), type_vec.end(), id), type_vec.end());
   552→        
   553→        auto& sub_vec = m_by_subsystem[static_cast<size_t>(evd.subsystem)];
   554→        sub_vec.erase(std::remove(sub_vec.begin(), sub_vec.end(), id), sub_vec.end());
   555→        
   556→        m_errors.erase(std::remove(m_errors.begin(), m_errors.end(), id), m_errors.end());
   557→        
   558→        m_evidences.erase(it);
   559→        return true;
   560→    }
   561→    
   562→    std::optional<Evidence> find(const std::string& id) const {
   563→        auto it = m_evidences.find(id);
   564→        if (it != m_evidences.end()) return it->second;
   565→        return std::nullopt;
   566→    }
   567→    
   568→    std::vector<Evidence> findByType(EvidenceType type) const {
   569→        std::vector<Evidence> result;
   570→        for (const auto& id : m_by_type[static_cast<size_t>(type)]) {
   571→            if (auto evd = find(id)) result.push_back(*evd);
   572→        }
   573→        return result;
   574→    }
   575→    
   576→    std::vector<Evidence> findBySubsystem(Subsystem subs) const {
   577→        std::vector<Evidence> result;
   578→        for (const auto& id : m_by_subsystem[static_cast<size_t>(subs)]) {
   579→            if (auto evd = find(id)) result.push_back(*evd);
   580→        }
   581→        return result;
   582→    }
   583→    
   584→    std::vector<Evidence> findErrors() const {
   585→        std::vector<Evidence> result;
   586→        for (const auto& id : m_errors) {
   587→            if (auto evd = find(id)) result.push_back(*evd);
   588→        }
   589→        return result;
   590→    }
   591→    
   592→    std::vector<Evidence> findByTimeRange(DebugTimestamp start, DebugTimestamp end) const {
   593→        std::vector<Evidence> result;
   594→        for (const auto& [id, evd] : m_evidences) {
   595→            if (evd.timestamp >= start && evd.timestamp <= end) {
   596→                result.push_back(evd);
   597→            }
   598→        }
   599→        return result;
   600→    }
   601→    
   602→    std::vector<Evidence> all() const {
   603→        std::vector<Evidence> result;
   604→        result.reserve(m_evidences.size());
   605→        for (const auto& [id, evd] : m_evidences) {
   606→            result.push_back(evd);
   607→        }
   608→        return result;
   609→    }
   610→    
   611→    size_t size() const { return m_evidences.size(); }
   612→    bool empty() const { return m_evidences.empty(); }
   613→    
   614→    void clear() {
   615→        m_evidences.clear();
   616→        for (auto& vec : m_by_type) vec.clear();
   617→        for (auto& vec : m_by_subsystem) vec.clear();
   618→        m_errors.clear();
   619→    }
   620→
   621→private:
   622→    std::map<std::string, Evidence> m_evidences;
   623→

   659→// ============================================================================
   660→// Hypothesis Structure
   661→// ============================================================================
   662→
   663→struct Hypothesis {
   664→    std::string id;
   665→    std::string title;
   666→    std::string description;
   667→    InvestigationStatus status{InvestigationStatus::Open};
   668→    
   669→    // Timing
   670→    DebugTimestamp created_at;
   671→    DebugTimestamp updated_at;
   672→    DebugTimestamp confirmed_at;
   673→    DebugTimestamp rejected_at;
   674→    
   675→    // Evidence relationships
   676→    std::vector<std::string> supporting_evidence_ids;
   677→    std::vector<std::string> refuting_evidence_ids;
   678→    
   679→    // Confidence scoring
   680→    double confidence_score{0.0};  // 0.0 to 1.0
   681→    std::string confidence_reasoning;
   682→    
   683→    // Relationships
   684→    std::vector<std::string> related_hypothesis_ids;
   685→    std::string superseded_by_id;
   686→    
   687→    // Notes and findings
   688→    std::string notes;
   689→    std::vector<std::string> rejection_reasons;
   690→    
   691→    // Root cause chain (if confirmed)
   692→    std::vector<std::string> causal_chain_evidence_ids;
   693→    
   694→    static std::string generateId() {
   695→        static uint64_t counter = 0;
   696→        auto now = std::chrono::high_resolution_clock::now().time_since_epoch().count();
   697→        return "hyp_" + std::to_string(now) + "_" + std::to_string(++counter);
   698→    }
   699→    
   700→    Hypothesis() {
   701→        id = generateId();
   702→        created_at = DebugTimestamp::now();
   703→        updated_at = created_at;
   704→    }
   705→    
   706→    void addSupportingEvidence(const std::string& evidence_id) {
   707→        if (std::find(supporting_evidence_ids.begin(), supporting_evidence_ids.end(), evidence_id) 
   708→            == supporting_evidence_ids.end()) {
   709→            supporting_evidence_ids.push_back(evidence_id);
   710→            updated_at = DebugTimestamp::now();
   711→            updateConfidence();
   712→        }
   713→    }
   714→    
   715→    void addRefutingEvidence(const std::string& evidence_id) {
   716→        if (std::find(refuting_evidence_ids.begin(), refuting_evidence_ids.end(), evidence_id) 
   717→            == refuting_evidence_ids.end()) {
   718→            refuting_evidence_ids.push_back(evidence_id);
   719→            updated_at = DebugTimestamp::now();
   720→            updateConfidence();
   721→        }
   722→    }
   723→    
   724→    void confirm() {
   725→        status = InvestigationStatus::Confirmed;
   726→        confirmed_at = DebugTimestamp::now();
   727→        updated_at = confirmed_at;
   728→        confidence_score = 1.0;
   729→        confidence_reasoning = "Confirmed by direct evidence";
   730→    }
   731→    
   732→    void reject(const std::string& reason = "") {
   733→        status = InvestigationStatus::Rejected;
   734→        rejected_at = DebugTimestamp::now();
   735→        updated_at = rejected_at;
   736→        confidence_score = 0.0;
   737→        if (!reason.empty()) {
   738→            rejection_reasons.push_back(reason);
   739→        }
   740→    }
   741→    
   742→private:
   743→    void updateConfidence() {
   744→        // Simple Bayesian-like update
   745→        double support = supporting_evidence_ids.size();
   746→        double refute = refuting_evidence_ids.size();
   747→        double total = support + refute;
   748→        
   749→        if (total > 0) {
   750→            confidence_score = support / total;
   751→        }
   752→        
   753→        confidence_reasoning = std::to_string(supporting_evidence_ids.size()) + 
   754→                              " supporting, " + 
   755→                              std::to_string(refuting_evidence_ids.size()) + 
   756→                              " refuting";
   757→    }
   758→};

   760→// ============================================================================
   761→// JSON Utilities (Lightweight - No External Dependencies)
   762→// ============================================================================
   763→
   764→class JsonUtils {
   765→public:
   766→    static std::string escapeJson(const std::string& input) {
   767→        std::string output;
   768→        output.reserve(input.length() * 2);
   769→        for (char c : input) {
   770→            switch (c) {
   771→                case '"': output += "\\\""; break;
   772→                case '\\': output += "\\\\"; break;
   773→                case '\b': output += "\\b"; break;
   774→                case '\f': output += "\\f"; break;
   775→                case '\n': output += "\\n"; break;
   776→                case '\r': output += "\\r"; break;
   777→                case '\t': output += "\\t"; break;
   778→                default:
   779→                    if ('\x00' <= c && c <= '\x1f') {
   780→                        char buf[8];
   781→                        snprintf(buf, sizeof(buf), "\\u%04x", (unsigned)(unsigned char)c);
   782→                        output += buf;
   783→                    } else {
   784→                        output += c;
   785→                    }
   786→            }
   787→        }
   788→        return output;
   789→    }
   790→    
   791→    static std::string unescapeJson(const std::string& input) {
   792→        std::string output;
   793→        output.reserve(input.length());
   794→        
   795→        for (size_t i = 0; i < input.length(); ++i) {
   796→            if (input[i] == '\\' && i + 1 < input.length()) {
   797→                switch (input[i + 1]) {
   798→                    case '"': output += '"'; i++; break;
   799→                    case '\\': output += '\\'; i++; break;
   800→                    case 'b': output += '\b'; i++; break;
   801→                    case 'f': output += '\f'; i++; break;
   802→                    case 'n': output += '\n'; i++; break;
   803→                    case 'r': output += '\r'; i++; break;
   804→                    case 't': output += '\t'; i++; break;
   805→                    case 'u':
   806→                        if (i + 5 < input.length()) {
   807→                            std::string hex = input.substr(i + 2, 4);
   808→                            unsigned int codepoint;
   809→                            std::istringstream(hex) >> std::hex >> codepoint;
   810→                            output += static_cast<char>(codepoint);
   811→                            i += 5;
   812→                        }
   813→                        break;
   814→                    default:
   815→                        output += input[i];
   816→                }
   817→            } else {
   818→                output += input[i];
   819→            }
   820→        }
   821→        
   822→        return output;
   823→    }
   824→};

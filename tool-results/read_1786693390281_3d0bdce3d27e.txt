    88→    /**
    89→     * Generate complete root cause report from experiment record
    90→     */
    91→    RootCauseReport generateReport(const ExperimentRecord& record) const {
    92→        RootCauseReport report;
    93→        
    94→        report.experiment_id = record.id;
    95→        report.title = record.title.empty() ? "Debug Investigation" : record.title;
    96→        report.generated_at = Timestamp::now();
    97→        
    98→        // Build timeline from evidence and hypotheses
    99→        buildTimeline(record, report);
   100→        
   101→        // Collect all hypotheses
   102→        collectHypotheses(record, report);
   103→        
   104→        // Identify key evidence (high severity or attached to confirmed hypothesis)
   105→        identifyKeyEvidence(record, report);
   106→        
   107→        // Determine root cause if we have a confirmed hypothesis
   108→        determineRootCause(report);
   109→        
   110→        // Generate recommendations
   111→        generateRecommendations(record, report);
   112→        
   113→        // Calculate metadata
   114→        calculateStats(record, report);
   115→        
   116→        return report;
   117→    }

   119→    /**
   120→     * Generate report as formatted text (for CLI output)
   121→     */
   122→    std::string generateTextReport(const RootCauseReport& report) const {
   123→        std::stringstream output;
   124→        
   125→        // Header
   126→        output << "=" << std::string(78, '=') << "\n";
   127→        output << "  ROOT CAUSE ANALYSIS REPORT\n";
   128→        output << "  Experiment: " << report.experiment_id << "\n";
   129→        output << "  Generated: " << report.generated_at << "\n";
   130→        output << "=" << std::string(78, '=') << "\n\n";
   131→        
   132→        // Executive Summary
   133→        output << "EXECUTIVE SUMMARY\n";
   134→        output << "-" << std::string(40, '-') << "\n";
   135→        output << report.executive_summary << "\n\n";
   136→        
   137→        // Root Cause
   138→        if (!report.root_cause_description.empty()) {
   139→            output << "ROOT CAUSE\n";
   140→            output << "-" << std::string(40, '-') << "\n";
   141→            output << report.root_cause_description << "\n";
   142→            output << "Confidence Level: " << std::fixed << std::setprecision(1) 
   143→                   << (report.confidence_level * 100.0) << "%\n";
   144→            output << "Confirmed Hypothesis ID: " << report.confirmed_hypothesis_id << "\n\n";
   145→        }
   146→        
   147→        // Timeline
   148→        if (m_config.include_timeline && !report.timeline.empty()) {
   149→            output << "INVESTIGATION TIMELINE\n";
   150→            output << "-" << std::string(40, '-') << "\n";
   151→            
   152→            for (const auto& event : report.timeline) {
   153→                output << "[" << event.timestamp << "] ";
   154→                output << "[" << event.event_type << "] ";
   155→                output << event.description << "\n";
   156→                
   157→                if (m_config.ai_friendly_format && !event.metadata.empty()) {
   158→                    for (const auto& [k, v] : event.metadata) {
   159→                        output << "  " << k << ": " << truncateString(v, m_config.max_evidence_detail_length) << "\n";
   160→                    }
   161→                }
   162→            }
   163→            output << "\n";
   164→        }
   165→        
   166→        // Key Evidence
   167→        if (!report.key_evidence.empty()) {
   168→            output << "KEY EVIDENCE\n";
   169→            output << "-" << std::string(40, '-') << "\n";
   170→            
   171→            for (size_t i = 0; i < report.key_evidence.size(); ++i) {
   172→                const auto& evd = report.key_evidence[i];
   173→                output << i + 1 << ". [" << Evidence::severityToString(evd.severity) << "] " 
   174→                       << evd.description << "\n";
   175→                
   176→                if (m_config.include_full_evidence) {
   177→                    output << "   Content: " << truncateString(evd.content, m_config.max_evidence_detail_length) << "\n";
   178→                }
   179→                output << "   Source: " << evd.source_path << "\n";
   180→                output << "   Verified: " << (evd.verified ? "Yes" : "No") << "\n";
   181→                output << "\n";
   182→            }
   183→        }
   184→        
   185→        // Confirmed Hypothesis Details
   186→        if (!report.all_hypotheses.empty()) {
   187→            output << "HYPOTHESIS ANALYSIS\n";
   188→            output << "-" << std::string(40, '-') << "\n";
   189→            
   190→            for (const auto& hyp : report.all_hypotheses) {
   191→                if (hyp.status == InvestigationStatus::Confirmed) {
   192→                    output << "[CONFIRMED] " << hyp.title << "\n";
   193→                    output << "  Description: " << hyp.description << "\n";
   194→                    output << "  Confidence: " << std::fixed << std::setprecision(1) 
   195→                           << (hyp.confidence_score * 100.0) << "%\n";
   196→                    
   197→                    if (!hyp.supporting_evidence_ids.empty()) {
   198→                        output << "  Supporting Evidence: " << hyp.supporting_evidence_ids.size() << " items\n";
   199→                    }
   200→                    output << "\n";
   201→                }
   202→            }
   203→        }
   204→        
   205→        // Rejected Hypotheses
   206→        if (m_config.include_rejected && !report.rejected_hypotheses.empty()) {
   207→            output << "REJECTED HYPOTHESES\n";
   208→            output << "-" << std::string(40, '-') << "\n";
   209→            
   210→            for (const auto& hyp : report.rejected_hypotheses) {
   211→                output << "[REJECTED] " << hyp.title << "\n";
   212→                output << "  Reason: " << (hyp.notes.empty() ? "Not specified" : hyp.notes) << "\n";
   213→                
   214→                if (!hyp.refuting_evidence_ids.empty()) {
   215→                    output << "  Refuting Evidence: " << hyp.refuting_evidence_ids.size() << " items\n";
   216→                }
   217→                output << "\n";
   218→            }
   219→        }
   220→        
   221→        // Recommendations
   222→        if (m_config.include_recommendations && !report.recommended_fix.empty()) {
   223→            output << "RECOMMENDATIONS\n";
   224→            output << "-" << std::string(40, '-') << "\n";
   225→            output << report.recommended_fix << "\n\n";
   226→            
   227→            if (!report.verification_steps.empty()) {
   228→                output << "VERIFICATION STEPS:\n";
   229→                output << report.verification_steps << "\n\n";
   230→            }
   231→        }
   232→        
   233→        // Lessons Learned
   234→        if (!report.lessons_learned.empty()) {
   235→            output << "LESSONS LEARNED\n";
   236→            output << "-" << std::string(40, '-') << "\n";
   237→            output << report.lessons_learned << "\n\n";
   238→        }
   239→        
   240→        // Statistics
   241→        if (report.metadata.count("stats")) {
   242→            output << "STATISTICS\n";
   243→            output << "-" << std::string(40, '-') << "\n";
   244→            output << report.metadata.at("stats") << "\n";
   245→        }
   246→        
   247→        // Footer
   248→        output << "=" << std::string(78, '=') << "\n";
   249→        output << "End of Report | Debug Intelligence Layer v" << VERSION << "\n";
   250→        
   251→        return output.str();
   252→    }

   425→    void buildTimeline(const ExperimentRecord& record, RootCauseReport& report) const {
   426→        // Add creation event
   427→        TimelineEvent create_evt;
   428→        create_evt.timestamp = record.created_at;
   429→        create_evt.event_type = "experiment_started";
   430→        create_evt.description = "Experiment started: " + record.title;
   431→        create_evt.reference_id = record.id;
   432→        report.timeline.push_back(create_evt);
   433→        
   434→        // Add evidence events
   435→        for (const auto& evd : record.evidences.all()) {
   436→            report.timeline.push_back(TimelineEvent::evidenceAdded(evd));
   437→        }
   438→        
   439→        // Add hypothesis events
   440→        for (const auto& hyp : record.hypotheses.all()) {
   441→            report.timeline.push_back(TimelineEvent::hypothesisCreated(hyp));
   442→            
   443→            if (hyp.status == InvestigationStatus::Confirmed) {
   444→                report.timeline.push_back(TimelineEvent::hypothesisConfirmed(hyp));
   445→            } else if (hyp.status == InvestigationStatus::Rejected) {
   446→                report.timeline.push_back(TimelineEvent::hypothesisRejected(hyp));
   447→            }
   448→        }
   449→        
   450→        // Sort by timestamp
   451→        std::sort(report.timeline.begin(), report.timeline.end(),
   452→                 [](const TimelineEvent& a, const TimelineEvent& b) {
   453→                     return a.timestamp < b.timestamp;
   454→                 });
   455→    }
   456→    
   457→    void collectHypotheses(const ExperimentRecord& record, RootCauseReport& report) const {
   458→        report.all_hypotheses = record.hypotheses.all();
   459→        
   460→        // Separate rejected
   461→        for (const auto& hyp : report.all_hypotheses) {
   462→            if (hyp.status == InvestigationStatus::Rejected || 
   463→                hyp.status == InvestigationStatus::Superseded) {
   464→                report.rejected_hypotheses.push_back(hyp);
   465→            }
   466→        }
   467→    }
   468→    
   469→    void identifyKeyEvidence(const ExperimentRecord& record, RootCauseReport& report) const {
   470→        // Get high-severity evidence
   471→        auto all_evd = record.evidences.all();
   472→        
   473→        // First pass: critical and error severity
   474→        for (const auto& evd : all_evd) {
   475→            if (evd.severity == Severity::Critical || evd.severity == Severity::Error) {
   476→                report.key_evidence.push_back(evd);
   477→            }
   478→        }
   479→        
   480→        // Second pass: evidence attached to confirmed hypotheses
   481→        for (const auto& hyp : record.hypotheses.getConfirmed()) {
   482→            for (const auto& evd_id : hyp.supporting_evidence_ids) {
   483→                auto evd = record.evidences.find(evd_id);
   484→                if (evd) {
   485→                    // Check if already added
   486→                    bool found = false;
   487→                    for (const auto& existing : report.key_evidence) {
   488→                        if (existing.id == evd->id) {
   489→                            found = true;
   490→                            break;
   491→                        }
   492→                    }
   493→                    if (!found) {
   494→                        report.key_evidence.push_back(*evd);
   495→                    }
   496→                }
   497→            }
   498→        }
   499→        
   500→        // If no key evidence yet, take some warnings too
   501→        if (report.key_evidence.empty()) {
   502→            for (const auto& evd : all_evd) {
   503→                if (evd.severity >= Severity::Warning) {
   504→                    report.key_evidence.push_back(evd);
   505→                    if (report.key_evidence.size() >= 5) break;  // Limit
   506→                }
   507→            }
   508→        }
   509→    }

   511→    void determineRootCause(RootCauseReport& report) const {
   512→        auto confirmed = [&]() -> std::optional<Hypothesis> {
   513→            for (const auto& hyp : report.all_hypotheses) {
   514→                if (hyp.status == InvestigationStatus::Confirmed) {
   515→                    return hyp;
   516→                }
   517→            }
   518→            return std::nullopt;
   519→        }();
   520→        
   521→        if (confirmed) {
   522→            report.root_cause_description = confirmed->description.empty() 
   523→                ? confirmed->title 
   524→                : confirmed->description;
   525→            report.confirmed_hypothesis_id = confirmed->id;
   526→            report.confidence_level = confirmed->confidence_score;
   527→            
   528→            // Build executive summary
   529→            report.executive_summary = "Root cause identified: " + confirmed->title + ". " +
   530→                "Investigation completed with " + std::to_string(report.all_hypotheses.size()) + 
   531→                " hypotheses tested, " + std::to_string(report.rejected_hypotheses.size()) + " rejected.";
   532→        } else if (!report.all_hypotheses.empty()) {
   533→            // No confirmed hypothesis - report status
   534→            report.executive_summary = "Investigation in progress: " + 
   535→                std::to_string(report.all_hypotheses.size()) + " hypotheses under consideration. " +
   536→                "No root cause confirmed yet.";
   537→            report.root_cause_description = "Under investigation";
   538→            report.confidence_level = 0.0;
   539→        } else {
   540→            report.executive_summary = "Experiment recorded but no hypotheses proposed yet.";
   541→            report.root_cause_description = "Not determined";
   542→        }
   543→    }
   544→    
   545→    void generateRecommendations(const ExperimentRecord& record, RootCauseReport& report) const {
   546→        // Find the confirmed hypothesis to base recommendations on
   547→        for (const auto& hyp : record.hypotheses.getConfirmed()) {
   548→            // Extract potential fix from hypothesis description/notes
   549→            if (!hyp.notes.empty()) {
   550→                report.recommended_fix = hyp.notes;
   551→            } else {
   552→                report.recommended_fix = "Address issue: " + hyp.title;
   553→            }
   554→            
   555→            // Generic verification steps
   556→            report.verification_steps = buildVerificationSteps(hyp);
   557→            break;
   558→        }
   559→        
   560→        // Lessons learned template
   561→        if (!report.rejected_hypotheses.empty()) {
   562→            std::stringstream lessons;
   563→            lessons << "Investigation ruled out " << report.rejected_hypotheses.size() 
   564→                    << " incorrect hypotheses before finding root cause.\n";
   565→            
   566→            // List what was learned from rejections
   567→            for (const auto& rej : report.rejected_hypotheses) {
   568→                if (!rej.notes.empty()) {
   569→                    lessons << "- " << rej.notes << "\n";
   570→                }
   571→            }
   572→            
   573→            report.lessons_learned = lessons.str();
   574→        }
   575→    }
   576→    
   577→    std::string buildVerificationSteps(const Hypothesis& hyp) const {
   578→        std::stringstream steps;
   579→        steps << "1. Apply fix addressing: " << hyp.title << "\n";
   580→        steps << "2. Rebuild with configuration matching this experiment\n";
   581→        steps << "3. Reproduce original test case\n";
   582→        steps << "4. Verify issue is resolved\n";
   583→        steps << "5. Run regression tests\n";
   584→        steps << "6. Document findings in EXP package\n";
   585→        return steps.str();
   586→    }

   588→    void calculateStats(const ExperimentRecord& record, RootCauseReport& report) const {
   589→        ReportStats stats = getStats(report);
   590→        report.metadata["stats"] = stats.toString();
   591→        report.metadata["generator_version"] = VERSION;
   592→        report.metadata["total_evidence"] = std::to_string(record.evidences.size());
   593→        report.metadata["total_hypotheses"] = std::to_string(record.hypotheses.size());
   594→    }

   596→    int countByStatus(const std::vector<Hypothesis>& hypotheses, InvestigationStatus status) const {
   597→        return std::count_if(hypotheses.begin(), hypotheses.end(),
   598→                            [status](const Hypothesis& h) { return h.status == status; });
   599→    }

   601→    std::string truncateString(const std::string& str, size_t max_len) const {
   602→        if (str.length() <= max_len) return str;
   603→        return str.substr(0, max_len) + "...[truncated]";
   604→    }

   606→    std::string escapeJson(const std::string& input) const {
   607→        std::string output;
   608→        output.reserve(input.length() * 2);
   609→        for (char c : input) {
   610→            switch (c) {
   611→                case '"': output += "\\\""; break;
   612→                case '\\': output += "\\\\"; break;
   613→                case '\b': output += "\\b"; break;
   614→                case '\f': output += "\\f"; break;
   615→                case '\n': output += "\\n"; break;
   616→                case '\r': output += "\\r"; break;
   617→                case '\t': output += "\\t"; break;
   618→                default:
   619→                    if ('\x00' <= c && c <= '\x1f') {
   620→                        char buf[8];
   621→                        snprintf(buf, sizeof(buf), "\\u%04x", (unsigned)c);
   622→                        output += buf;
   623→                    } else {
   624→                        output += c;
   625→                    }
   626→            }
   627→        }
   628→        return output;
   629→    }

    33→enum class ExperimentStatus : uint8_t {
    34→    Running,
    35→    Completed,
    36→    Failed,
    37→    Abandoned,
    38→    Superseded

    41→inline std::string experimentStatusToString(ExperimentStatus status) {
    42→    switch (status) {
    43→        case ExperimentStatus::Running: return "Running";
    44→        case ExperimentStatus::Completed: return "Completed";
    45→        case ExperimentStatus::Failed: return "Failed";
    46→        case ExperimentStatus::Abandoned: return "Abandoned";
    47→        case ExperimentStatus::Superseded: return "Superseded";
    48→    }
    49→    return "Unknown";
    50→}

    73→    std::string result_summary;

    90→    static std::string generateId() {

   108→    
   109→    void markFailed(const std::string& reason = "") {
   110→        status = ExperimentStatus::Failed;
   111→        updated_at = DebugTimestamp::now();
   112→        if (!reason.empty()) result_summary = "Failed: " + reason;
   113→    }
   114→    
   115→    void markAbandoned() {
   116→        status = ExperimentStatus::Abandoned;
   117→        updated_at = DebugTimestamp::now();
   118→    }

   121→        std::stringstream json;
   122→        json << "{\n";
   123→        json << "  \"id\": \"" << JsonUtils::escapeJson(id) << "\",\n";
   124→        json << "  \"title\": \"" << JsonUtils::escapeJson(title) << "\",\n";
   125→        json << "  \"status\": \"" << experimentStatusToString(status) << "\",\n";
   126→        json << "  \"created\": \"" << created_at.toISO8601() << "\",\n";
   127→        if (!commit_hash.empty()) {
   128→            json << "  \"commit\": \"" << JsonUtils::escapeJson(commit_hash) << "\",\n";
   129→        }
   130→        if (!result_summary.empty()) {
   131→            json << "  \"result\": \"" << JsonUtils::escapeJson(result_summary) << "\",\n";
   132→        }
   133→        if (root_cause_found) {
   134→            json << "  \"root_cause\": \"" << JsonUtils::escapeJson(root_cause) << "\"\n";

   219→struct UpstreamFixRecord {
   220→    std::string id;

   234→    std::string merged_in_version;

   236→    
   237→    // Applicability

   240→    std::vector<std::string> related_issues;
   241→    
   242→    // Detection metadata
   243→    DebugTimestamp added_to_database;
   244→    std::string source;  // Manual, automated, etc.
   245→    
   246→    /**
   247→     * Check relevance to a query
   248→     */
   249→    double relevanceToQuery(const std::string& query) const {
   250→        std::string combined = pr_title + " " + description + " " + fixed_issue_description + " ";
   251→        for (const auto& kw : keywords) combined += kw + " ";
   252→        for (const auto& comp : affected_components) combined += comp + " ";
   253→        
   254→        std::string lower_query = query;
   255→        std::transform(lower_query.begin(), lower_query.end(), lower_query.begin(), ::tolower);
   256→        
   257→        std::string lower_combined = combined;
   258→        std::transform(lower_combined.begin(), lower_combined.end(), lower_combined.begin(), ::tolower);
   259→        
   260→        // Direct match
   261→        if (lower_combined.find(lower_query) != std::string::npos) {
   262→            return 0.95;
   263→        }
   264→        
   265→        // Word-level matching
   266→        std::vector<std::string> query_words = splitWords(lower_query);
   267→        int matches = 0;
   268→        for (const auto& word : query_words) {
   269→            if (lower_combined.find(word) != std::string::npos) {
   270→                matches++;
   271→            }
   272→        }
   273→        
   274→        if (!query_words.empty()) {
   275→            return static_cast<double>(matches) / query_words.size() * 0.8;
   276→        }
   277→        
   278→        return 0.0;
   279→    }
   280→
   281→private:
   282→    static std::vector<std::string> splitWords(const std::string& text) {
   283→        std::vector<std::string> words;
   284→        std::stringstream ss(text);
   285→        std::string word;
   286→        while (ss >> word) {
   287→            if (word.length() > 2) words.push_back(word);
   288→        }
   289→        return words;
   290→    }
   291→};

   572→    std::vector<UpstreamFixRecord> findRelevantUpstreamFixes(
   573→        const std::string& query,
   574→        double min_relevance = 0.5,
   575→        size_t limit = 10) const {
   576→        
   577→        std::shared_lock lock(m_mutex);
   578→        
   579→        std::vector<std::pair<double, UpstreamFixRecord>> scored;
   580→        
   581→        for (const auto& [id, fix] : m_upstream_fixes) {
   582→            double relevance = fix.relevanceToQuery(query);
   583→            if (relevance >= min_relevance) {
   584→                scored.push_back({relevance, fix});
   585→            }
   586→        }
   587→        
   588→        // Sort by relevance
   589→        std::sort(scored.begin(), scored.end(),
   590→            [](const auto& a, const auto& b) { return a.first > b.first; });
   591→        
   592→        // Extract top results
   593→        std::vector<UpstreamFixRecord> result;
   594→        for (size_t i = 0; i < scored.size() && i < limit; ++i) {
   595→            result.push_back(scored[i].second);
   596→        }
   597→        
   598→        return result;
   599→    }

   605→    struct SearchQuery {
   606→        std::string text;
   607→        std::vector<std::string> tags;
   608→        std::optional<ExperimentStatus> status_filter;
   609→        std::optional<InvestigationStatus> hypothesis_status_filter;
   610→        int max_results{20};
   611→        double min_relevance{0.3};
   612→        
   613→        static SearchQuery forText(const std::string& text, int max_results = 20) {
   614→            return {text, {}, std::nullopt, std::nullopt, max_results, 0.3};
   615→        }
   616→        
   617→        static SearchQuery forTags(const std::vector<std::string>& tags) {
   618→            return {"", tags, std::nullopt, std::nullopt, 50, 0.0};
   619→        }
   620→    };
   621→    
   622→    /**
   623→     * Search across all data
   624→     */
   625→    std::vector<SearchResult> search(const SearchQuery& query) const {
   626→        std::vector<SearchResult> results;
   627→        
   628→        // Search experiments
   629→        {
   630→            std::shared_lock lock(m_mutex);
   631→            
   632→            for (const auto& [id, exp] : m_experiments) {
   633→                double score = 0.0;
   634→                std::string matched;
   635→                
   636→                if (!query.text.empty()) {
   637→                    score = calculateTextSimilarity(query.text, exp.title + " " + exp.description);
   638→                    if (score > 0) matched = "text";
   639→                }
   640→                
   641→                // Tag matching boosts score
   642→                for (const auto& tag : query.tags) {
   643→                    auto it = std::find(exp.tags.begin(), exp.tags.end(), tag);
   644→                    if (it != exp.tags.end()) {
   645→                        score = std::max(score, 0.8);
   646→                        if (!matched.empty()) matched += "+tag:" + tag;
   647→                        else matched = "tag:" + tag;
   648→                    }
   649→                }
   650→                
   651→                // Status filter
   652→                if (query.status_filter.has_value() && exp.status != *query.status_filter) {
   653→                    score = 0.0;
   654→                }
   655→                
   656→                if (score >= query.min_relevance) {
   657→                    SearchResult sr;
   658→                    sr.relevance_score = score;
   659→                    sr.matched_fields = matched;
   660→                    sr.experiment = exp;
   661→                    results.push_back(sr);
   662→                }
   663→            }
   664→        }
   665→        
   666→        // Sort by relevance
   667→        std::sort(results.begin(), results.end(), std::greater<SearchResult>());
   668→        
   669→        // Limit results
   670→        if (results.size() > static_cast<size_t>(query.max_results)) {
   671→            results.resize(query.max_results);
   672→        }
   673→        
   674→        return results;
   675→    }
   676→    
   677→    /**
   678→     * Simple text search
   679→     */
   680→    std::vector<SearchResult> searchText(const std::string& query, int max_results = 20) const {
   681→        return search(SearchQuery::forText(query, max_results));
   682→    }

   688→    /**
   689→     * Check if a new experiment would duplicate existing work
   690→     */
   691→    std::vector<DuplicateWarning> checkForDuplicates(
   692→        const std::string& title,
   693→        const std::vector<std::string>& tags = {},
   694→        double threshold = 0.7) const {
   695→        
   696→        std::vector<DuplicateWarning> warnings;
   697→        
   698→        std::shared_lock lock(m_mutex);
   699→        
   700→        for (const auto& [id, exp] : m_experiments) {
   701→            double similarity = calculateTextSimilarity(title, exp.title);
   702→            
   703→            // Boost for matching tags
   704→            if (!tags.empty()) {
   705→                size_t matching_tags = 0;
   706→                for (const auto& tag : tags) {
   707→                    if (std::find(exp.tags.begin(), exp.tags.end(), tag) != exp.tags.end()) {
   708→                        matching_tags++;
   709→                    }
   710→                }
   711→                
   712→                if (!exp.tags.empty()) {
   713→                    double tag_ratio = static_cast<double>(matching_tags) / tags.size();
   714→                    similarity = std::max(similarity, tag_ratio * 0.8);
   715→                }
   716→            }
   717→            
   718→            if (similarity >= threshold) {
   719→                DuplicateWarning warn;
   720→                warn.existing_id = id;
   721→                warn.existing_title = exp.title;
   722→                warn.existing_status = exp.status;
   723→                warn.similarity_score = similarity;
   724→                
   725→                if (similarity >= 0.9) {
   726→                    warn.reason = "Near-exact match to existing investigation";
   727→                    warn.recommendation = "Review existing experiment before proceeding.";
   728→                } else if (exp.status == ExperimentStatus::Completed) {
   729→                    warn.reason = "Similar to completed investigation";
   730→                    warn.recommendation = "Check if this experiment already solved your issue.";
   731→                } else if (exp.status == ExperimentStatus::Running) {
   732→                    warn.reason = "Similar investigation currently in progress";
   733→                    warn.recommendation = "Coordinate with ongoing investigation or wait for results.";
   734→                } else {
   735→                    warn.reason = "Similar topic previously investigated";
   736→                    warn.recommendation = "Review findings before starting new work.";
   737→                }
   738→                
   739→                warnings.push_back(warn);
   740→            }
   741→        }
   742→        
   743→        return warnings;
   744→    }
   745→    
   746→    /**
   747→     * Quick duplicate check
   748→     */
   749→    std::pair<bool, std::optional<DuplicateWarning>> isDuplicate(
   750→        const std::string& title,
   751→        const std::vector<std::string>& tags = {}) const {
   752→        
   753→        auto warnings = checkForDuplicates(title, tags, 0.8);
   754→        
   755→        if (warnings.empty()) {
   756→            return {false, std::nullopt};
   757→        }
   758→        
   759→        // Return highest similarity
   760→        auto max_it = std::max_element(warnings.begin(), warnings.end(),
   761→            [](const DuplicateWarning& a, const DuplicateWarning& b) {
   762→                return a.similarity_score < b.similarity_score;
   763→            });
   764→        
   765→        return {true, *max_it};
   766→    }

   854→    double calculateTextSimilarity(const std::string& a, const std::string& b) const {
   855→        // Bigram overlap similarity
   856→        auto bigrams = [](const std::string& s) -> std::set<std::string> {
   857→            std::set<std::string> result;
   858→            std::string lower = s;
   859→            std::transform(lower.begin(), lower.end(), lower.begin(), ::tolower);
   860→            
   861→            for (size_t i = 0; i + 1 < lower.length(); ++i) {
   862→                result.insert(lower.substr(i, 2));
   863→            }
   864→            return result;
   865→        };
   866→        
   867→        auto bgA = bigrams(a);
   868→        auto bgB = bigrams(b);
   869→        
   870→        if (bgA.empty() && bgB.empty()) return 1.0;
   871→        if (bgA.empty() || bgB.empty()) return 0.0;
   872→        
   873→        std::set<std::string> intersection;
   874→        std::set_intersection(bgA.begin(), bgA.end(), bgB.begin(), bgB.end(),
   875→                             std::inserter(intersection, intersection.begin()));
   876→        
   877→        std::set<std::string> union_set;
   878→        std::union(bgA.begin(), bgA.end(), bgB.begin(), bgB.end(),
   879→                  std::inserter(union_set, union_set.begin()));
   880→        
   881→        return static_cast<double>(intersection.size()) / union_set.size();
   882→    }

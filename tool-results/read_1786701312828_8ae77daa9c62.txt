     1→/**
     2→ * State Timeline System - Priority 3
     3→ * 
     4→ * PROBLEM SOLVED:
     5→ * Crash location is usually not corruption location.
     6→ * Need historical state to trace back to root cause.
     7→ * 
     8→ * EXAMPLE:
     9→ *   Current: "Crash at frame 1300"
    10→ *   Required: 
    11→ *     Object created:  frame 200
    12→ *     Modified:        frame 600
    13→ *     Invalidated:     frame 900
    14→ *     Used incorrectly: frame 1250
    15→ *     Crash:           frame 1300
    16→ * 
    17→ * DESIGN:
    18→ * - Records state changes with frame-level precision
    19→ * - Tracks object/resource lifecycles
    20→ * - Enables time-travel queries ("what was state at frame X?")
    21→ * - Supports multiple timeline streams (GPU, memory, HLE, etc.)
    22→ */

   144→// ============================================================================
   145→// Timeline Event
   146→// ============================================================================
   147→
   148→struct TimelineEvent {
   149→    std::string id;
   150→    DebugTimestamp timestamp;
   151→    uint64_t frame_number{0};
   152→    TimelineEventType type;
   153→    
   154→    // What object/state?
   155→    std::string object_id;       // Unique identifier for the object
   156→    std::string object_type;     // Class/type of object
   157→    std::string object_description;  // Human-readable description
   158→    
   159→    // What changed?
   160→    std::string attribute_name;   // Which attribute changed
   161→    std::string old_value;        // Previous value (JSON string)
   162→    std::string new_value;        // New value (JSON string)
   163→    
   164→    // Provenance
   165→    Subsystem subsystem{Subsystem::Unknown};
   166→    SourceLocation source;
   167→    std::string thread_id;
   168→    
   169→    // Additional data
   170→    std::map<std::string, std::string> metadata;
   171→    
   172→    // Relationships
   173→    std::string parent_event_id;  // Causal parent
   174→    std::vector<std::string> related_event_ids;
   175→    
   176→    static std::string generateId() {
   177→        static std::atomic<uint64_t> counter{0};
   178→        auto now = std::chrono::high_resolution_clock::now().time_since_epoch().count();
   179→        return "tle_" + std::to_string(now) << "_" << std::to_string(++counter);
   180→    }
   181→    
   182→    TimelineEvent() {
   183→        id = generateId();
   184→        timestamp = DebugTimestamp::now();
   185→    }
   186→    
   187→    bool isErrorOrCorruption() const {
   188→        return type == TimelineEventType::ErrorDetected ||
   189→               type == TimelineEventType::CorruptionDetected ||
   190→               type == TimelineEventType::AssertionFailed ||
   191→               type == TimelineEventType::AnomalyDetected;
   192→    }
   193→    
   194→    bool isLifecycleEvent() const {
   195→        return type == TimelineEventType::ObjectCreated ||
   196→               type == TimelineEventType::ObjectDestroyed ||
   197→               type == TimelineEventType::ResourceAllocated ||
   198→               type == TimelineEventType::ResourceFreed;
   199→    }
   200→    
   201→    std::string toJson() const {
   202→        std::stringstream json;
   203→        json << "{\n";
   204→        json << "  \"id\": \"" << JsonUtils::escapeJson(id) << "\",\n";
   205→        json << "  \"timestamp\": \"" << timestamp.toISO8601() << "\",\n";
   206→        json << "  \"frame\": " << frame_number << ",\n";
   207→        json << "  \"type\": \"" << timelineEventTypeToString(type) << "\",\n";
   208→        json << "  \"object_id\": \"" << JsonUtils::escapeJson(object_id) << "\",\n";
   209→        json << "  \"object_type\": \"" << JsonUtils::escapeJson(object_type) << "\",\n";
   210→        if (!object_description.empty()) {
   211→            json << "  \"description\": \"" << JsonUtils::escapeJson(object_description) << "\",\n";
   212→        }
   213→        if (!attribute_name.empty()) {
   214→            json << "  \"attribute\": \"" << JsonUtils::escapeJson(attribute_name) << "\",\n";
   215→            json << "  \"old_value\": \"" << JsonUtils::escapeJson(old_value) << "\",\n";
   216→            json << "  \"new_value\": \"" << JsonUtils::escapeJson(new_value) << "\",\n";
   217→        }
   218→        json << "  \"subsystem\": \"" << subsystemToString(subsystem) << "\"\n";
   219→        json << "}";
   220→        return json.str();
   221→    }
   222→};

   224→// ============================================================================
   225→// Object Lifecycle Tracker
   226→// ============================================================================
   227→
   228→struct ObjectLifecycle {
   229→    std::string object_id;
   230→    std::string object_type;
   231→    
   232→    // Key lifecycle events
   233→    std::optional<TimelineEvent> created;
   234→    std::optional<TimelineEvent> destroyed;
   235→    std::vector<TimelineEvent> modifications;
   236→    
   237→    // Current state (reconstructed from events)
   238→    std::map<std::string, std::string> current_state;
   239→    
   240→    // Statistics
   241→    uint64_t creation_frame{0};
   242→    uint64_t destruction_frame{0};  // 0 if still alive
   243→    size_t total_modifications{0};
   244→    uint64_t last_access_frame{0};
   245→    
   246→    bool isAlive() const { return !destroyed.has_value(); }
   247→    uint64_t lifetime() const { 
   248→        return destruction_frame > 0 ? destruction_frame - creation_frame : 0;
   249→    }
   250→    
   251→    std::string toJson() const {
   252→        std::stringstream json;
   253→        json << "{\n";
   254→        json << "  \"object_id\": \"" << JsonUtils::escapeJson(object_id) << "\",\n";
   255→        json << "  \"object_type\": \"" << JsonUtils::escapeJson(object_type) << "\",\n";
   256→        json << "  \"alive\": " << (isAlive() ? "true" : "false") << ",\n";
   257→        json << "  \"created_frame\": " << creation_frame << ",\n";
   258→        json << "  \"modifications\": " << total_modifications << ",\n";
   259→        json << "  \"lifetime_frames\": " << lifetime() << "\n";
   260→        json << "}";
   261→        return json.str();
   262→    }
   263→};

   269→struct TimelineQueryResult {
   270→    std::vector<TimelineEvent> events;
   271→    
   272→    // Time range of results
   273→    uint64_t start_frame{0};
   274→    uint64_t end_frame{0};
   275→

   278→    Subsystem subsystem_filter{Subsystem::Unknown};
   279→    std::string object_id_filter;
   280→    
   281→    // Aggregated statistics
   282→    size_t total_events{0};
   283→    size_t error_events{0};
   284→    size_t lifecycle_events{0};
   285→    
   286→    // Object lifecycles found (if querying by object)
   287→    std::map<std::string, ObjectLifecycle> object_lifecycles;
   288→    
   289→    std::string toJson() const {
   290→        std::stringstream json;
   291→        json << "{\n";
   292→        json << "  \"total_events\": " << total_events << ",\n";
   293→        json << "  \"error_events\": " << error_events << ",\n";
   294→        json << "  \"frame_range\": [" << start_frame << ", " << end_frame << "],\n";
   295→        
   296→        json << "  \"events\": [\n";
   297→        for (size_t i = 0; i < events.size() && i < 100; ++i) {  // Limit output
   298→            json << "    " << events[i].toJson();
   299→            if (i < events.size() - 1 && i < 99) json << ",";
   300→            json << "\n";
   301→        }
   302→        if (events.size() > 100) {
   303→            json << "    ... and " << (events.size() - 100) << " more\n";
   304→        }
   305→        json << "  ]\n";
   306→        
   307→        json << "}\n";
   308→        return json.str();
   309→    }
   310→};

   316→struct StateSnapshot {
   317→    uint64_t target_frame;
   318→    DebugTimestamp approximate_time;
   319→    
   320→    // All active objects at this point
   321→    std::map<std::string, ObjectLifecycle> active_objects;
   322→    
   323→    // Recent events leading to this state

   325→

   330→        std::stringstream json;
   331→        json << "{\n";
   332→        json << "  \"frame\": " << target_frame << ",\n";
   333→        json << "  \"active_objects\": " << active_objects.size() << ",\n";
   334→        json << "  \"prior_errors\": " << prior_errors.size() << "\n";
   335→        json << "}";
   336→        return json.str();

   344→class StateTimelineSystem {
   345→public:
   346→    struct Config {
   347→        bool enabled{true};
   348→        size_t max_events{1000000};          // Global event limit
   349→        size_t max_events_per_object{10000}; // Per-object limit
   350→        bool track_all_objects{false};       // Track everything or just watched
   351→        bool auto_track_gpu{true};           // Auto-track GPU resources
   352→        bool auto_track_memory{true};        // Auto-track major allocations
   353→        bool record_full_state_changes{false}; // Record old+new values (expensive)
   354→        
   355→        static Config minimal() {
   356→            return {true, 100000, 1000, false, false, false, false};
   357→        }
   358→        
   359→        static Config comprehensive() {
   360→            return {true, 10000000, 50000, true, true, true, true};
   361→        }
   362→    };
   363→    
   364→    explicit StateTimelineSystem(const Config& config = Config())
   365→        : m_config(config), m_enabled(config.enabled), m_current_frame(0) {}
   366→    
   367→    // ========================================================================
   368→    // Control Interface
   369→    // ========================================================================
   370→    
   371→    void enable() { m_enabled.store(true); }
   372→    void disable() { m_enabled.store(false); }
   373→    bool isEnabled() const { return m_enabled.load(); }
   374→    
   375→    void clear() {
   376→        std::unique_lock lock(m_mutex);
   377→        m_events.clear();
   378→        m_object_lifecycles.clear();
   379→        m_watch_list.clear();
   380→        m_current_frame = 0;
   381→    }
   382→    
   383→    // ========================================================================
   384→    // Frame Tracking
   385→    // ========================================================================
   386→    
   387→    /**
   388→     * Advance to next frame (call once per emulated frame)
   389→     */
   390→    void advanceFrame() {
   391→        if (!isEnabled()) return;
   392→        
   393→        ++m_current_frame;
   394→        
   395→        // Optionally emit frame marker
   396→        if (m_config.track_all_objects) {
   397→            TimelineEvent marker;
   398→            marker.type = TimelineEventType::FrameMarker;
   399→            marker.frame_number = m_current_frame;
   400→            marker.description = "Frame " + std::to_string(m_current_frame);
   401→            
   402→            addEventInternal(marker);
   403→        }
   404→    }
   405→    
   406→    void setFrame(uint64_t frame) {
   407→        m_current_frame = frame;
   408→    }
   409→    
   410→    uint64_t getCurrentFrame() const { return m_current_frame.load(); }
   411→    
   412→    // ========================================================================
   413→    // Object Watching
   414→    // ========================================================================
   415→    
   416→    /**
   417→     * Watch a specific object for all state changes
   418→     */
   419→    void watchObject(const std::string& object_id, const std::string& object_type = "") {
   420→        std::unique_lock lock(m_mutex);
   421→        m_watch_list[object_id] = object_type;
   422→    }
   423→    
   424→    void unwatchObject(const std::string& object_id) {
   425→        std::unique_lock lock(m_mutex);
   426→        m_watch_list.erase(object_id);
   427→    }
   428→    
   429→    bool isWatching(const std::string& object_id) const {
   430→        std::shared_lock lock(m_mutex);
   431→        return m_watch_list.count(object_id) > 0 || m_config.track_all_objects;
   432→    }

   581→    // ========================================================================
   582→    // Query Interface - Time Travel
   583→    // ========================================================================
   584→    
   585→    /**
   586→     * Get all events in a frame range
   587→     */
   588→    TimelineQueryResult queryByFrameRange(uint64_t start_frame, uint64_t end_frame) const {
   589→        TimelineQueryResult result;
   590→        result.start_frame = start_frame;
   591→        result.end_frame = end_frame;
   592→        
   593→        std::shared_lock lock(m_mutex);
   594→        
   595→        for (const auto& event : m_events) {
   596→            if (event.frame_number >= start_frame && event.frame_number <= end_frame) {
   597→                result.events.push_back(event);
   598→                
   599→                if (event.isErrorOrCorruption()) result.error_events++;
   600→                if (event.isLifecycleEvent()) result.lifecycle_events++;
   601→                result.total_events++;
   602→            }
   603→        }
   604→        
   605→        return result;
   606→    }

   608→    /**
   609→     * Get complete lifecycle of an object
   610→     */
   611→    std::optional<ObjectLifecycle> getObjectLifecycle(const std::string& object_id) const {
   612→        std::shared_lock lock(m_mutex);
   613→        
   614→        auto it = m_object_lifecycles.find(object_id);
   615→        if (it != m_object_lifecycles.end()) {
   616→            return it->second;
   617→        }
   618→        return std::nullopt;
   619→    }
   620→    
   621→    /**
   622→     * Get state snapshot at a specific frame
   623→     */
   624→    StateSnapshot getSnapshotAtFrame(uint64_t target_frame) const {
   625→        StateSnapshot snapshot;
   626→        snapshot.target_frame = target_frame;
   627→        
   628→        std::shared_lock lock(m_mutex);
   629→        
   630→        // Find all objects that existed at this frame
   631→        for (const auto& [obj_id, lifecycle] : m_object_lifecycles) {
   632→            if (lifecycle.creation_frame <= target_frame &&
   633→                (lifecycle.destruction_frame == 0 || lifecycle.destruction_frame > target_frame)) {
   634→                snapshot.active_objects[obj_id] = lifecycle;
   635→            }
   636→        }
   637→        
   638→        // Get recent events leading up to this frame
   639→        int count = 0;
   640→        for (auto it = m_events.rbegin(); it != m_events.rend() && count < 50; ++it) {
   641→            if (it->frame_number <= target_frame) {
   642→                snapshot.preceding_events.push_back(*it);
   643→                count++;
   644→                
   645→                if (it->isErrorOrCorruption()) {
   646→                    snapshot.prior_errors.push_back(*it);
   647→                }
   648→            }
   649→        }
   650→        
   651→        return snapshot;
   652→    }

   654→    /**
   655→     * Find what happened to an object around a specific frame
   656→     */
   657→    std::vector<TimelineEvent> getObjectEventsAroundFrame(
   658→        const std::string& object_id,
   659→        uint64_t center_frame,
   660→        uint64_t window_size = 100) const {
   661→        
   662→        std::vector<TimelineEvent> result;
   663→        uint64_t start = window_size > center_frame ? 0 : center_frame - window_size;
   664→        uint64_t end = center_frame + window_size;
   665→        
   666→        std::shared_lock lock(m_mutex);
   667→        
   668→        for (const auto& event : m_events) {
   669→            if (event.object_id == object_id &&
   670→                event.frame_number >= start &&
   671→                event.frame_number <= end) {
   672→                result.push_back(event);
   673→            }
   674→        }
   675→        
   676→        return result;
   677→    }

   679→    /**
   680→     * Find all errors/corruptions in a frame range
   681→     */
   682→    std::vector<TimelineEvent> findErrorsInFrameRange(
   683→        uint64_t start_frame,
   684→        uint64_t end_frame) const {
   685→        
   686→        std::vector<TimelineEvent> errors;
   687→        
   688→        std::shared_lock lock(m_mutex);
   689→        
   690→        for (const auto& event : m_events) {
   691→            if (event.isErrorOrCorruption() &&
   692→                event.frame_number >= start_frame &&
   693→                event.frame_number <= end_frame) {
   694→                errors.push_back(event);
   695→            }
   696→        }
   697→        
   698→        return errors;
   699→    }

   701→    /**
   702→     * Find objects that were destroyed shortly before a crash
   703→     * (Common pattern: use-after-free)
   704→     */
   705→    std::vector<ObjectLifetime> findObjectsDestroyedBeforeFrame(
   706→        uint64_t crash_frame,
   707→        uint64_t lookback_window = 1000) const {
   708→        
   709→        struct ObjectLifetime {
   710→            std::string object_id;
   711→            std::string object_type;
   712→            uint64_t destruction_frame;
   713→            uint64_t frames_before_crash;
   714→        };
   715→        
   716→        std::vector<ObjectLifetime> recently_destroyed;
   717→        uint64_t start = lookback_window > crash_frame ? 0 : crash_frame - lookback_window;
   718→        
   719→        std::shared_lock lock(m_mutex);
   720→        
   721→        for (const auto& [obj_id, lifecycle] : m_object_lifecycles) {
   722→            if (lifecycle.destruction_frame > 0 &&
   723→                lifecycle.destruction_frame >= start &&
   724→                lifecycle.destruction_frame < crash_frame) {
   725→                
   726→                ObjectLifetime lt;
   727→                lt.object_id = obj_id;
   728→                lt.object_type = lifecycle.object_type;
   729→                lt.destruction_frame = lifecycle.destruction_frame;
   730→                lt.frames_before_crash = crash_frame - lifecycle.destruction_frame;
   731→                
   732→                recently_destroyed.push_back(lt);
   733→            }
   734→        }
   735→        
   736→        // Sort by proximity to crash
   737→        std::sort(recently_destroyed.begin(), recently_destroyed.end(),
   738→            [](const ObjectLifetime& a, const ObjectLifetime& b) {
   739→                return a.frames_before_crash < b.frames_before_crash;
   740→            });
   741→        
   742→        return recently_destroyed;
   743→    }

   749→    struct Stats {
   750→        size_t total_events{0};
   751→        uint64_t current_frame{0};
   752→        size_t tracked_objects{0};
   753→        size_t watched_objects{0};
   754→        size_t active_objects{0};
   755→        size_t total_errors{0};
   756→        size_t total_corruptions{0};
   757→        
   758→        std::map<TimelineEventType, size_t> events_by_type;
   759→        std::map<Subsystem, size_t> events_by_subsystem;
   760→    };
   761→    
   762→    Stats getStats() const {
   763→        std::shared_lock lock(m_mutex);
   764→        
   765→        Stats stats;
   766→        stats.current_frame = m_current_frame;
   767→        stats.tracked_objects = m_object_lifecycles.size();
   768→        stats.watched_objects = m_watch_list.size();
   769→        
   770→        for (const auto& [obj_id, lifecycle] : m_object_lifecycles) {
   771→            if (lifecycle.isAlive()) stats.active_objects++;
   772→        }
   773→        
   774→        for (const auto& event : m_events) {
   775→            stats.total_events++;
   776→            stats.events_by_type[event.type]++;
   777→            stats.events_by_subsystem[event.subsystem]++;
   778→            
   779→            if (event.type == TimelineEventType::ErrorDetected) stats.total_errors++;
   780→            if (event.type == TimelineEventType::CorruptionDetected) stats.total_corruptions++;
   781→        }
   782→        
   783→        return stats;
   784→    }

   805→    bool shouldTrack(const std::string& object_id) const {
   806→        if (m_config.track_all_objects) return true;
   807→        return m_watch_list.count(object_id) > 0;
   808→    }
   809→    
   810→    void addEventInternal(const TimelineEvent& event) {
   811→        std::unique_lock lock(m_mutex);
   812→        
   813→        // Check global limit
   814→        if (m_events.size() >= m_config.max_events) {
   815→            m_events.pop_front();  // Evict oldest
   816→        }
   817→        
   818→        m_events.push_back(event);
   819→        
   820→        // Update object lifecycle tracking
   821→        updateObjectLifecycle(event);
   822→    }
   823→    
   824→    void updateObjectLifecycle(const TimelineEvent& event) {
   825→        // Get or create lifecycle entry
   826→        auto& lifecycle = m_object_lifecycles[event.object_id];
   827→        
   828→        if (lifecycle.object_id.empty()) {
   829→            lifecycle.object_id = event.object_id;
   830→            lifecycle.object_type = event.object_type;
   831→        }
   832→        
   833→        switch (event.type) {
   834→            case TimelineEventType::ObjectCreated:
   835→            case TimelineEventType::ResourceAllocated:
   836→                if (!lifecycle.created.has_value()) {
   837→                    lifecycle.created = event;
   838→                    lifecycle.creation_frame = event.frame_number;
   839→                }
   840→                break;
   841→                
   842→            case TimelineEventType::ObjectDestroyed:
   843→            case TimelineEventType::ResourceFreed:
   844→                if (!lifecycle.destroyed.has_value()) {
   845→                    lifecycle.destroyed = event;
   846→                    lifecycle.destruction_frame = event.frame_number;
   847→                }
   848→                break;
   849→                
   850→            case TimelineEventType::ObjectModified:
   851→            case TimelineEventType::StateChanged:
   852→                lifecycle.modifications.push_back(event);
   853→                lifecycle.total_modifications++;
   854→                
   855→                // Update current state
   856→                if (!event.attribute_name.empty()) {
   857→                    lifecycle.current_state[event.attribute_name] = event.new_value;
   858→                }
   859→                break;
   860→                
   861→            default:
   862→                break;
   863→        }
   864→        
   865→        // Update last access
   866→        if (event.frame_number > lifecycle.last_access_frame) {
   867→            lifecycle.last_access_frame = event.frame_number;
   868→        }
   869→    }

   151→   151→        std::set<int> failure_codes;     // These codes indicate known failures

   153→   153→    };
   154→   154→    
   155→   155→    ReturnValueConstraint return_constraint;
   156→   156→    
   157→   157→    // Pre-conditions (should be true before call)

   226→   226→    switch (status) {

   228→   228→        case CallStatus::Completed: return "Completed";
   229→   229→        case CallStatus::ContractPassed: return "ContractPassed";

   238→   238→struct HLECallRecord {
   239→   239→    std::string id;
   240→   240→    DebugTimestamp start_time;
   241→   241→    DebugTimestamp end_time;
   242→   242→    
   243→   243→    // Function identification
   244→   244→    std::string function_name;
   245→   245→    int function_id{-1};
   246→   246→    std::string library_name;
   247→   247→    
   248→   248→    // Call parameters (serialized)
   249→   249→    std::string parameters_json;
   250→   250→    
   251→   251→    // Result
   252→   252→    int return_value{0};
   253→   253→    CallStatus status{CallStatus::Started};
   254→   254→    std::string error_message;
   255→   255→    
   256→   256→    // Contract validation
   257→   257→    std::vector<SideEffectRecord> observed_effects;
   258→   258→    std::vector<std::string> missing_required_effects;
   259→   259→    std::vector<std::string> failed_validations;
   260→   260→    
   261→   261→    // Provenance
   262→   262→    SourceLocation caller_location;
   263→   263→    std::string thread_id;
   264→   264→    uint64_t guest_call_site{0};
   265→   265→    
   266→   266→    static std::string generateId() {
   267→   267→        static std::atomic<uint64_t> counter{0};
   268→   268→        auto now = std::chrono::high_resolution_clock::now().time_since_epoch().count();
   269→   269→        return "call_" + std::to_string(now) + "_" + std::to_string(++counter);
   270→   270→    }
   271→   271→    
   272→   272→    HLECallRecord() {
   273→   273→        id = generateId();
   274→   274→        start_time = DebugTimestamp::now();
   275→   275→    }
   276→   276→    
   277→   277→    /**
   278→   278→     * Calculate duration in microseconds
   279→   279→     */
   280→   280→    uint64_t durationMicroseconds() const {
   281→   281→        if (end_time.steady_clock.time_since_epoch().count() == 0) {
   282→   282→            return 0;  // Not yet completed
   283→   283→        }
   284→   284→        auto duration = end_time.steady_clock - start_time.steady_clock;
   285→   285→        return std::chrono::duration_cast<std::chrono::microseconds>(duration).count();
   286→   286→    }
   287→   287→    
   288→   288→    std::string toJson() const {
   289→   289→        std::stringstream json;
   290→   290→        json << "{\n";
   291→   291→        json << "  \"id\": \"" << JsonUtils::escapeJson(id) << "\",\n";
   292→   292→        json << "  \"function\": \"" << JsonUtils::escapeJson(function_name) << "\",\n";
   293→   293→        json << "  \"library\": \"" << JsonUtils::escapeJson(library_name) << "\",\n";
   294→   294→        json << "  \"return_value\": " << return_value << ",\n";
   295→   295→        json << "  \"status\": \"" << callStatusToString(status) << "\",\n";
   296→   296→        json << "  \"duration_us\": " << durationMicroseconds() << ",\n";
   297→   297→        json << "  \"observed_effects\": " << observed_effects.size() << ",\n";
   298→   298→        json << "  \"missing_effects\": " << missing_required_effects.size() << "\n";
   299→   299→        
   300→   300→        if (!missing_required_effects.empty()) {
   301→   301→            json << ",\n  \"missing_effect_details\": [\n";
   302→   302→            for (size_t i = 0; i < missing_required_effects.size(); ++i) {
   303→   303→                json << "    \"" << JsonUtils::escapeJson(missing_required_effects[i]) << "\"";
   304→   304→                if (i < missing_required_effects.size() - 1) json << ",";
   305→   305→                json << "\n";
   306→   306→            }
   307→   307→            json << "  ]";
   308→   308→        }
   309→   309→        
   310→   310→        json << "\n}";
   311→   311→        return json.str();
   312→   312→    }
   313→   313→};

   505→   505→     * Complete an HLE call and validate contract
   506→   506→     */
   507→   507→    void completeCall(
   508→   508→        const std::string& call_id,
   509→   509→        int return_value,
   510→   510→        const std::vector<SideEffectRecord>& observed_effects = {}) {
   511→   511→        
   512→   512→        if (!isEnabled() || call_id.empty()) return;
   513→   513→        
   514→   514→        std::unique_lock lock(m_mutex);
   515→   515→        
   516→   516→        auto it = m_active_calls.find(call_id);
   517→   517→        if (it == m_active_calls.end()) return;
   518→   518→        
   519→   519→        HLECallRecord& record = it->second;
   520→   520→        record.return_value = return_value;
   521→   521→        record.end_time = DebugTimestamp::now();
   522→   522→        record.observed_effects = observed_effects;
   523→   523→        
   524→   524→        // Validate against registered contract
   525→   525→        validateCall(record);
   526→   526→        
   527→   527→        // Move from active to history
   528→   528→        m_call_history.push_back(record);
   529→   529→        m_active_calls.erase(it);
   530→   530→        
   531→   531→        // Enforce history limit
   532→   532→        while (m_call_history.size() > m_config.max_history) {
   533→   533→            m_call_history.pop_front();
   534→   534→        }
   535→   535→    }
   536→   536→    
   537→   537→    /**
   538→   538→     * Convenience: Complete call with no side effects recorded
   539→   539→     */
   540→   540→    void completeCallNoEffects(const std::string& call_id, int return_value) {
   541→   541→        completeCall(call_id, return_value, {});
   542→   542→    }

   545→   545→     * Record that a specific side effect occurred during a call
   546→   546→     * (Can be called between startCall and completeCall)
   547→   547→     */
   548→   548→    void recordSideEffect(
   549→   549→        const std::string& call_id,
   550→   550→        SideEffectType type,
   551→   551→        const std::string& description = "",
   552→   552→        const std::map<std::string, std::string>& data = {},
   553→   553→        const SourceLocation& source = DEBUG_HERE()) {
   554→   554→        
   555→   555→        if (!isEnabled() || call_id.empty()) return;
   556→   556→        
   557→   557→        std::unique_lock lock(m_mutex);
   558→   558→        
   559→   559→        auto it = m_active_calls.find(call_id);
   560→   560→        if (it == m_active_calls.end()) return;
   561→   561→        
   562→   562→        SideEffectRecord effect;
   563→   563→        effect.type = type;
   564→   564→        effect.description = description.empty() ? sideEffectTypeToString(type) : description;
   565→   565→        effect.data = data;
   566→   566→        effect.source = source;
   567→   567→        effect.hle_function_name = it->second.function_name;
   568→   568→        effect.verified = true;
   569→   569→        
   570→   570→        it->second.observed_effects.push_back(effect);
   571→   571→    }

   625→   625→    /**
   626→   626→     * Get statistics about contract compliance
   627→   627→     */
   628→   628→    struct ComplianceStats {
   629→   629→        size_t total_calls{0};
   630→   630→        size_t passed_calls{0};
   631→   631→        size_t failed_calls{0};
   632→   632→        size_t calls_with_missing_effects{0};
   633→   633→        
   634→   634→        // By library
   635→   635→        std::map<std::string, size_t> calls_by_library;
   636→   636→        std::map<std::string, size_t> violations_by_library;
   637→   637→        
   638→   638→        // Most problematic functions
   639→   639→        std::vector<std::pair<std::string, size_t>> top_violations;
   640→   640→        
   641→   641→        double pass_rate() const {
   642→   642→            return total_calls > 0 ? 
   643→   643→                static_cast<double>(passed_calls) / total_calls : 1.0;
   644→   644→        }
   645→   645→        
   646→   646→        std::string toJson() const {
   647→   647→            std::stringstream json;
   648→   648→            json << "{\n";
   649→   649→            json << "  \"total_calls\": " << total_calls << ",\n";
   650→   650→            json << "  \"passed\": " << passed_calls << ",\n";
   651→   651→            json << "  \"failed\": " << failed_calls << ",\n";
   652→   652→            json << "  \"pass_rate\": " << std::fixed << std::setprecision(2) 
   653→   653→                 << (pass_rate() * 100.0) << "%\n";
   654→   654→            json << "}";
   655→   655→            return json.str();
   656→   656→        }
   657→   657→    };
   658→   658→    
   659→   659→    ComplianceStats getComplianceStats() const {
   660→   660→        std::shared_lock lock(m_mutex);
   661→   661→        
   662→   662→        ComplianceStats stats;
   663→   663→        stats.total_calls = m_call_history.size();
   664→   664→        
   665→   665→        // Count by function for top violators
   666→   666→        std::map<std::string, size_t> violation_counts;
   667→   667→        
   668→   668→        for (const auto& call : m_call_history) {
   669→   669→            stats.calls_by_library[call.library_name]++;
   670→   670→            
   671→   671→            switch (call.status) {
   672→   672→                case CallStatus::ContractPassed:
   673→   673→                    stats.passed_calls++;
   674→   674→                    break;
   675→   675→                case CallStatus::ContractFailed:
   676→   676→                case CallStatus::MissingSideEffects:
   677→   677→                    stats.failed_calls++;
   678→   678→                    stats.calls_with_missing_effects++;
   679→   679→                    violation_counts[call.function_name]++;
   680→   680→                    stats.violations_by_library[call.library_name]++;
   681→   681→                    break;
   682→   682→                default:
   683→   683→                    break;
   684→   684→            }
   685→   685→        }
   686→   686→        
   687→   687→        // Sort for top violators
   688→   688→        using PairType = std::pair<std::string, size_t>;
   689→   689→        std::vector<PairType> sorted(violation_counts.begin(), violation_counts.end());
   690→   690→        std::sort(sorted.begin(), sorted.end(),
   691→   691→            [](const PairType& a, const PairType& b) { return b.second < a.second; });
   692→   692→        
   693→   693→        stats.top_violations.assign(sorted.begin(), 
   694→   694→            sorted.begin() + std::min(sorted.size(), size_t(10)));
   695→   695→        
   696→   696→        return stats;
   697→   697→    }

   753→   753→    std::string makeContractKey(const std::string& func, const std::string& lib) {
   754→   754→        return lib + "::" + func;
   755→   755→    }
   756→   756→    
   757→   757→    void validateCall(HLECallRecord& record) {
   758→   758→        std::string key = makeContractKey(record.function_name, record.library_name);
   759→   759→        
   760→   760→        auto contract_it = m_contracts.find(key);
   761→   761→        if (contract_it == m_contracts.end()) {
   762→   762→            // No contract registered - mark as passed (or warning in strict mode)
   763→   763→            record.status = m_config.strict_mode ? 
   764→   764→                CallStatus::MissingSideEffects : CallStatus::Completed;
   765→   765→            return;
   766→   766→        }
   767→   767→        
   768→   768→        const HLEContract& contract = contract_it->second;
   769→   769→        ContractViolation violation;
   770→   770→        violation.call_record = record;
   771→   771→        violation.contract = contract;
   772→   772→        
   773→   773→        bool has_failure = false;
   774→   774→        
   775→   775→        // Check each expected effect
   776→   776→        for (const auto& expected : contract.expected_effects) {
   777→   777→            if (!expected.required) continue;
   778→   778→            
   779→   779→            bool found = false;
   780→   780→            for (const auto& observed : record.observed_effects) {
   781→   781→                if (observed.type == expected.type) {
   782→   782→                    found = true;
   783→   783→                    
   784→   784→                    // Run custom validator if provided
   785→   785→                    if (expected.validator && !expected.validator()) {
   786→   786→                        violation.failed_validations.push_back(
   787→   787→                            "Validator failed for: " + expected.description);
   788→   788→                        has_failure = true;
   789→   789→                    }
   790→   790→                    
   791→   791→                    break;
   792→   792→                }
   793→   793→            }
   794→   794→            
   795→   795→            if (!found) {
   796→   796→                violation.missing_side_effects.push_back(expected.description);
   797→   797→                has_failure = true;
   798→   798→                
   799→   799→                // Also add to call record
   800→   800→                record.missing_required_effects.push_back(expected.description);
   801→   801→            }
   802→   802→        }
   803→   803→        
   804→   804→        // Check return value constraint
   805→   805→        if (contract.return_constraint.check_return_value) {
   806→   806→            bool is_success = contract.return_constraint.success_codes.count(record.return_value) > 0;
   807→   807→            bool is_known_failure = contract.return_constraint.failure_codes.count(record.return_value) > 0;
   808→   808→            
   809→   809→            if (!is_success && !is_known_failure && !contract.return_constraint.allow_other_codes) {
   810→   810→                violation.unexpected_errors.push_back(
   811→   811→                    "Unexpected return value: " + std::to_string(record.return_value));
   812→   812→                has_failure = true;
   813→   813→            }
   814→   814→        }
   815→   815→        
   816→   816→        // Determine final status
   817→   817→        if (has_failure) {
   818→   818→            record.status = violation.missing_side_effects.empty() ?
   819→   819→                CallStatus::ContractFailed : CallStatus::MissingSideEffects;
   820→   820→            
   821→   821→            // Assess severity
   822→   822→            assessViolationSeverity(violation);
   823→   823→            
   824→   824→            // Store violation
   825→   825→            m_violations.push_back(violation);
   826→   826→            
   827→   827→            // Keep violations bounded
   828→   828→            while (m_violations.size() > m_config.max_history) {
   829→   829→                m_violations.erase(m_violations.begin());
   830→   830→            }
   831→   831→        } else {
   832→   832→            record.status = CallStatus::ContractPassed;
   833→   833→        }
   834→   834→    }
   835→   835→    
   836→   836→    void assessViolationSeverity(ContractViolation& violation) {
   837→   837→        // Critical: File handle claimed valid but wasn't created
   838→   838→        for (const auto& missing : violation.missing_side_effects) {
   839→   839→            if (missing.find("FileHandle") != std::string::npos ||
   840→   840→                missing.find("Handle") != std::string::npos) {
   841→   841→                violation.severity = Severity::Error;
   842→   842→                violation.impact_description = 
   843→   843→                    "Guest may use invalid handle, leading to crashes or corruption";
   844→   844→                violation.suggested_fix =
   845→   845→                    "Ensure handle is properly created and tracked before returning success";
   846→   846→                return;
   847→   847→            }
   848→   848→        }
   849→   849→        
   850→   850→        // Warning: State change not performed
   851→   851→        for (const auto& missing : violation.missing_side_effects) {
   852→   852→            if (missing.find("State") != std::string::npos ||
   853→   853→                missing.find("Modified") != std::string::npos) {
   854→   854→                violation.severity = Severity::Warning;
   855→   855→                violation.impact_description =
   856→   856→                    "Guest may operate on stale state, causing incorrect behavior";
   857→   857→                return;
   858→   858→            }
   859→   859→        }
   860→   860→        
   861→   861→        // Default
   862→   862→        violation.severity = Severity::Info;
   863→   863→        violation.impact_description =
   864→   864→            "HLE function may not fully implement expected behavior";
   865→   865→    }

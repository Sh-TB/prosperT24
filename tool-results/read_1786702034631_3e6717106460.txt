     1→/**
     2→ * Diagnostic Quality Analyzer - Priority 4
     3→ * 
     4→ * PROBLEM SOLVED:
     5→ * A lot of time wasted because diagnostics were added but never executed.
     6→ * 
     7→ * EXAMPLES OF MISLEADING SITUATIONS:
     8→ *   "Added breakpoint, no output, therefore not the issue"
     9→ *   
    10→ *   REALITY:
    11→ *   - Wrong code path (branch not taken)
    12→ *   - Unreachable code (dead path)
    13→ *   - Condition impossible (contradictory preconditions)
    14→ *   - Diagnostic removed during optimization
    15→ *   - Exception thrown before diagnostic reached
    16→ * 
    17→ * REQUIREMENT:
    18→ * Every diagnostic must report:
    19→ *   - Registered: When it was added to the system
    20→ *   - Executed: How many times it ran
    21→ *   - Last execution: When it last ran
    22→ *   - Not reached: Why it might not have run
    23→ */

    39→// ============================================================================
    40→// Diagnostic Status
    41→// ============================================================================
    42→
    43→enum class DiagnosticStatus : uint8_t {
    44→    Registered,      // Created and registered in system
    45→    Executed,        // Ran at least once
    46→    Triggered,       // Condition was met (for conditional diagnostics)
    47→    NotReached,      // Code path never entered
    48→    Removed,         // Explicitly removed
    49→    Error            // Diagnostic itself had an error
    50→};
    51→
    52→inline std::string diagnosticStatusToString(DiagnosticStatus status) {
    53→    switch (status) {
    54→        case DiagnosticStatus::Registered: return "Registered";
    55→        case DiagnosticStatus::Executed: return "Executed";
    56→        case DiagnosticStatus::Triggered: return "Triggered";
    57→        case DiagnosticStatus::NotReached: return "NotReached";
    58→        case DiagnosticStatus::Removed: return "Removed";
    59→        case DiagnosticStatus::Error: return "Error";
    60→    }
    61→    return "Unknown";
    62→}

   187→struct DiagnosticDefinition {
   188→    std::string id;
   189→    std::string name;
   190→    std::string description;
   191→    DiagnosticType type;
   192→    
   193→    // Location
   194→    SourceLocation location;
   195→    Subsystem subsystem{Subsystem::Unknown};
   196→    
   197→    // Condition (if conditional)
   198→    std::function<bool()> condition;  // When should this run?
   199→    bool is_conditional{false};
   200→    
   201→    // What it checks
   202→    std::function<bool()> check;     // The actual diagnostic logic
   203→    
   204→    // What to do when triggered
   205→    std::function<void(const std::string&)> on_triggered;
   206→    
   207→    // Metadata
   208→    std::map<std::string, std::string> tags;
   209→    std::string category;
   210→    int priority{0};  // Higher = more important
   211→    
   212→    // Expected frequency
   213→    size_t expected_executions_per_session{1};  // 0 = unknown/unlimited
   214→    
   215→    static std::string generateId() {
   216→        static std::atomic<uint64_t> counter{0};
   217→        auto now = std::chrono::high_resolution_clock::now().time_since_epoch().count();
   218→        return "diag_" + std::to_string(now) + "_" + std::to_string(++counter);
   219→    }
   220→    
   221→    DiagnosticDefinition() {
   222→        id = generateId();
   223→    }
   224→};

   226→// ============================================================================
   227→// Diagnostic Execution Record
   228→// ============================================================================
   229→
   230→struct DiagnosticExecution {
   231→    std::string diagnostic_id;
   232→    DebugTimestamp timestamp;
   233→    uint64_t frame_number{0};
   234→    
   235→    // Result
   236→    DiagnosticStatus status{DiagnosticStatus::Registered};
   237→    bool passed{true};
   238→    std::string result_message;
   239→    std::string failure_detail;
   240→    
   241→    // Context
   242→    std::string thread_id;
   243→    std::map<std::string, std::string> context_data;  // Values at time of execution
   244→    
   245→    // Performance
   246→    uint64_t execution_time_ns{0};
   247→    
   248→    static std::string generateId() {
   249→        static std::atomic<uint64_t> counter{0};
   250→        auto now = std::chrono::high_resolution_clock::now().time_since_epoch().count();
   251→        return "exec_" + std::to_string(now) + "_" + std::to_string(++counter);
   252→    }
   253→    
   254→    DiagnosticExecution() {
   255→        timestamp = DebugTimestamp::now();
   256→    }
   257→    
   258→    std::string toJson() const {
   259→        std::stringstream json;
   260→        json << "{\n";
   261→        json << "  \"diagnostic_id\": \"" << JsonUtils::escapeJson(diagnostic_id) << "\",\n";
   262→        json << "  \"timestamp\": \"" << timestamp.toISO8601() << "\",\n";
   263→        json << "  \"frame\": " << frame_number << ",\n";
   264→        json << "  \"status\": \"" << diagnosticStatusToString(status) << "\",\n";
   265→        json << "  \"passed\": " << (passed ? "true" : "false") << ",\n";
   266→        if (!result_message.empty()) {
   267→            json << "  \"message\": \"" << JsonUtils::escapeJson(result_message) << "\"\n";
   268→        }
   269→        json << "}";
   270→        return json.str();
   271→    }
   272→};

   274→// ============================================================================
   275→// Diagnostic Summary - Current State of a Diagnostic
   276→// ============================================================================
   277→
   278→struct DiagnosticSummary {
   279→    DiagnosticDefinition definition;
   280→    
   281→    // Execution history
   282→    std::vector<DiagnosticExecution> executions;
   283→    
   284→    // Aggregated stats
   285→    DiagnosticStatus current_status{DiagnosticStatus::Registered};
   286→    size_t total_executions{0};
   287→    size_t total_passes{0};
   288→    size_t total_failures{0};
   289→    
   290→    // Timing
   291→    DebugTimestamp registration_time;
   292→    DebugTimestamp first_execution;
   293→    DebugTimestamp last_execution;
   294→    
   295→    // If not executing, why?
   296→    NonExecutionReason suspected_reason{NonExecutionReason::Unknown};
   297→    std::string non_execution_analysis;
   298→    
   299→    // Health assessment
   300→    enum class Health {
   301→        Healthy,       // Running as expected
   302→        Suspicious,    // Should have run but hasn't
   303→        Broken,        // Failing consistently
   304→        Unknown         // Not enough data
   305→    } health{Health::Unknown};
   306→    
   307→    std::string health_reason;
   308→    
   309→    double passRate() const {
   310→        return total_executions > 0 ?
   311→            static_cast<double>(total_passes) / total_executions : 1.0;
   312→    }
   313→    
   314→    bool hasEverExecuted() const { return total_executions > 0; }
   315→    
   316→    std::string toJson() const {
   317→        std::stringstream json;
   318→        json << "{\n";
   319→        json << "  \"id\": \"" << JsonUtils::escapeJson(definition.id) << "\",\n";
   320→        json << "  \"name\": \"" << JsonUtils::escapeJson(definition.name) << "\",\n";
   321→        json << "  \"type\": \"" << diagnosticTypeToString(definition.type) << "\",\n";
   322→        json << "  \"status\": \"" << diagnosticStatusToString(current_status) << "\",\n";
   323→        json << "  \"executions\": " << total_executions << ",\n";
   324→        json << "  \"passes\": " << total_passes << ",\n";
   325→        json << "  \"failures\": " << total_failures << ",\n";
   326→        json << "  \"pass_rate\": " << std::fixed << std::setprecision(2) 
   327→             << (passRate() * 100.0) << "%,\n";
   328→        
   329→        if (!hasEverExecuted()) {
   330→            json << "  \"not_reached_reason\": \"" 
   331→                 << JsonUtils::escapeJson(nonExecutionReasonToString(suspected_reason)) << "\",\n";
   332→            json << "  \"analysis\": \"" << JsonUtils::escapeJson(non_execution_analysis) << "\",\n";
   333→        }
   334→        
   335→        const char* health_str = "";
   336→        switch (health) {
   337→            case Health::Healthy: health_str = "Healthy"; break;
   338→            case Health::Suspicious: health_str = "Suspicious"; break;
   339→            case Health::Broken: health_str = "Broken"; break;
   340→            case Health::Unknown: health_str = "Unknown"; break;
   341→        }
   342→        json << "  \"health\": \"" << health_str << "\"\n";
   343→        
   344→        json << "}";
   345→        return json.str();
   346→    }
   347→};

   370→    // Analysis

   372→    std::vector<std::string> recommendations;

   374→    double coverage_score{0.0};  // Estimated % of diagnostics actually running

   382→        json << "  \"broken\": " << broken_count << ",\n";

   392→        }
   393→        json << "  ],\n";
   394→        
   395→        json << "  \"recommendations\": [\n";

   397→            json << "    \"" << JsonUtils::escapeJson(recommendations[i]) << "\"";
   398→            if (i < recommendations.size() - 1) json << ",";
   399→            json << "\n";
   400→        }
   401→        json << "  ]\n";
   402→        
   403→        json << "}\n";
   404→        return json.str();
   405→    }
   406→};

   408→// ============================================================================
   409→// Diagnostic Quality Analyzer - Main Class
   410→// ============================================================================
   411→
   412→class DiagnosticQualityAnalyzer {
   413→public:
   414→    struct Config {
   415→        bool enabled;
   416→        bool auto_analyze;          // Auto-analyze after each session
   417→        bool track_all_executions;  // Keep full history or just summaries
   418→        size_t max_executions_per_diagnostic;
   419→        size_t max_diagnostics;
   420→        
   421→        // Suspicion thresholds
   422→        size_t min_executions_for_health;  // Need this many before judging
   423→        double suspicious_threshold;      // Below this pass rate = suspicious
   424→        
   425→        Config()
   426→            : enabled(true)
   427→            , auto_analyze(true)
   428→            , track_all_executions(true)
   429→            , max_executions_per_diagnostic(10000)
   430→            , max_diagnostics(10000)
   431→            , min_executions_for_health(3)
   432→            , suspicious_threshold(0.3) {}
   433→        
   434→        Config(bool en, bool aa, bool tae, size_t mepd, size_t md, size_t mefh, double st)
   435→            : enabled(en)
   436→            , auto_analyze(aa)
   437→            , track_all_executions(tae)
   438→            , max_executions_per_diagnostic(mepd)
   439→            , max_diagnostics(md)
   440→            , min_executions_for_health(mefh)
   441→            , suspicious_threshold(st) {}
   442→        
   443→        static Config defaultConfig() {
   444→            return Config(true, true, true, 1000, 5000, 5, 0.5);
   445→        }
   446→    };
   447→    
   448→    explicit DiagnosticQualityAnalyzer(const Config& config = Config())
   449→        : m_config(config), m_enabled(config.enabled) {}
   450→    
   451→    // ========================================================================
   452→    // Control Interface
   453→    // ========================================================================
   454→    
   455→    void enable() { m_enabled.store(true); }
   456→    void disable() { m_enabled.store(false); }
   457→    bool isEnabled() const { return m_enabled.load(); }
   458→    
   459→    void clear() {
   460→        std::unique_lock lock(m_mutex);
   461→        m_diagnostics.clear();
   462→        m_executions.clear();
   463→    }
   464→    
   465→    // ========================================================================
   466→    // Diagnostic Registration
   467→    // ========================================================================
   468→    
   469→    /**
   470→     * Register a new diagnostic
   471→     */
   472→    std::string registerDiagnostic(
   473→        const std::string& name,
   474→        DiagnosticType type,
   475→        std::function<bool()> check_func,
   476→        const SourceLocation& loc = DEBUG_HERE(),
   477→        Subsystem subs = Subsystem::Unknown,
   478→        const std::string& description = "") {
   479→        
   480→        if (!isEnabled()) return "";
   481→        
   482→        DiagnosticDefinition def;
   483→        def.name = name;
   484→        def.type = type;
   485→        def.check = check_func;
   486→        def.location = loc;
   487→        def.subsystem = subs;
   488→        def.description = description;
   489→        
   490→        return registerDiagnostic(def);
   491→    }
   492→    
   493→    /**
   494→     * Register a diagnostic with full definition
   495→     */
   496→    std::string registerDiagnostic(const DiagnosticDefinition& definition) {
   497→        if (!isEnabled()) return "";
   498→        
   499→        std::unique_lock lock(m_mutex);
   500→        
   501→        if (m_diagnostics.size() >= m_config.max_diagnostics) {
   502→            return "";  // At capacity
   503→        }
   504→        
   505→        DiagnosticSummary summary;
   506→        summary.definition = definition;
   507→        summary.registration_time = DebugTimestamp::now();
   508→        summary.current_status = DiagnosticStatus::Registered;
   509→        
   510→        m_diagnostics[definition.id] = summary;
   511→        
   512→        return definition.id;
   513→    }
   514→    
   515→    /**
   516→     * Unregister a diagnostic
   517→     */
   518→    bool unregisterDiagnostic(const std::string& id) {
   519→        std::unique_lock lock(m_mutex);
   520→        
   521→        auto it = m_diagnostics.find(id);
   522→        if (it == m_diagnostics.end()) return false;
   523→        
   524→        it->second.current_status = DiagnosticStatus::Removed;
   525→        // Keep in map for historical analysis, just mark as removed
   526→        return true;
   527→    }
   528→    
   529→    // ========================================================================
   530→    // Execution Interface
   531→    // ========================================================================
   532→    
   533→    /**
   534→     * Execute a diagnostic by ID
   535→     * Returns whether it passed
   536→     */
   537→    std::optional<bool> executeDiagnostic(const std::string& id) {
   538→        if (!isEnabled()) return std::nullopt;
   539→        
   540→        DiagnosticExecution exec;
   541→        exec.diagnostic_id = id;
   542→        exec.thread_id = getCurrentThreadId();
   543→        
   544→        std::unique_lock lock(m_mutex);
   545→        
   546→        auto it = m_diagnostics.find(id);
   547→        if (it == m_diagnostics.end()) {
   548→            exec.status = DiagnosticStatus::Error;
   549→            exec.result_message = "Diagnostic not found";
   550→            
   551→            if (m_config.track_all_executions) {
   552→                m_executions.push_back(exec);
   553→            }
   554→            return std::nullopt;
   555→        }
   556→        
   557→        DiagnosticSummary& summary = it->second;
   558→        lock.unlock();  // Don't hold lock during execution
   559→        
   560→        // Run the diagnostic
   561→        auto start = std::chrono::high_resolution_clock::now();
   562→        
   563→        try {
   564→            if (summary.definition.check) {
   565→                exec.passed = summary.definition.check();
   566→            }
   567→            exec.status = DiagnosticStatus::Executed;
   568→            
   569→            if (exec.passed) {
   570→                exec.status = DiagnosticStatus::Triggered;
   571→                if (summary.definition.on_triggered) {
   572→                    summary.definition.on_triggered(summary.definition.id);
   573→                }
   574→            }
   575→        } catch (const std::exception& e) {
   576→            exec.passed = false;
   577→            exec.status = DiagnosticStatus::Error;
   578→            exec.failure_detail = e.what();
   579→        }
   580→        
   581→        auto end = std::chrono::high_resolution_clock::now();
   582→        exec.execution_time_ns = std::chrono::duration_cast<std::chrono::nanoseconds>(end - start).count();
   583→        
   584→        // Update summary
   585→        lock.lock();
   586→        summary.executions.push_back(exec);
   587→        summary.total_executions++;
   588→        summary.current_status = exec.status;
   589→        summary.last_execution = exec.timestamp;
   590→        
   591→        if (summary.total_executions == 1) {
   592→            summary.first_execution = exec.timestamp;
   593→        }
   594→        
   595→        if (exec.passed) {
   596→            summary.total_passes++;
   597→        } else {
   598→            summary.total_failures++;
   599→        }
   600→        
   601→        // Enforce per-diagnostic limit
   602→        while (summary.executions.size() > m_config.max_executions_per_diagnostic) {
   603→            summary.executions.erase(summary.executions.begin());
   604→        }
   605→        
   606→        if (m_config.track_all_executions) {
   607→            m_executions.push_back(exec);
   608→        }
   609→        
   610→        return exec.passed;
   611→    }
   612→    
   613→    /**
   614→     * Mark a code path as reached (lightweight diagnostic)
   615→     */
   616→    void markCodePathReached(const std::string& id) {
   617→        if (!isEnabled()) return;
   618→        
   619→        DiagnosticExecution exec;
   620→        exec.diagnostic_id = id;
   621→        exec.status = DiagnosticStatus::Executed;
   622→        exec.passed = true;
   623→        exec.result_message = "Code path reached";
   624→        
   625→        std::unique_lock lock(m_mutex);
   626→        
   627→        auto it = m_diagnostics.find(id);
   628→        if (it != m_diagnostics.end()) {
   629→            DiagnosticSummary& summary = it->second;
   630→            summary.executions.push_back(exec);
   631→            summary.total_executions++;
   632→            summary.total_passes++;
   633→            summary.last_execution = exec.timestamp;
   634→            
   635→            if (summary.total_executions == 1) {
   636→                summary.first_execution = exec.timestamp;
   637→            }
   638→        }
   639→    }
   640→    
   641→    /**
   642→     * Record that a diagnostic was not reachable with reason
   643→     */
   644→    void markNotReachable(const std::string& id, NonExecutionReason reason,
   645→                          const std::string& analysis = "") {
   646→        if (!isEnabled()) return;
   647→        
   648→        std::unique_lock lock(m_mutex);
   649→        
   650→        auto it = m_diagnostics.find(id);
   651→        if (it != m_diagnostics.end()) {
   652→            it->second.suspected_reason = reason;
   653→            it->second.non_execution_analysis = analysis;
   654→            it->second.current_status = DiagnosticStatus::NotReached;
   655→        }
   656→    }

   662→    /**
   663→     * Get summary for a specific diagnostic
   664→     */
   665→    std::optional<DiagnosticSummary> getDiagnosticSummary(const std::string& id) const {
   666→        std::shared_lock lock(m_mutex);
   667→        
   668→        auto it = m_diagnostics.find(id);
   669→        if (it != m_diagnostics.end()) {
   670→            return it->second;
   671→        }
   672→        return std::nullopt;
   673→    }
   674→    
   675→    /**
   676→     * Get all diagnostics that have never executed
   677→     */
   678→    std::vector<DiagnosticSummary> getNeverExecuted() const {
   679→        std::shared_lock lock(m_mutex);
   680→        
   681→        std::vector<DiagnosticSummary> result;
   682→        for (const auto& [id, summary] : m_diagnostics) {
   683→            if (!summary.hasEverExecuted() && 
   684→                summary.current_status != DiagnosticStatus::Removed) {
   685→                result.push_back(summary);
   686→            }
   687→        }
   688→        return result;
   689→    }
   690→    
   691→    /**
   692→     * Get all suspicious diagnostics (should be running but aren't)
   693→     */
   694→    std::vector<DiagnosticSummary> getSuspicious() const {
   695→        std::shared_lock lock(m_mutex);
   696→        
   697→        std::vector<DiagnosticSummary> result;
   698→        for (const auto& [id, summary] : m_diagnostics) {
   699→            if (summary.health == DiagnosticSummary::Health::Suspicious ||
   700→                summary.health == DiagnosticSummary::Health::Broken) {
   701→                result.push_back(summary);
   702→            }
   703→        }
   704→        return result;
   705→    }
   706→    
   707→    /**
   708→     * Generate quality report
   709→     */
   710→    DiagnosticQualityReport generateReport() const {
   711→        DiagnosticQualityReport report;
   712→        
   713→        std::shared_lock lock(m_mutex);
   714→        
   715→        report.total_diagnostics = m_diagnostics.size();
   716→        
   717→        for (const auto& [id, summary] : m_diagnostics) {
   718→            if (summary.current_status == DiagnosticStatus::Removed) continue;
   719→            
   720→            // Categorize
   721→            switch (summary.health) {
   722→                case DiagnosticSummary::Health::Healthy:
   723→                    report.healthy_count++;
   724→                    break;
   725→                case DiagnosticSummary::Health::Suspicious:
   726→                    report.suspicious_count++;
   727→                    report.suspicious_diagnostics.push_back(summary);
   728→                    break;
   729→                case DiagnosticSummary::Health::Broken:
   730→                    report.broken_count++;
   731→                    report.broken_diagnostics.push_back(summary);
   732→                    break;
   733→                default:
   734→                    report.unknown_count++;
   735→                    break;
   736→            }
   737→            
   738→            if (!summary.hasEverExecuted()) {
   739→                report.never_executed_count++;
   740→                report.never_executed_diagnostics.push_back(summary);
   741→            }
   742→            
   743→            // By subsystem
   744→            auto& sub_stats = report.by_subsystem[summary.definition.subsystem];
   745→            sub_stats.first++;  // Total
   746→            
   747→            if (summary.health == DiagnosticSummary::Health::Suspicious ||
   748→                summary.health == DiagnosticSummary::Health::Broken) {
   749→                sub_stats.second++;  // Problematic
   750→            }
   751→        }
   752→        
   753→        // Calculate coverage score
   754→        size_t executed = report.total_diagnostics - report.never_executed_count;
   755→        report.coverage_score = report.total_diagnostics > 0 ?
   756→            static_cast<double>(executed) / report.total_diagnostics : 1.0;
   757→        
   758→        // Generate findings and recommendations
   759→        generateFindings(report);
   760→        
   761→        return report;
   762→    }

   764→private:
   765→    Config m_config;
   766→    std::atomic<bool> m_enabled;
   767→    mutable std::shared_mutex m_mutex;
   768→    
   769→    std::map<std::string, DiagnosticSummary> m_diagnostics;
   770→    std::deque<DiagnosticExecution> m_executions;
   771→    
   772→    void generateFindings(DiagnosticQualityReport& report) const {
   773→        // Finding: Many diagnostics never execute
   774→        if (report.never_executed_count > report.total_diagnostics * 0.3) {
   775→            report.findings.push_back(
   776→                std::to_string(report.never_executed_count) + " of " +
   777→                std::to_string(report.total_diagnostics) + " diagnostics (" +
   778→                std::to_string(static_cast<int>((report.never_executed_count * 100.0) / report.total_diagnostics)) +
   779→                "%) have never executed");
   780→            report.recommendations.push_back(
   781→                "Review unreachable diagnostics - may indicate dead code or incorrect assumptions about control flow");
   782→        }
   783→        
   784→        // Finding: Broken diagnostics
   785→        if (report.broken_count > 0) {
   786→            report.findings.push_back(
   787→                std::to_string(report.broken_count) + " diagnostics are failing consistently");
   788→            report.recommendations.push_back(
   789→                "Investigate failing diagnostics - they may indicate real bugs or outdated assertions");
   790→        }
   791→        
   792→        // Finding: Subsystem-specific issues
   793→        for (const auto& [sub, stats] : report.by_subsystem) {
   794→            if (stats.second > 5) {  // More than 5 problematic in one subsystem
   795→                report.findings.push_back(
   796→                    "Subsystem " + subsystemToString(sub) + " has " +
   797→                    std::to_string(stats.second) + " problematic diagnostics");
   798→            }
   799→        }
   800→        
   801→        // General recommendations
   802→        if (report.coverage_score < 0.7) {
   803→            report.recommendations.insert(report.recommendations.begin(),
   804→                "Overall diagnostic coverage is low (" +
   805→                std::to_string(static_cast<int>(report.coverage_score * 100)) +
   806→                "%) - consider adding reachability markers to verify code paths");
   807→        }
   808→    }

   811→        std::stringstream ss;
   812→        ss << std::this_thread::get_id();
   813→        return ss.str();
   814→    }

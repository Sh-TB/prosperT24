     1→     1→/**
     2→     2→ * State Timeline System - Priority 3
     3→     3→ * 
     4→     4→ * PROBLEM SOLVED:
     5→     5→ * Crash location is usually not corruption location.
     6→     6→ * Need historical state to trace back to root cause.
     7→     7→ * 
     8→     8→ * EXAMPLE:
     9→     9→ *   Current: "Crash at frame 1300"
    10→    10→ *   Required: 
    11→    11→ *     Object created:  frame 200
    12→    12→ *     Modified:        frame 600
    13→    13→ *     Invalidated:     frame 900
    14→    14→ *     Used incorrectly: frame 1250
    15→    15→ *     Crash:           frame 1300
    16→    16→ * 
    17→    17→ * DESIGN:
    18→    18→ * - Records state changes with frame-level precision
    19→    19→ * - Tracks object/resource lifecycles
    20→    20→ * - Enables time-travel queries ("what was state at frame X?")
    21→    21→ * - Supports multiple timeline streams (GPU, memory, HLE, etc.)
    22→    22→ */

   144→   144→// ============================================================================
   145→   145→// Timeline Event
   146→   146→// ============================================================================
   147→   147→
   148→   148→struct TimelineEvent {
   149→   149→    std::string id;
   150→   150→    DebugTimestamp timestamp;
   151→   151→    uint64_t frame_number{0};
   152→   152→    TimelineEventType type;
   153→   153→    
   154→   154→    // What object/state?
   155→   155→    std::string object_id;       // Unique identifier for the object
   156→   156→    std::string object_type;     // Class/type of object
   157→   157→    std::string object_description;  // Human-readable description
   158→   158→    
   159→   159→    // What changed?
   160→   160→    std::string attribute_name;   // Which attribute changed
   161→   161→    std::string old_value;        // Previous value (JSON string)
   162→   162→    std::string new_value;        // New value (JSON string)
   163→   163→    
   164→   164→    // Provenance
   165→   165→    Subsystem subsystem{Subsystem::Unknown};
   166→   166→    SourceLocation source;
   167→   167→    std::string thread_id;
   168→   168→    
   169→   169→    // Additional data
   170→   170→    std::map<std::string, std::string> metadata;
   171→   171→    
   172→   172→    // Relationships
   173→   173→    std::string parent_event_id;  // Causal parent
   174→   174→    std::vector<std::string> related_event_ids;
   175→   175→    
   176→   176→    static std::string generateId() {
   177→   177→        static std::atomic<uint64_t> counter{0};
   178→   178→        auto now = std::chrono::high_resolution_clock::now().time_since_epoch().count();
   179→   179→        return "tle_" + std::to_string(now) << "_" << std::to_string(++counter);
   180→   180→    }
   181→   181→    
   182→   182→    TimelineEvent() {
   183→   183→        id = generateId();
   184→   184→        timestamp = DebugTimestamp::now();
   185→   185→    }
   186→   186→    
   187→   187→    bool isErrorOrCorruption() const {
   188→   188→        return type == TimelineEventType::ErrorDetected ||
   189→   189→               type == TimelineEventType::CorruptionDetected ||
   190→   190→               type == TimelineEventType::AssertionFailed ||
   191→   191→               type == TimelineEventType::AnomalyDetected;
   192→   192→    }
   193→   193→    
   194→   194→    bool isLifecycleEvent() const {
   195→   195→        return type == TimelineEventType::ObjectCreated ||
   196→   196→               type == TimelineEventType::ObjectDestroyed ||
   197→   197→               type == TimelineEventType::ResourceAllocated ||
   198→   198→               type == TimelineEventType::ResourceFreed;
   199→   199→    }
   200→   200→    
   201→   201→    std::string toJson() const {
   202→   202→        std::stringstream json;
   203→   203→        json << "{\n";
   204→   204→        json << "  \"id\": \"" << JsonUtils::escapeJson(id) << "\",\n";
   205→   205→        json << "  \"timestamp\": \"" << timestamp.toISO8601() << "\",\n";
   206→   206→        json << "  \"frame\": " << frame_number << ",\n";
   207→   207→        json << "  \"type\": \"" << timelineEventTypeToString(type) << "\",\n";
   208→   208→        json << "  \"object_id\": \"" << JsonUtils::escapeJson(object_id) << "\",\n";
   209→   209→        json << "  \"object_type\": \"" << JsonUtils::escapeJson(object_type) << "\",\n";
   210→   210→        if (!object_description.empty()) {
   211→   211→            json << "  \"description\": \"" << JsonUtils::escapeJson(object_description) << "\",\n";
   212→   212→        }
   213→   213→        if (!attribute_name.empty()) {
   214→   214→            json << "  \"attribute\": \"" << JsonUtils::escapeJson(attribute_name) << "\",\n";
   215→   215→            json << "  \"old_value\": \"" << JsonUtils::escapeJson(old_value) << "\",\n";
   216→   216→            json << "  \"new_value\": \"" << JsonUtils::escapeJson(new_value) << "\",\n";
   217→   217→        }
   218→   218→        json << "  \"subsystem\": \"" << subsystemToString(subsystem) << "\"\n";
   219→   219→        json << "}";
   220→   220→        return json.str();
   221→   221→    }
   222→   222→};

   225→   225→// Object Lifecycle Tracker
   226→   226→// ============================================================================
   227→   227→
   228→   228→struct ObjectLifecycle {
   229→   229→    std::string object_id;
   230→   230→    std::string object_type;
   231→   231→    
   232→   232→    // Key lifecycle events
   233→   233→    std::optional<TimelineEvent> created;
   234→   234→    std::optional<TimelineEvent> destroyed;
   235→   235→    std::vector<TimelineEvent> modifications;
   236→   236→    
   237→   237→    // Current state (reconstructed from events)
   238→   238→    std::map<std::string, std::string> current_state;
   239→   239→    
   240→   240→    // Statistics
   241→   241→    uint64_t creation_frame{0};
   242→   242→    uint64_t destruction_frame{0};  // 0 if still alive
   243→   243→    size_t total_modifications{0};
   244→   244→    uint64_t last_access_frame{0};
   245→   245→    
   246→   246→    bool isAlive() const { return !destroyed.has_value(); }
   247→   247→    uint64_t lifetime() const { 
   248→   248→        return destruction_frame > 0 ? destruction_frame - creation_frame : 0;
   249→   249→    }
   250→   250→    
   251→   251→    std::string toJson() const {
   252→   252→        std::stringstream json;
   253→   253→        json << "{\n";
   254→   254→        json << "  \"object_id\": \"" << JsonUtils::escapeJson(object_id) << "\",\n";
   255→   255→        json << "  \"object_type\": \"" << JsonUtils::escapeJson(object_type) << "\",\n";
   256→   256→        json << "  \"alive\": " << (isAlive() ? "true" : "false") << ",\n";
   257→   257→        json << "  \"created_frame\": " << creation_frame << ",\n";
   258→   258→        json << "  \"modifications\": " << total_modifications << ",\n";
   259→   259→        json << "  \"lifetime_frames\": " << lifetime() << "\n";
   260→   260→        json << "}";
   261→   261→        return json.str();
   262→   262→    }
   263→   263→};

   303→   303→            json << "    ... and " << (events.size() - 100) << " more\n";

   305→   305→        json << "  ]\n";
   306→   306→        
   307→   307→        json << "}\n";

   344→   344→class StateTimelineSystem {
   345→   345→public:
   346→   346→    struct Config {
   347→   347→        bool enabled{true};
   348→   348→        size_t max_events{1000000};          // Global event limit
   349→   349→        size_t max_events_per_object{10000}; // Per-object limit
   350→   350→        bool track_all_objects{false};       // Track everything or just watched
   351→   351→        bool auto_track_gpu{true};           // Auto-track GPU resources
   352→   352→        bool auto_track_memory{true};        // Auto-track major allocations
   353→   353→        bool record_full_state_changes{false}; // Record old+new values (expensive)
   354→   354→        
   355→   355→        static Config minimal() {
   356→   356→            return {true, 100000, 1000, false, false, false, false};
   357→   357→        }
   358→   358→        
   359→   359→        static Config comprehensive() {
   360→   360→            return {true, 10000000, 50000, true, true, true, true};
   361→   361→        }

   363→   363→    
   364→   364→    explicit StateTimelineSystem(const Config& config = Config())

   367→   367→    // ========================================================================
   368→   368→    // Control Interface

   370→   370→    
   371→   371→    void enable() { m_enabled.store(true); }

   374→   374→    
   375→   375→    void clear() {
   376→   376→        std::unique_lock lock(m_mutex);
   377→   377→        m_events.clear();
   378→   378→        m_object_lifecycles.clear();
   379→   379→        m_watch_list.clear();
   380→   380→        m_current_frame = 0;

   383→   383→    // ========================================================================
   384→   384→    // Frame Tracking

   387→   387→    /**
   388→   388→     * Advance to next frame (call once per emulated frame)
   389→   389→     */
   390→   390→    void advanceFrame() {
   391→   391→        if (!isEnabled()) return;
   392→   392→        
   393→   393→        ++m_current_frame;
   394→   394→        
   395→   395→        // Optionally emit frame marker
   396→   396→        if (m_config.track_all_objects) {
   397→   397→            TimelineEvent marker;
   398→   398→            marker.type = TimelineEventType::FrameMarker;
   399→   399→            marker.frame_number = m_current_frame;
   400→   400→            marker.description = "Frame " + std::to_string(m_current_frame);
   401→   401→            
   402→   402→            addEventInternal(marker);
   403→   403→        }
   404→   404→    }
   405→   405→    
   406→   406→    void setFrame(uint64_t frame) {
   407→   407→        m_current_frame = frame;
   408→   408→    }
   409→   409→    
   410→   410→    uint64_t getCurrentFrame() const { return m_current_frame.load(); }

   430→   430→        std::shared_lock lock(m_mutex);

   492→   492→        Subsystem subs = Subsystem::Unknown,
   493→   493→        const SourceLocation& loc = DEBUG_HERE()) {
   494→   494→

   496→   496→        if (!shouldTrack(object_id)) return;
   497→   497→

   585→   585→    /**
   586→   586→     * Get all events in a frame range
   587→   587→     */
   588→   588→    TimelineQueryResult queryByFrameRange(uint64_t start_frame, uint64_t end_frame) const {
   589→   589→        TimelineQueryResult result;
   590→   590→        result.start_frame = start_frame;
   591→   591→        result.end_frame = end_frame;
   592→   592→        
   593→   593→        std::shared_lock lock(m_mutex);
   594→   594→        
   595→   595→        for (const auto& event : m_events) {
   596→   596→            if (event.frame_number >= start_frame && event.frame_number <= end_frame) {
   597→   597→                result.events.push_back(event);
   598→   598→                
   599→   599→                if (event.isErrorOrCorruption()) result.error_events++;
   600→   600→                if (event.isLifecycleEvent()) result.lifecycle_events++;
   601→   601→                result.total_events++;
   602→   602→            }
   603→   603→        }
   604→   604→        
   605→   605→        return result;
   606→   606→    }

   608→   608→    /**
   609→   609→     * Get complete lifecycle of an object
   610→   610→     */
   611→   611→    std::optional<ObjectLifecycle> getObjectLifecycle(const std::string& object_id) const {
   612→   612→        std::shared_lock lock(m_mutex);
   613→   613→        
   614→   614→        auto it = m_object_lifecycles.find(object_id);
   615→   615→        if (it != m_object_lifecycles.end()) {
   616→   616→            return it->second;
   617→   617→        }
   618→   618→        return std::nullopt;
   619→   619→    }

   621→   621→    /**
   622→   622→     * Get state snapshot at a specific frame
   623→   623→     */
   624→   624→    StateSnapshot getSnapshotAtFrame(uint64_t target_frame) const {
   625→   625→        StateSnapshot snapshot;
   626→   626→        snapshot.target_frame = target_frame;
   627→   627→        
   628→   628→        std::shared_lock lock(m_mutex);
   629→   629→        
   630→   630→        // Find all objects that existed at this frame
   631→   631→        for (const auto& [obj_id, lifecycle] : m_object_lifecycles) {
   632→   632→            if (lifecycle.creation_frame <= target_frame &&
   633→   633→                (lifecycle.destruction_frame == 0 || lifecycle.destruction_frame > target_frame)) {
   634→   634→                snapshot.active_objects[obj_id] = lifecycle;
   635→   635→            }
   636→   636→        }
   637→   637→        
   638→   638→        // Get recent events leading up to this frame
   639→   639→        int count = 0;
   640→   640→        for (auto it = m_events.rbegin(); it != m_events.rend() && count < 50; ++it) {
   641→   641→            if (it->frame_number <= target_frame) {
   642→   642→                snapshot.preceding_events.push_back(*it);
   643→   643→                count++;
   644→   644→                
   645→   645→                if (it->isErrorOrCorruption()) {
   646→   646→                    snapshot.prior_errors.push_back(*it);
   647→   647→                }
   648→   648→            }
   649→   649→        }
   650→   650→        
   651→   651→        return snapshot;
   652→   652→    }

   654→   654→    /**
   655→   655→     * Find what happened to an object around a specific frame
   656→   656→     */
   657→   657→    std::vector<TimelineEvent> getObjectEventsAroundFrame(
   658→   658→        const std::string& object_id,
   659→   659→        uint64_t center_frame,
   660→   660→        uint64_t window_size = 100) const {
   661→   661→        
   662→   662→        std::vector<TimelineEvent> result;
   663→   663→        uint64_t start = window_size > center_frame ? 0 : center_frame - window_size;
   664→   664→        uint64_t end = center_frame + window_size;
   665→   665→        
   666→   666→        std::shared_lock lock(m_mutex);
   667→   667→        
   668→   668→        for (const auto& event : m_events) {
   669→   669→            if (event.object_id == object_id &&
   670→   670→                event.frame_number >= start &&
   671→   671→                event.frame_number <= end) {
   672→   672→                result.push_back(event);
   673→   673→            }
   674→   674→        }
   675→   675→        
   676→   676→        return result;
   677→   677→    }

   682→   682→    std::vector<TimelineEvent> findErrorsInFrameRange(

   701→   701→    /**
   702→   702→     * Find objects that were destroyed shortly before a crash
   703→   703→     * (Common pattern: use-after-free)
   704→   704→     */
   705→   705→    std::vector<ObjectLifetime> findObjectsDestroyedBeforeFrame(
   706→   706→        uint64_t crash_frame,
   707→   707→        uint64_t lookback_window = 1000) const {
   708→   708→        
   709→   709→        struct ObjectLifetime {
   710→   710→            std::string object_id;
   711→   711→            std::string object_type;
   712→   712→            uint64_t destruction_frame;
   713→   713→            uint64_t frames_before_crash;
   714→   714→        };
   715→   715→        
   716→   716→        std::vector<ObjectLifetime> recently_destroyed;
   717→   717→        uint64_t start = lookback_window > crash_frame ? 0 : crash_frame - lookback_window;
   718→   718→        
   719→   719→        std::shared_lock lock(m_mutex);
   720→   720→        
   721→   721→        for (const auto& [obj_id, lifecycle] : m_object_lifecycles) {
   722→   722→            if (lifecycle.destruction_frame > 0 &&
   723→   723→                lifecycle.destruction_frame >= start &&
   724→   724→                lifecycle.destruction_frame < crash_frame) {
   725→   725→                
   726→   726→                ObjectLifetime lt;
   727→   727→                lt.object_id = obj_id;
   728→   728→                lt.object_type = lifecycle.object_type;
   729→   729→                lt.destruction_frame = lifecycle.destruction_frame;
   730→   730→                lt.frames_before_crash = crash_frame - lifecycle.destruction_frame;
   731→   731→                
   732→   732→                recently_destroyed.push_back(lt);
   733→   733→            }
   734→   734→        }
   735→   735→        
   736→   736→        // Sort by proximity to crash
   737→   737→        std::sort(recently_destroyed.begin(), recently_destroyed.end(),
   738→   738→            [](const ObjectLifetime& a, const ObjectLifetime& b) {
   739→   739→                return a.frames_before_crash < b.frames_before_crash;
   740→   740→            });
   741→   741→        
   742→   742→        return recently_destroyed;
   743→   743→    }

   749→   749→    struct Stats {
   750→   750→        size_t total_events{0};
   751→   751→        uint64_t current_frame{0};
   752→   752→        size_t tracked_objects{0};
   753→   753→        size_t watched_objects{0};
   754→   754→        size_t active_objects{0};
   755→   755→        size_t total_errors{0};
   756→   756→        size_t total_corruptions{0};
   757→   757→        
   758→   758→        std::map<TimelineEventType, size_t> events_by_type;
   759→   759→        std::map<Subsystem, size_t> events_by_subsystem;
   760→   760→    };
   761→   761→    
   762→   762→    Stats getStats() const {
   763→   763→        std::shared_lock lock(m_mutex);
   764→   764→        
   765→   765→        Stats stats;
   766→   766→        stats.current_frame = m_current_frame;
   767→   767→        stats.tracked_objects = m_object_lifecycles.size();
   768→   768→        stats.watched_objects = m_watch_list.size();
   769→   769→        
   770→   770→        for (const auto& [obj_id, lifecycle] : m_object_lifecycles) {
   771→   771→            if (lifecycle.isAlive()) stats.active_objects++;
   772→   772→        }
   773→   773→        
   774→   774→        for (const auto& event : m_events) {
   775→   775→            stats.total_events++;
   776→   776→            stats.events_by_type[event.type]++;
   777→   777→            stats.events_by_subsystem[event.subsystem]++;
   778→   778→            
   779→   779→            if (event.type == TimelineEventType::ErrorDetected) stats.total_errors++;
   780→   780→            if (event.type == TimelineEventType::CorruptionDetected) stats.total_corruptions++;
   781→   781→        }
   782→   782→        
   783→   783→        return stats;
   784→   784→    }

   805→   805→    bool shouldTrack(const std::string& object_id) const {
   806→   806→        if (m_config.track_all_objects) return true;
   807→   807→        return m_watch_list.count(object_id) > 0;

   810→   810→    void addEventInternal(const TimelineEvent& event) {
   811→   811→        std::unique_lock lock(m_mutex);
   812→   812→        
   813→   813→        // Check global limit
   814→   814→        if (m_events.size() >= m_config.max_events) {
   815→   815→            m_events.pop_front();  // Evict oldest
   816→   816→        }
   817→   817→        
   818→   818→        m_events.push_back(event);
   819→   819→        
   820→   820→        // Update object lifecycle tracking
   821→   821→        updateObjectLifecycle(event);
   822→   822→    }
   823→   823→    
   824→   824→    void updateObjectLifecycle(const TimelineEvent& event) {
   825→   825→        // Get or create lifecycle entry
   826→   826→        auto& lifecycle = m_object_lifecycles[event.object_id];
   827→   827→        
   828→   828→        if (lifecycle.object_id.empty()) {
   829→   829→            lifecycle.object_id = event.object_id;
   830→   830→            lifecycle.object_type = event.object_type;
   831→   831→        }
   832→   832→        
   833→   833→        switch (event.type) {
   834→   834→            case TimelineEventType::ObjectCreated:
   835→   835→            case TimelineEventType::ResourceAllocated:
   836→   836→                if (!lifecycle.created.has_value()) {
   837→   837→                    lifecycle.created = event;
   838→   838→                    lifecycle.creation_frame = event.frame_number;
   839→   839→                }
   840→   840→                break;
   841→   841→                
   842→   842→            case TimelineEventType::ObjectDestroyed:
   843→   843→            case TimelineEventType::ResourceFreed:
   844→   844→                if (!lifecycle.destroyed.has_value()) {
   845→   845→                    lifecycle.destroyed = event;
   846→   846→                    lifecycle.destruction_frame = event.frame_number;
   847→   847→                }
   848→   848→                break;
   849→   849→                
   850→   850→            case TimelineEventType::ObjectModified:
   851→   851→            case TimelineEventType::StateChanged:
   852→   852→                lifecycle.modifications.push_back(event);
   853→   853→                lifecycle.total_modifications++;
   854→   854→                
   855→   855→                // Update current state
   856→   856→                if (!event.attribute_name.empty()) {
   857→   857→                    lifecycle.current_state[event.attribute_name] = event.new_value;
   858→   858→                }
   859→   859→                break;
   860→   860→                
   861→   861→            default:
   862→   862→                break;
   863→   863→        }
   864→   864→        
   865→   865→        // Update last access
   866→   866→        if (event.frame_number > lifecycle.last_access_frame) {
   867→   867→            lifecycle.last_access_frame = event.frame_number;
   868→   868→        }
   869→   869→    }

   872→   872→        std::stringstream ss;
   873→   873→        ss << std::this_thread::get_id();
   874→   874→        return ss.str();
   875→   875→    }

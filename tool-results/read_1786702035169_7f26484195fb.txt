    36→enum class ExperimentStatus : uint8_t {
    37→    Running,
    38→    Completed,
    39→    Failed,
    40→    Abandoned,
    41→    Superseded

    44→inline std::string experimentStatusToString(ExperimentStatus status) {
    45→    switch (status) {
    46→        case ExperimentStatus::Running: return "Running";
    47→        case ExperimentStatus::Completed: return "Completed";
    48→        case ExperimentStatus::Failed: return "Failed";
    49→        case ExperimentStatus::Abandoned: return "Abandoned";
    50→        case ExperimentStatus::Superseded: return "Superseded";
    51→    }
    52→    return "Unknown";
    53→}

    55→struct ExperimentRecord {
    56→    std::string id;
    57→    std::string title;
    58→    std::string description;
    59→    
    60→    // Timing
    61→    DebugTimestamp created_at;
    62→    DebugTimestamp updated_at;
    63→    DebugTimestamp completed_at;
    64→    
    65→    ExperimentStatus status{ExperimentStatus::Running};
    66→    
    67→    // Context
    68→    std::string commit_hash;
    69→    std::string branch_name;
    70→    std::string issue_reference;  // GitHub issue, JIRA, etc.
    71→    
    72→    // Build environment
    73→    std::map<std::string, std::string> build_info;
    74→    
    75→    // Results
    76→    std::string result_summary;
    77→    std::string root_cause;  // If found
    78→    bool root_cause_found{false};
    79→    
    80→    // Classification
    81→    std::vector<std::string> tags;
    82→    std::vector<std::string> affected_components;
    83→    Severity severity{Severity::Info};
    84→    
    85→    // Related items
    86→    std::vector<std::string> hypothesis_ids;
    87→    std::vector<std::string> evidence_ids;
    88→    std::string superseded_by;  // If this was superseded by another experiment
    89→    
    90→    // Package location
    91→    fs::path package_path;
    92→    
    93→    static std::string generateId() {
    94→        static std::atomic<uint64_t> counter{0};
    95→        auto now = std::chrono::high_resolution_clock::now().time_since_epoch().count();
    96→        return "exp_" + std::to_string(now) + "_" + std::to_string(++counter);
    97→    }
    98→    
    99→    ExperimentRecord() {
   100→        id = generateId();
   101→        created_at = DebugTimestamp::now();
   102→        updated_at = created_at;
   103→    }
   104→    
   105→    void markCompleted(const std::string& result = "") {
   106→        status = ExperimentStatus::Completed;
   107→        completed_at = DebugTimestamp::now();
   108→        updated_at = completed_at;
   109→        if (!result.empty()) result_summary = result;
   110→    }
   111→    
   112→    void markFailed(const std::string& reason = "") {
   113→        status = ExperimentStatus::Failed;
   114→        updated_at = DebugTimestamp::now();
   115→        if (!reason.empty()) result_summary = "Failed: " + reason;
   116→    }
   117→    
   118→    void markAbandoned() {
   119→        status = ExperimentStatus::Abandoned;
   120→        updated_at = DebugTimestamp::now();
   121→    }
   122→    
   123→    std::string toJson() const {
   124→        std::stringstream json;
   125→        json << "{\n";
   126→        json << "  \"id\": \"" << JsonUtils::escapeJson(id) << "\",\n";
   127→        json << "  \"title\": \"" << JsonUtils::escapeJson(title) << "\",\n";
   128→        json << "  \"status\": \"" << experimentStatusToString(status) << "\",\n";
   129→        json << "  \"created\": \"" << created_at.toISO8601() << "\",\n";
   130→        if (!commit_hash.empty()) {
   131→            json << "  \"commit\": \"" << JsonUtils::escapeJson(commit_hash) << "\",\n";
   132→        }
   133→        if (!result_summary.empty()) {
   134→            json << "  \"result\": \"" << JsonUtils::escapeJson(result_summary) << "\",\n";
   135→        }
   136→        if (root_cause_found) {
   137→            json << "  \"root_cause\": \"" << JsonUtils::escapeJson(root_cause) << "\"\n";
   138→        }
   139→        
   140→        json << "}";
   141→        return json.str();
   142→    }

   211→        json << "  \"supporting_evidence\": " << supporting_evidence_ids.size() << ",\n";
   212→        json << "  \"refuting_evidence\": " << refuting_evidence_ids.size() << "\n";
   213→        json << "}";

   222→struct UpstreamFixRecord {
   223→    std::string id;
   224→    
   225→    // Identification
   226→    std::string pr_url;

   234→    
   235→    // Status
   236→    bool is_merged{false};
   237→    std::string merged_in_version;
   238→    DebugTimestamp merged_at;
   239→    
   240→    // Applicability
   241→    std::vector<std::string> affected_components;
   242→    std::vector<std::string> keywords;
   243→    std::vector<std::string> related_issues;
   244→    
   245→    // Detection metadata
   246→    DebugTimestamp added_to_database;
   247→    std::string source;  // Manual, automated, etc.
   248→    
   249→    /**
   250→     * Check relevance to a query
   251→     */
   252→    double relevanceToQuery(const std::string& query) const {
   253→        std::string combined = pr_title + " " + description + " " + fixed_issue_description + " ";
   254→        for (const auto& kw : keywords) combined += kw + " ";
   255→        for (const auto& comp : affected_components) combined += comp + " ";
   256→        
   257→        std::string lower_query = query;
   258→        std::transform(lower_query.begin(), lower_query.end(), lower_query.begin(), ::tolower);
   259→        
   260→        std::string lower_combined = combined;
   261→        std::transform(lower_combined.begin(), lower_combined.end(), lower_combined.begin(), ::tolower);
   262→        
   263→        // Direct match
   264→        if (lower_combined.find(lower_query) != std::string::npos) {
   265→            return 0.95;
   266→        }
   267→        
   268→        // Word-level matching
   269→        std::vector<std::string> query_words = splitWords(lower_query);
   270→        int matches = 0;
   271→        for (const auto& word : query_words) {
   272→            if (lower_combined.find(word) != std::string::npos) {
   273→                matches++;
   274→            }
   275→        }
   276→        
   277→        if (!query_words.empty()) {
   278→            return static_cast<double>(matches) / query_words.size() * 0.8;
   279→        }
   280→        
   281→        return 0.0;
   282→    }
   283→
   284→private:
   285→    static std::vector<std::string> splitWords(const std::string& text) {
   286→        std::vector<std::string> words;
   287→        std::stringstream ss(text);
   288→        std::string word;
   289→        while (ss >> word) {
   290→            if (word.length() > 2) words.push_back(word);
   291→        }
   292→        return words;
   293→    }
   294→};

   586→    /**
   587→     * Find relevant upstream fixes for a query
   588→     */
   589→    std::vector<UpstreamFixRecord> findRelevantUpstreamFixes(
   590→        const std::string& query,
   591→        double min_relevance = 0.5,
   592→        size_t limit = 10) const {
   593→        
   594→        std::shared_lock lock(m_mutex);
   595→        
   596→        std::vector<std::pair<double, UpstreamFixRecord>> scored;
   597→        
   598→        for (const auto& [id, fix] : m_upstream_fixes) {
   599→            double relevance = fix.relevanceToQuery(query);
   600→            if (relevance >= min_relevance) {
   601→                scored.push_back({relevance, fix});
   602→            }
   603→        }
   604→        
   605→        // Sort by relevance
   606→        std::sort(scored.begin(), scored.end(),
   607→            [](const auto& a, const auto& b) { return a.first > b.first; });
   608→        
   609→        // Extract top results
   610→        std::vector<UpstreamFixRecord> result;
   611→        for (size_t i = 0; i < scored.size() && i < limit; ++i) {
   612→            result.push_back(scored[i].second);
   613→        }
   614→        
   615→        return result;
   616→    }

   622→    struct SearchQuery {
   623→        std::string text;
   624→        std::vector<std::string> tags;
   625→        std::optional<ExperimentStatus> status_filter;
   626→        std::optional<InvestigationStatus> hypothesis_status_filter;
   627→        int max_results{20};
   628→        double min_relevance{0.3};
   629→        
   630→        static SearchQuery forText(const std::string& text, int max_results = 20) {
   631→            return {text, {}, std::nullopt, std::nullopt, max_results, 0.3};
   632→        }
   633→        
   634→        static SearchQuery forTags(const std::vector<std::string>& tags) {
   635→            return {"", tags, std::nullopt, std::nullopt, 50, 0.0};
   636→        }
   637→    };
   638→    
   639→    /**
   640→     * Search across all data
   641→     */
   642→    std::vector<SearchResult> search(const SearchQuery& query) const {
   643→        std::vector<SearchResult> results;
   644→        
   645→        // Search experiments
   646→        {
   647→            std::shared_lock lock(m_mutex);
   648→            
   649→            for (const auto& [id, exp] : m_experiments) {
   650→                double score = 0.0;
   651→                std::string matched;
   652→                
   653→                if (!query.text.empty()) {
   654→                    score = calculateTextSimilarity(query.text, exp.title + " " + exp.description);
   655→                    if (score > 0) matched = "text";
   656→                }
   657→                
   658→                // Tag matching boosts score
   659→                for (const auto& tag : query.tags) {
   660→                    auto it = std::find(exp.tags.begin(), exp.tags.end(), tag);
   661→                    if (it != exp.tags.end()) {
   662→                        score = std::max(score, 0.8);
   663→                        if (!matched.empty()) matched += "+tag:" + tag;
   664→                        else matched = "tag:" + tag;
   665→                    }
   666→                }
   667→                
   668→                // Status filter
   669→                if (query.status_filter.has_value() && exp.status != *query.status_filter) {
   670→                    score = 0.0;
   671→                }
   672→                
   673→                if (score >= query.min_relevance) {
   674→                    SearchResult sr;
   675→                    sr.relevance_score = score;
   676→                    sr.matched_fields = matched;
   677→                    sr.experiment = exp;
   678→                    results.push_back(sr);
   679→                }
   680→            }
   681→        }
   682→        
   683→        // Sort by relevance
   684→        std::sort(results.begin(), results.end(), std::greater<SearchResult>());
   685→        
   686→        // Limit results
   687→        if (results.size() > static_cast<size_t>(query.max_results)) {
   688→            results.resize(query.max_results);
   689→        }
   690→        
   691→        return results;
   692→    }
   693→    
   694→    /**
   695→     * Simple text search
   696→     */
   697→    std::vector<SearchResult> searchText(const std::string& query, int max_results = 20) const {
   698→        return search(SearchQuery::forText(query, max_results));

   705→    /**
   706→     * Check if a new experiment would duplicate existing work
   707→     */
   708→    std::vector<DuplicateWarning> checkForDuplicates(
   709→        const std::string& title,
   710→        const std::vector<std::string>& tags = {},
   711→        double threshold = 0.7) const {
   712→        
   713→        std::vector<DuplicateWarning> warnings;
   714→        
   715→        std::shared_lock lock(m_mutex);
   716→        
   717→        for (const auto& [id, exp] : m_experiments) {
   718→            double similarity = calculateTextSimilarity(title, exp.title);
   719→            
   720→            // Boost for matching tags
   721→            if (!tags.empty()) {
   722→                size_t matching_tags = 0;
   723→                for (const auto& tag : tags) {
   724→                    if (std::find(exp.tags.begin(), exp.tags.end(), tag) != exp.tags.end()) {
   725→                        matching_tags++;
   726→                    }
   727→                }
   728→                
   729→                if (!exp.tags.empty()) {
   730→                    double tag_ratio = static_cast<double>(matching_tags) / tags.size();
   731→                    similarity = std::max(similarity, tag_ratio * 0.8);
   732→                }
   733→            }
   734→            
   735→            if (similarity >= threshold) {
   736→                DuplicateWarning warn;
   737→                warn.existing_id = id;
   738→                warn.existing_title = exp.title;
   739→                warn.existing_status = exp.status;
   740→                warn.similarity_score = similarity;
   741→                
   742→                if (similarity >= 0.9) {
   743→                    warn.reason = "Near-exact match to existing investigation";
   744→                    warn.recommendation = "Review existing experiment before proceeding.";
   745→                } else if (exp.status == ExperimentStatus::Completed) {
   746→                    warn.reason = "Similar to completed investigation";
   747→                    warn.recommendation = "Check if this experiment already solved your issue.";
   748→                } else if (exp.status == ExperimentStatus::Running) {
   749→                    warn.reason = "Similar investigation currently in progress";
   750→                    warn.recommendation = "Coordinate with ongoing investigation or wait for results.";
   751→                } else {
   752→                    warn.reason = "Similar topic previously investigated";
   753→                    warn.recommendation = "Review findings before starting new work.";
   754→                }
   755→                
   756→                warnings.push_back(warn);
   757→            }
   758→        }
   759→        
   760→        return warnings;
   761→    }
   762→    
   763→    /**
   764→     * Quick duplicate check
   765→     */
   766→    std::pair<bool, std::optional<DuplicateWarning>> isDuplicate(
   767→        const std::string& title,
   768→        const std::vector<std::string>& tags = {}) const {
   769→        
   770→        auto warnings = checkForDuplicates(title, tags, 0.8);
   771→        
   772→        if (warnings.empty()) {
   773→            return {false, std::nullopt};
   774→        }
   775→        
   776→        // Return highest similarity
   777→        auto max_it = std::max_element(warnings.begin(), warnings.end(),
   778→            [](const DuplicateWarning& a, const DuplicateWarning& b) {
   779→                return a.similarity_score < b.similarity_score;
   780→            });
   781→        
   782→        return {true, *max_it};
   783→    }

   871→    double calculateTextSimilarity(const std::string& a, const std::string& b) const {
   872→        // Bigram overlap similarity
   873→        auto bigrams = [](const std::string& s) -> std::set<std::string> {
   874→            std::set<std::string> result;
   875→            std::string lower = s;
   876→            std::transform(lower.begin(), lower.end(), lower.begin(), ::tolower);
   877→            
   878→            for (size_t i = 0; i + 1 < lower.length(); ++i) {
   879→                result.insert(lower.substr(i, 2));
   880→            }
   881→            return result;
   882→        };
   883→        
   884→        auto bgA = bigrams(a);
   885→        auto bgB = bigrams(b);
   886→        
   887→        if (bgA.empty() && bgB.empty()) return 1.0;
   888→        if (bgA.empty() || bgB.empty()) return 0.0;
   889→        
   890→        std::set<std::string> intersection;
   891→        std::set_intersection(bgA.begin(), bgA.end(), bgB.begin(), bgB.end(),
   892→                             std::inserter(intersection, intersection.begin()));
   893→        
   894→        std::set<std::string> union_set;
   895→        std::set_union(bgA.begin(), bgA.end(), bgB.begin(), bgB.end(),
   896→                  std::inserter(union_set, union_set.begin()));
   897→        
   898→        return static_cast<double>(intersection.size()) / union_set.size();
   899→    }

    34→enum class ExperimentStatus : uint8_t {
    35→    Running,
    36→    Completed,
    37→    Failed,
    38→    Abandoned,
    39→    Superseded

    42→inline std::string experimentStatusToString(ExperimentStatus status) {
    43→    switch (status) {
    44→        case ExperimentStatus::Running: return "Running";
    45→        case ExperimentStatus::Completed: return "Completed";
    46→        case ExperimentStatus::Failed: return "Failed";
    47→        case ExperimentStatus::Abandoned: return "Abandoned";
    48→        case ExperimentStatus::Superseded: return "Superseded";
    49→    }
    50→    return "Unknown";
    51→}

    64→    
    65→    // Context
    66→    std::string commit_hash;

    69→    
    70→    // Build environment
    71→    std::map<std::string, std::string> build_info;

    73→    // Results
    74→    std::string result_summary;
    75→    std::string root_cause;  // If found

    77→

    80→    std::vector<std::string> affected_components;
    81→    Severity severity{Severity::Info};

    83→    // Related items

    87→    
    88→    // Package location
    89→    fs::path package_path;
    90→    
    91→    static std::string generateId() {
    92→        static std::atomic<uint64_t> counter{0};
    93→        auto now = std::chrono::high_resolution_clock::now().time_since_epoch().count();
    94→        return "exp_" + std::to_string(now) + "_" + std::to_string(++counter);
    95→    }
    96→    
    97→    ExperimentRecord() {
    98→        id = generateId();
    99→        created_at = DebugTimestamp::now();
   100→        updated_at = created_at;

   103→    void markCompleted(const std::string& result = "") {
   104→        status = ExperimentStatus::Completed;
   105→        completed_at = DebugTimestamp::now();
   106→        updated_at = completed_at;
   107→        if (!result.empty()) result_summary = result;
   108→    }
   109→    
   110→    void markFailed(const std::string& reason = "") {
   111→        status = ExperimentStatus::Failed;
   112→        updated_at = DebugTimestamp::now();
   113→        if (!reason.empty()) result_summary = "Failed: " + reason;
   114→    }
   115→    
   116→    void markAbandoned() {
   117→        status = ExperimentStatus::Abandoned;
   118→        updated_at = DebugTimestamp::now();
   119→    }
   120→    
   121→    std::string toJson() const {
   122→        std::stringstream json;
   123→        json << "{\n";
   124→        json << "  \"id\": \"" << JsonUtils::escapeJson(id) << "\",\n";
   125→        json << "  \"title\": \"" << JsonUtils::escapeJson(title) << "\",\n";
   126→        json << "  \"status\": \"" << experimentStatusToString(status) << "\",\n";
   127→        json << "  \"created\": \"" << created_at.toISO8601() << "\",\n";
   128→        if (!commit_hash.empty()) {
   129→            json << "  \"commit\": \"" << JsonUtils::escapeJson(commit_hash) << "\",\n";
   130→        }
   131→        if (!result_summary.empty()) {
   132→            json << "  \"result\": \"" << JsonUtils::escapeJson(result_summary) << "\",\n";
   133→        }
   134→        if (root_cause_found) {
   135→            json << "  \"root_cause\": \"" << JsonUtils::escapeJson(root_cause) << "\"\n";

   138→        json << "}";

   245→    std::string source;  // Manual, automated, etc.
   246→    
   247→    /**
   248→     * Check relevance to a query
   249→     */
   250→    double relevanceToQuery(const std::string& query) const {
   251→        std::string combined = pr_title + " " + description + " " + fixed_issue_description + " ";
   252→        for (const auto& kw : keywords) combined += kw + " ";
   253→        for (const auto& comp : affected_components) combined += comp + " ";
   254→        
   255→        std::string lower_query = query;
   256→        std::transform(lower_query.begin(), lower_query.end(), lower_query.begin(), ::tolower);
   257→        
   258→        std::string lower_combined = combined;
   259→        std::transform(lower_combined.begin(), lower_combined.end(), lower_combined.begin(), ::tolower);
   260→        
   261→        // Direct match
   262→        if (lower_combined.find(lower_query) != std::string::npos) {
   263→            return 0.95;
   264→        }
   265→        
   266→        // Word-level matching
   267→        std::vector<std::string> query_words = splitWords(lower_query);
   268→        int matches = 0;
   269→        for (const auto& word : query_words) {
   270→            if (lower_combined.find(word) != std::string::npos) {
   271→                matches++;
   272→            }
   273→        }
   274→        
   275→        if (!query_words.empty()) {
   276→            return static_cast<double>(matches) / query_words.size() * 0.8;
   277→        }
   278→        
   279→        return 0.0;
   280→    }
   281→
   282→private:
   283→    static std::vector<std::string> splitWords(const std::string& text) {
   284→        std::vector<std::string> words;
   285→        std::stringstream ss(text);
   286→        std::string word;
   287→        while (ss >> word) {
   288→            if (word.length() > 2) words.push_back(word);
   289→        }
   290→        return words;
   291→    }

   573→    std::vector<UpstreamFixRecord> findRelevantUpstreamFixes(
   574→        const std::string& query,
   575→        double min_relevance = 0.5,
   576→        size_t limit = 10) const {
   577→        
   578→        std::shared_lock lock(m_mutex);
   579→        
   580→        std::vector<std::pair<double, UpstreamFixRecord>> scored;
   581→        
   582→        for (const auto& [id, fix] : m_upstream_fixes) {
   583→            double relevance = fix.relevanceToQuery(query);
   584→            if (relevance >= min_relevance) {
   585→                scored.push_back({relevance, fix});
   586→            }
   587→        }
   588→        
   589→        // Sort by relevance
   590→        std::sort(scored.begin(), scored.end(),
   591→            [](const auto& a, const auto& b) { return a.first > b.first; });
   592→        
   593→        // Extract top results
   594→        std::vector<UpstreamFixRecord> result;
   595→        for (size_t i = 0; i < scored.size() && i < limit; ++i) {
   596→            result.push_back(scored[i].second);
   597→        }
   598→        
   599→        return result;
   600→    }

   606→    struct SearchQuery {
   607→        std::string text;
   608→        std::vector<std::string> tags;
   609→        std::optional<ExperimentStatus> status_filter;
   610→        std::optional<InvestigationStatus> hypothesis_status_filter;
   611→        int max_results{20};
   612→        double min_relevance{0.3};
   613→        
   614→        static SearchQuery forText(const std::string& text, int max_results = 20) {
   615→            return {text, {}, std::nullopt, std::nullopt, max_results, 0.3};
   616→        }
   617→        
   618→        static SearchQuery forTags(const std::vector<std::string>& tags) {
   619→            return {"", tags, std::nullopt, std::nullopt, 50, 0.0};
   620→        }
   621→    };
   622→    
   623→    /**
   624→     * Search across all data
   625→     */
   626→    std::vector<SearchResult> search(const SearchQuery& query) const {
   627→        std::vector<SearchResult> results;
   628→        
   629→        // Search experiments
   630→        {
   631→            std::shared_lock lock(m_mutex);
   632→            
   633→            for (const auto& [id, exp] : m_experiments) {
   634→                double score = 0.0;
   635→                std::string matched;
   636→                
   637→                if (!query.text.empty()) {
   638→                    score = calculateTextSimilarity(query.text, exp.title + " " + exp.description);
   639→                    if (score > 0) matched = "text";
   640→                }
   641→                
   642→                // Tag matching boosts score
   643→                for (const auto& tag : query.tags) {
   644→                    auto it = std::find(exp.tags.begin(), exp.tags.end(), tag);
   645→                    if (it != exp.tags.end()) {
   646→                        score = std::max(score, 0.8);
   647→                        if (!matched.empty()) matched += "+tag:" + tag;
   648→                        else matched = "tag:" + tag;
   649→                    }
   650→                }
   651→                
   652→                // Status filter
   653→                if (query.status_filter.has_value() && exp.status != *query.status_filter) {
   654→                    score = 0.0;
   655→                }
   656→                
   657→                if (score >= query.min_relevance) {
   658→                    SearchResult sr;
   659→                    sr.relevance_score = score;
   660→                    sr.matched_fields = matched;
   661→                    sr.experiment = exp;
   662→                    results.push_back(sr);
   663→                }
   664→            }
   665→        }
   666→        
   667→        // Sort by relevance
   668→        std::sort(results.begin(), results.end(), std::greater<SearchResult>());
   669→        
   670→        // Limit results
   671→        if (results.size() > static_cast<size_t>(query.max_results)) {
   672→            results.resize(query.max_results);
   673→        }
   674→        
   675→        return results;
   676→    }
   677→    
   678→    /**
   679→     * Simple text search
   680→     */
   681→    std::vector<SearchResult> searchText(const std::string& query, int max_results = 20) const {
   682→        return search(SearchQuery::forText(query, max_results));
   683→    }

   689→    /**
   690→     * Check if a new experiment would duplicate existing work
   691→     */
   692→    std::vector<DuplicateWarning> checkForDuplicates(
   693→        const std::string& title,
   694→        const std::vector<std::string>& tags = {},
   695→        double threshold = 0.7) const {
   696→        
   697→        std::vector<DuplicateWarning> warnings;
   698→        
   699→        std::shared_lock lock(m_mutex);
   700→        
   701→        for (const auto& [id, exp] : m_experiments) {
   702→            double similarity = calculateTextSimilarity(title, exp.title);
   703→            
   704→            // Boost for matching tags
   705→            if (!tags.empty()) {
   706→                size_t matching_tags = 0;
   707→                for (const auto& tag : tags) {
   708→                    if (std::find(exp.tags.begin(), exp.tags.end(), tag) != exp.tags.end()) {
   709→                        matching_tags++;
   710→                    }
   711→                }
   712→                
   713→                if (!exp.tags.empty()) {
   714→                    double tag_ratio = static_cast<double>(matching_tags) / tags.size();
   715→                    similarity = std::max(similarity, tag_ratio * 0.8);
   716→                }
   717→            }
   718→            
   719→            if (similarity >= threshold) {
   720→                DuplicateWarning warn;
   721→                warn.existing_id = id;
   722→                warn.existing_title = exp.title;
   723→                warn.existing_status = exp.status;
   724→                warn.similarity_score = similarity;
   725→                
   726→                if (similarity >= 0.9) {
   727→                    warn.reason = "Near-exact match to existing investigation";
   728→                    warn.recommendation = "Review existing experiment before proceeding.";
   729→                } else if (exp.status == ExperimentStatus::Completed) {
   730→                    warn.reason = "Similar to completed investigation";
   731→                    warn.recommendation = "Check if this experiment already solved your issue.";
   732→                } else if (exp.status == ExperimentStatus::Running) {
   733→                    warn.reason = "Similar investigation currently in progress";
   734→                    warn.recommendation = "Coordinate with ongoing investigation or wait for results.";
   735→                } else {
   736→                    warn.reason = "Similar topic previously investigated";
   737→                    warn.recommendation = "Review findings before starting new work.";
   738→                }
   739→                
   740→                warnings.push_back(warn);
   741→            }
   742→        }
   743→        
   744→        return warnings;
   745→    }
   746→    
   747→    /**
   748→     * Quick duplicate check
   749→     */
   750→    std::pair<bool, std::optional<DuplicateWarning>> isDuplicate(
   751→        const std::string& title,
   752→        const std::vector<std::string>& tags = {}) const {
   753→        
   754→        auto warnings = checkForDuplicates(title, tags, 0.8);
   755→        
   756→        if (warnings.empty()) {
   757→            return {false, std::nullopt};
   758→        }
   759→        
   760→        // Return highest similarity
   761→        auto max_it = std::max_element(warnings.begin(), warnings.end(),
   762→            [](const DuplicateWarning& a, const DuplicateWarning& b) {
   763→                return a.similarity_score < b.similarity_score;
   764→            });
   765→        
   766→        return {true, *max_it};
   767→    }

   855→    double calculateTextSimilarity(const std::string& a, const std::string& b) const {
   856→        // Bigram overlap similarity
   857→        auto bigrams = [](const std::string& s) -> std::set<std::string> {
   858→            std::set<std::string> result;
   859→            std::string lower = s;
   860→            std::transform(lower.begin(), lower.end(), lower.begin(), ::tolower);
   861→            
   862→            for (size_t i = 0; i + 1 < lower.length(); ++i) {
   863→                result.insert(lower.substr(i, 2));
   864→            }
   865→            return result;
   866→        };
   867→        
   868→        auto bgA = bigrams(a);
   869→        auto bgB = bigrams(b);
   870→        
   871→        if (bgA.empty() && bgB.empty()) return 1.0;
   872→        if (bgA.empty() || bgB.empty()) return 0.0;
   873→        
   874→        std::set<std::string> intersection;
   875→        std::set_intersection(bgA.begin(), bgA.end(), bgB.begin(), bgB.end(),
   876→                             std::inserter(intersection, intersection.begin()));
   877→        
   878→        std::set<std::string> union_set;
   879→        std::union(bgA.begin(), bgA.end(), bgB.begin(), bgB.end(),
   880→                  std::inserter(union_set, union_set.begin()));
   881→        
   882→        return static_cast<double>(intersection.size()) / union_set.size();
   883→    }

    92→/**

    95→class Timestamp {
    96→public:
    97→    static std::string now() {
    98→        auto now = std::chrono::system_clock::now();
    99→        auto time_t = std::chrono::system_clock::to_time_t(now);
   100→        std::stringstream ss;
   101→        ss << std::put_time(std::gmtime(&time_t), "%Y-%m-%dT%H:%M:%SZ");
   102→        return ss.str();
   103→    }
   104→    
   105→    static std::string fileFriendly() {
   106→        auto now = std::chrono::system_clock::now();
   107→        auto time_t = std::chrono::system_clock::to_time_t(now);
   108→        std::stringstream ss;
   109→        ss << std::put_time(std::gmtime(&time_t), "%Y%m%d_%H%M%S");
   110→        return ss.str();
   111→    }

   435→    /**
   436→     * Simple text similarity using bigram overlap
   437→     */
   438→    double calculateSimilarity(const std::string& a, const std::string& b) const {
   439→        auto bigrams = [](const std::string& s) -> std::set<std::string> {
   440→            std::set<std::string> result;
   441→            std::string lower = s;
   442→            std::transform(lower.begin(), lower.end(), lower.begin(), ::tolower);
   443→            for (size_t i = 0; i + 1 < lower.length(); ++i) {
   444→                result.insert(lower.substr(i, 2));
   445→            }
   446→            return result;
   447→        };
   448→        
   449→        auto bgA = bigrams(a);
   450→        auto bgB = bigrams(b);
   451→        
   452→        if (bgA.empty() && bgB.empty()) return 1.0;
   453→        if (bgA.empty() || bgB.empty()) return 0.0;
   454→        
   455→        std::set<std::string> intersection;
   456→        std::set_intersection(bgA.begin(), bgA.end(), bgB.begin(), bgB.end(),
   457→                             std::inserter(intersection, intersection.begin()));
   458→        
   459→        std::set<std::string> union_set;
   460→        std::set_union(bgA.begin(), bgA.end(), bgB.begin(), bgB.end(),
   461→                      std::inserter(union_set, union_set.begin()));
   462→        
   463→        return static_cast<double>(intersection.size()) / union_set.size();
   464→    }
   465→};

   613→/**
   614→ * Entry in experiment history
   615→ */
   616→struct HistoryEntry {
   617→    std::string experiment_id;
   618→    std::string title;
   619→    std::string status;
   620→    std::string timestamp;
   621→    std::vector<std::string> tags;
   622→    std::string summary;
   623→    fs::path package_path;
   624→    
   625→    /**
   626→     * Check if this entry matches a search query
   627→     */
   628→    bool matchesQuery(const std::string& query) const {
   629→        std::string lower_query = query;
   630→        std::transform(lower_query.begin(), lower_query.end(), lower_query.begin(), ::tolower);
   631→        
   632→        std::string combined = title + " " + summary + " ";
   633→        for (const auto& tag : tags) {
   634→            combined += tag + " ";
   635→        }
   636→        std::transform(combined.begin(), combined.end(), combined.begin(), ::tolower);
   637→        
   638→        return combined.find(lower_query) != std::string::npos;
   639→    }
   640→    
   641→    /**
   642→     * Calculate relevance score for a query (0.0 to 1.0)
   643→     */
   644→    double relevanceScore(const std::string& query) const {
   645→        std::string lower_query = query;
   646→        std::transform(lower_query.begin(), lower_query.end(), lower_query.begin(), ::tolower);
   647→        
   648→        // Exact title match gets highest score
   649→        std::string lower_title = title;
   650→        std::transform(lower_title.begin(), lower_title.end(), lower_title.begin(), ::tolower);
   651→        if (lower_title == lower_query) return 1.0;
   652→        
   653→        // Title contains query
   654→        if (lower_title.find(lower_query) != std::string::npos) return 0.9;
   655→        
   656→        // Check tags
   657→        for (const auto& tag : tags) {
   658→            std::string lower_tag = tag;
   659→            std::transform(lower_tag.begin(), lower_tag.end(), lower_tag.begin(), ::tolower);
   660→            if (lower_tag == lower_query) return 0.8;
   661→            if (lower_tag.find(lower_query) != std::string::npos) return 0.7;
   662→        }
   663→        
   664→        // Summary match
   665→        std::string lower_summary = summary;
   666→        std::transform(lower_summary.begin(), lower_summary.end(), lower_summary.begin(), ::tolower);
   667→        if (lower_summary.find(lower_query) != std::string::npos) return 0.5;
   668→        
   669→        return 0.0;
   670→    }
   671→};

   709→    std::string event_type;
   710→    std::string description;
   711→    std::string reference_id;  // Evidence ID, hypothesis ID, etc.
   712→    std::map<std::string, std::string> metadata;

   714→    static TimelineEvent evidenceAdded(const Evidence& e) {

   716→        evt.timestamp = e.timestamp;
   717→        evt.event_type = "evidence_added";

   753→ * Final root cause report
   754→ */
   755→struct RootCauseReport {
   756→    std::string experiment_id;
   757→    std::string generated_at;
   758→    std::string title;
   759→    std::string executive_summary;
   760→    
   761→    std::string root_cause_description;
   762→    std::string confirmed_hypothesis_id;
   763→    double confidence_level;
   764→    
   765→    std::vector<TimelineEvent> timeline;
   766→    std::vector<Evidence> key_evidence;
   767→    std::vector<Hypothesis> all_hypotheses;
   768→    std::vector<Hypothesis> rejected_hypotheses;
   769→    
   770→    std::string recommended_fix;
   771→    std::string verification_steps;
   772→    std::string lessons_learned;
   773→    
   774→    std::vector<UpstreamFixNotification> upstream_fixes;
   775→    std::vector<DuplicateWarning> duplicates_found;
   776→    
   777→    std::map<std::string, std::string> metadata;
   778→    
   779→    RootCauseReport() : confidence_level(0.0) {
   780→        generated_at = Timestamp::now();
   781→    }
   782→    
   783→    /**
   784→     * Validate that report has minimum required fields populated
   785→     */
   786→    bool isValid() const {
   787→        return !experiment_id.empty() &&
   788→               !root_cause_description.empty() &&
   789→               !confirmed_hypothesis_id.empty() &&
   790→               confidence_level > 0.0 &&
   791→               !key_evidence.empty();
   792→    }
   793→};

     1→/**
     2→ * State Timeline System - Priority 3
     3→ * 
     4→ * PROBLEM SOLVED:
     5→ * Crash location is usually not corruption location.
     6→ * Need historical state to trace back to root cause.
     7→ * 
     8→ * EXAMPLE:
     9→ *   Current: "Crash at frame 1300"
    10→ *   Required: 
    11→ *     Object created:  frame 200
    12→ *     Modified:        frame 600
    13→ *     Invalidated:     frame 900
    14→ *     Used incorrectly: frame 1250
    15→ *     Crash:           frame 1300
    16→ * 
    17→ * DESIGN:
    18→ * - Records state changes with frame-level precision
    19→ * - Tracks object/resource lifecycles
    20→ * - Enables time-travel queries ("what was state at frame X?")
    21→ * - Supports multiple timeline streams (GPU, memory, HLE, etc.)
    22→ */

   146→// ============================================================================
   147→// Timeline Event
   148→// ============================================================================
   149→
   150→struct TimelineEvent {
   151→    std::string id;
   152→    DebugTimestamp timestamp;
   153→    uint64_t frame_number{0};
   154→    TimelineEventType type;
   155→    
   156→    // What object/state?
   157→    std::string object_id;       // Unique identifier for the object
   158→    std::string object_type;     // Class/type of object
   159→    std::string object_description;  // Human-readable description
   160→    
   161→    // What changed?
   162→    std::string attribute_name;   // Which attribute changed
   163→    std::string old_value;        // Previous value (JSON string)
   164→    std::string new_value;        // New value (JSON string)
   165→    
   166→    // Provenance
   167→    Subsystem subsystem{Subsystem::Unknown};
   168→    SourceLocation source;
   169→    std::string thread_id;
   170→    
   171→    // Additional data
   172→    std::map<std::string, std::string> metadata;
   173→    
   174→    // Relationships
   175→    std::string parent_event_id;  // Causal parent
   176→    std::vector<std::string> related_event_ids;
   177→    
   178→    static std::string generateId() {
   179→        static std::atomic<uint64_t> counter{0};
   180→        auto now = std::chrono::high_resolution_clock::now().time_since_epoch().count();
   181→        return "tle_" + std::to_string(now) + "_" + std::to_string(++counter);
   182→    }
   183→    
   184→    TimelineEvent() {
   185→        id = generateId();
   186→        timestamp = DebugTimestamp::now();
   187→    }
   188→    
   189→    bool isErrorOrCorruption() const {
   190→        return type == TimelineEventType::ErrorDetected ||
   191→               type == TimelineEventType::CorruptionDetected ||
   192→               type == TimelineEventType::AssertionFailed ||
   193→               type == TimelineEventType::AnomalyDetected;
   194→    }
   195→    
   196→    bool isLifecycleEvent() const {
   197→        return type == TimelineEventType::ObjectCreated ||
   198→               type == TimelineEventType::ObjectDestroyed ||
   199→               type == TimelineEventType::ResourceAllocated ||
   200→               type == TimelineEventType::ResourceFreed;
   201→    }
   202→    
   203→    std::string toJson() const {
   204→        std::stringstream json;
   205→        json << "{\n";
   206→        json << "  \"id\": \"" << JsonUtils::escapeJson(id) << "\",\n";
   207→        json << "  \"timestamp\": \"" << timestamp.toISO8601() << "\",\n";
   208→        json << "  \"frame\": " << frame_number << ",\n";
   209→        json << "  \"type\": \"" << timelineEventTypeToString(type) << "\",\n";
   210→        json << "  \"object_id\": \"" << JsonUtils::escapeJson(object_id) << "\",\n";
   211→        json << "  \"object_type\": \"" << JsonUtils::escapeJson(object_type) << "\",\n";
   212→        if (!object_description.empty()) {
   213→            json << "  \"description\": \"" << JsonUtils::escapeJson(object_description) << "\",\n";
   214→        }
   215→        if (!attribute_name.empty()) {
   216→            json << "  \"attribute\": \"" << JsonUtils::escapeJson(attribute_name) << "\",\n";
   217→            json << "  \"old_value\": \"" << JsonUtils::escapeJson(old_value) << "\",\n";
   218→            json << "  \"new_value\": \"" << JsonUtils::escapeJson(new_value) << "\",\n";
   219→        }
   220→        json << "  \"subsystem\": \"" << subsystemToString(subsystem) << "\"\n";
   221→        json << "}";
   222→        return json.str();
   223→    }
   224→};

   226→// ============================================================================
   227→// Object Lifecycle Tracker
   228→// ============================================================================
   229→
   230→struct ObjectLifecycle {
   231→    std::string object_id;
   232→    std::string object_type;
   233→    
   234→    // Key lifecycle events
   235→    std::optional<TimelineEvent> created;
   236→    std::optional<TimelineEvent> destroyed;
   237→    std::vector<TimelineEvent> modifications;
   238→    
   239→    // Current state (reconstructed from events)
   240→    std::map<std::string, std::string> current_state;
   241→    
   242→    // Statistics
   243→    uint64_t creation_frame{0};
   244→    uint64_t destruction_frame{0};  // 0 if still alive
   245→    uint64_t frames_before_crash{0};  // For crash analysis
   246→    size_t total_modifications{0};
   247→    uint64_t last_access_frame{0};
   248→    
   249→    bool isAlive() const { return !destroyed.has_value(); }
   250→    uint64_t lifetime() const { 
   251→        return destruction_frame > 0 ? destruction_frame - creation_frame : 0;
   252→    }
   253→    
   254→    std::string toJson() const {
   255→        std::stringstream json;
   256→        json << "{\n";
   257→        json << "  \"object_id\": \"" << JsonUtils::escapeJson(object_id) << "\",\n";
   258→        json << "  \"object_type\": \"" << JsonUtils::escapeJson(object_type) << "\",\n";
   259→        json << "  \"alive\": " << (isAlive() ? "true" : "false") << ",\n";
   260→        json << "  \"created_frame\": " << creation_frame << ",\n";
   261→        json << "  \"modifications\": " << total_modifications << ",\n";
   262→        json << "  \"lifetime_frames\": " << lifetime() << "\n";
   263→        json << "}";
   264→        return json.str();
   265→    }
   266→};

   269→// Timeline Query Results

   271→
   272→struct TimelineQueryResult {
   273→    std::vector<TimelineEvent> events;
   274→    
   275→    // Time range of results
   276→    uint64_t start_frame{0};
   277→    uint64_t end_frame{0};
   278→    
   279→    // Filtering applied
   280→    std::vector<TimelineEventType> event_types_filter;
   281→    Subsystem subsystem_filter{Subsystem::Unknown};
   282→    std::string object_id_filter;
   283→    
   284→    // Aggregated statistics
   285→    size_t total_events{0};
   286→    size_t error_events{0};
   287→    size_t lifecycle_events{0};
   288→    
   289→    // Object lifecycles found (if querying by object)
   290→    std::map<std::string, ObjectLifecycle> object_lifecycles;
   291→    
   292→    std::string toJson() const {
   293→        std::stringstream json;
   294→        json << "{\n";
   295→        json << "  \"total_events\": " << total_events << ",\n";
   296→        json << "  \"error_events\": " << error_events << ",\n";
   297→        json << "  \"frame_range\": [" << start_frame << ", " << end_frame << "],\n";
   298→        
   299→        json << "  \"events\": [\n";
   300→        for (size_t i = 0; i < events.size() && i < 100; ++i) {  // Limit output
   301→            json << "    " << events[i].toJson();
   302→            if (i < events.size() - 1 && i < 99) json << ",";
   303→            json << "\n";
   304→        }
   305→        if (events.size() > 100) {
   306→            json << "    ... and " << (events.size() - 100) << " more\n";
   307→        }
   308→        json << "  ]\n";
   309→        
   310→        json << "}\n";
   311→        return json.str();
   312→    }
   313→};

   318→
   319→struct StateSnapshot {
   320→    uint64_t target_frame;
   321→    DebugTimestamp approximate_time;
   322→    
   323→    // All active objects at this point
   324→    std::map<std::string, ObjectLifecycle> active_objects;
   325→    
   326→    // Recent events leading to this state
   327→    std::vector<TimelineEvent> preceding_events;
   328→

   330→    std::vector<TimelineEvent> prior_errors;

   333→        std::stringstream json;
   334→        json << "{\n";
   335→        json << "  \"frame\": " << target_frame << ",\n";
   336→        json << "  \"active_objects\": " << active_objects.size() << ",\n";
   337→        json << "  \"prior_errors\": " << prior_errors.size() << "\n";
   338→        json << "}";
   339→        return json.str();
   340→    }

   344→// Timeline System - Main Class

   346→
   347→class StateTimelineSystem {
   348→public:
   349→    struct Config {
   350→        bool enabled;
   351→        size_t max_events;          // Global event limit
   352→        size_t max_events_per_object; // Per-object limit
   353→        bool track_all_objects;       // Track everything or just watched
   354→        bool auto_track_gpu;           // Auto-track GPU resources
   355→        bool auto_track_memory;        // Auto-track major allocations
   356→        bool record_full_state_changes; // Record old+new values (expensive)
   357→        
   358→        Config()
   359→            : enabled(true)
   360→            , max_events(1000000)
   361→            , max_events_per_object(10000)
   362→            , track_all_objects(false)
   363→            , auto_track_gpu(true)
   364→            , auto_track_memory(true)
   365→            , record_full_state_changes(false) {}
   366→        
   367→        Config(bool en, size_t me, size_t mepo, bool tao, bool atg, bool atm, bool rfsc)
   368→            : enabled(en)
   369→            , max_events(me)
   370→            , max_events_per_object(mepo)
   371→            , track_all_objects(tao)
   372→            , auto_track_gpu(atg)
   373→            , auto_track_memory(atm)
   374→            , record_full_state_changes(rfsc) {}
   375→        
   376→        static Config minimal() {
   377→            return Config(true, 100000, 1000, false, false, false, false);
   378→        }
   379→        
   380→        static Config comprehensive() {
   381→            return Config(true, 10000000, 50000, true, true, true, true);
   382→        }
   383→    };
   384→    
   385→    explicit StateTimelineSystem(const Config& config = Config())
   386→        : m_config(config), m_enabled(config.enabled), m_current_frame(0) {}
   387→    
   388→    // ========================================================================
   389→    // Control Interface
   390→    // ========================================================================
   391→    
   392→    void enable() { m_enabled.store(true); }
   393→    void disable() { m_enabled.store(false); }
   394→    bool isEnabled() const { return m_enabled.load(); }
   395→    
   396→    void clear() {
   397→        std::unique_lock lock(m_mutex);
   398→        m_events.clear();
   399→        m_object_lifecycles.clear();
   400→        m_watch_list.clear();
   401→        m_current_frame = 0;
   402→    }
   403→    
   404→    // ========================================================================
   405→    // Frame Tracking
   406→    // ========================================================================
   407→    
   408→    /**
   409→     * Advance to next frame (call once per emulated frame)
   410→     */
   411→    void advanceFrame() {
   412→        if (!isEnabled()) return;
   413→        
   414→        ++m_current_frame;
   415→        
   416→        // Optionally emit frame marker
   417→        if (m_config.track_all_objects) {
   418→            TimelineEvent marker;
   419→            marker.type = TimelineEventType::FrameMarker;
   420→            marker.frame_number = m_current_frame;
   421→            marker.object_description = "Frame " + std::to_string(m_current_frame);
   422→            
   423→            addEventInternal(marker);
   424→        }
   425→    }
   426→    
   427→    void setFrame(uint64_t frame) {
   428→        m_current_frame = frame;
   429→    }
   430→    
   431→    uint64_t getCurrentFrame() const { return m_current_frame.load(); }
   432→    
   433→    // ========================================================================
   434→    // Object Watching
   435→    // ========================================================================
   436→    
   437→    /**
   438→     * Watch a specific object for all state changes
   439→     */
   440→    void watchObject(const std::string& object_id, const std::string& object_type = "") {
   441→        std::unique_lock lock(m_mutex);
   442→        m_watch_list[object_id] = object_type;
   443→    }
   444→    
   445→    void unwatchObject(const std::string& object_id) {
   446→        std::unique_lock lock(m_mutex);
   447→        m_watch_list.erase(object_id);
   448→    }
   449→    
   450→    bool isWatching(const std::string& object_id) const {
   451→        std::shared_lock lock(m_mutex);
   452→        return m_watch_list.count(object_id) > 0 || m_config.track_all_objects;
   453→    }

   455→    // ========================================================================
   456→    // Event Recording Interface
   457→    // ========================================================================
   458→    
   459→    /**
   460→     * Record generic timeline event

   476→        event.source = loc;
   477→        event.thread_id = getCurrentThreadId();

   479→        addEventInternal(event);
   480→    }

   605→    
   606→    /**
   607→     * Get all events in a frame range
   608→     */
   609→    TimelineQueryResult queryByFrameRange(uint64_t start_frame, uint64_t end_frame) const {
   610→        TimelineQueryResult result;
   611→        result.start_frame = start_frame;
   612→        result.end_frame = end_frame;
   613→        
   614→        std::shared_lock lock(m_mutex);
   615→        
   616→        for (const auto& event : m_events) {
   617→            if (event.frame_number >= start_frame && event.frame_number <= end_frame) {
   618→                result.events.push_back(event);
   619→                
   620→                if (event.isErrorOrCorruption()) result.error_events++;
   621→                if (event.isLifecycleEvent()) result.lifecycle_events++;
   622→                result.total_events++;
   623→            }
   624→        }
   625→        
   626→        return result;
   627→    }
   628→    
   629→    /**
   630→     * Get complete lifecycle of an object
   631→     */
   632→    std::optional<ObjectLifecycle> getObjectLifecycle(const std::string& object_id) const {
   633→        std::shared_lock lock(m_mutex);
   634→        
   635→        auto it = m_object_lifecycles.find(object_id);
   636→        if (it != m_object_lifecycles.end()) {
   637→            return it->second;
   638→        }
   639→        return std::nullopt;
   640→    }
   641→    
   642→    /**
   643→     * Get state snapshot at a specific frame
   644→     */
   645→    StateSnapshot getSnapshotAtFrame(uint64_t target_frame) const {
   646→        StateSnapshot snapshot;
   647→        snapshot.target_frame = target_frame;
   648→        
   649→        std::shared_lock lock(m_mutex);
   650→        
   651→        // Find all objects that existed at this frame
   652→        for (const auto& [obj_id, lifecycle] : m_object_lifecycles) {
   653→            if (lifecycle.creation_frame <= target_frame &&
   654→                (lifecycle.destruction_frame == 0 || lifecycle.destruction_frame > target_frame)) {
   655→                snapshot.active_objects[obj_id] = lifecycle;
   656→            }
   657→        }
   658→        
   659→        // Get recent events leading up to this frame
   660→        int count = 0;
   661→        for (auto it = m_events.rbegin(); it != m_events.rend() && count < 50; ++it) {
   662→            if (it->frame_number <= target_frame) {
   663→                snapshot.preceding_events.push_back(*it);
   664→                count++;
   665→                
   666→                if (it->isErrorOrCorruption()) {
   667→                    snapshot.prior_errors.push_back(*it);
   668→                }
   669→            }
   670→        }
   671→        
   672→        return snapshot;
   673→    }
   674→    
   675→    /**
   676→     * Find what happened to an object around a specific frame
   677→     */
   678→    std::vector<TimelineEvent> getObjectEventsAroundFrame(
   679→        const std::string& object_id,
   680→        uint64_t center_frame,
   681→        uint64_t window_size = 100) const {
   682→        
   683→        std::vector<TimelineEvent> result;
   684→        uint64_t start = window_size > center_frame ? 0 : center_frame - window_size;
   685→        uint64_t end = center_frame + window_size;
   686→        
   687→        std::shared_lock lock(m_mutex);
   688→        
   689→        for (const auto& event : m_events) {
   690→            if (event.object_id == object_id &&
   691→                event.frame_number >= start &&
   692→                event.frame_number <= end) {
   693→                result.push_back(event);
   694→            }
   695→        }
   696→        
   697→        return result;
   698→    }
   699→    
   700→    /**
   701→     * Find all errors/corruptions in a frame range
   702→     */
   703→    std::vector<TimelineEvent> findErrorsInFrameRange(
   704→        uint64_t start_frame,
   705→        uint64_t end_frame) const {
   706→        
   707→        std::vector<TimelineEvent> errors;
   708→        
   709→        std::shared_lock lock(m_mutex);
   710→        
   711→        for (const auto& event : m_events) {
   712→            if (event.isErrorOrCorruption() &&
   713→                event.frame_number >= start_frame &&
   714→                event.frame_number <= end_frame) {
   715→                errors.push_back(event);
   716→            }
   717→        }
   718→        
   719→        return errors;
   720→    }

   722→    /**
   723→     * Find objects that were destroyed shortly before a crash
   724→     * (Common pattern: use-after-free)
   725→     */
   726→    std::vector<ObjectLifecycle> findObjectsDestroyedBeforeFrame(
   727→        uint64_t crash_frame,
   728→        uint64_t lookback_window = 1000) const {
   729→        
   730→        std::vector<ObjectLifecycle> recently_destroyed;
   731→        uint64_t start = lookback_window > crash_frame ? 0 : crash_frame - lookback_window;
   732→        
   733→        std::shared_lock lock(m_mutex);
   734→        
   735→        for (const auto& [obj_id, lifecycle] : m_object_lifecycles) {
   736→            if (lifecycle.destruction_frame > 0 &&
   737→                lifecycle.destruction_frame >= start &&
   738→                lifecycle.destruction_frame < crash_frame) {
   739→                
   740→                ObjectLifecycle lt;
   741→                lt.object_id = obj_id;
   742→                lt.object_type = lifecycle.object_type;
   743→                lt.destruction_frame = lifecycle.destruction_frame;
   744→                lt.frames_before_crash = crash_frame - lifecycle.destruction_frame;
   745→                
   746→                recently_destroyed.push_back(lt);
   747→            }
   748→        }
   749→        
   750→        // Sort by proximity to crash
   751→        std::sort(recently_destroyed.begin(), recently_destroyed.end(),
   752→            [](const ObjectLifecycle& a, const ObjectLifecycle& b) {
   753→                return a.frames_before_crash < b.frames_before_crash;
   754→            });
   755→        
   756→        return recently_destroyed;
   757→    }

   763→    struct Stats {
   764→        size_t total_events{0};
   765→        uint64_t current_frame{0};
   766→        size_t tracked_objects{0};
   767→        size_t watched_objects{0};
   768→        size_t active_objects{0};
   769→        size_t total_errors{0};
   770→        size_t total_corruptions{0};
   771→        
   772→        std::map<TimelineEventType, size_t> events_by_type;
   773→        std::map<Subsystem, size_t> events_by_subsystem;
   774→    };
   775→    
   776→    Stats getStats() const {
   777→        std::shared_lock lock(m_mutex);
   778→        
   779→        Stats stats;
   780→        stats.current_frame = m_current_frame;
   781→        stats.tracked_objects = m_object_lifecycles.size();
   782→        stats.watched_objects = m_watch_list.size();
   783→        
   784→        for (const auto& [obj_id, lifecycle] : m_object_lifecycles) {
   785→            if (lifecycle.isAlive()) stats.active_objects++;
   786→        }
   787→        
   788→        for (const auto& event : m_events) {
   789→            stats.total_events++;
   790→            stats.events_by_type[event.type]++;
   791→            stats.events_by_subsystem[event.subsystem]++;
   792→            
   793→            if (event.type == TimelineEventType::ErrorDetected) stats.total_errors++;
   794→            if (event.type == TimelineEventType::CorruptionDetected) stats.total_corruptions++;
   795→        }
   796→        
   797→        return stats;
   798→    }

   805→    
   806→    // Event storage (ordered by insertion)
   807→    std::deque<TimelineEvent> m_events;
   808→    
   809→    // Object lifecycle tracking
   810→    std::map<std::string, ObjectLifecycle> m_object_lifecycles;
   811→

   813→    std::map<std::string, std::string> m_watch_list;  // object_id -> type

   816→    // Internal Methods

   818→    
   819→    bool shouldTrack(const std::string& object_id) const {
   820→        if (m_config.track_all_objects) return true;
   821→        return m_watch_list.count(object_id) > 0;
   822→    }
   823→    
   824→    void addEventInternal(const TimelineEvent& event) {
   825→        std::unique_lock lock(m_mutex);
   826→        
   827→        // Check global limit
   828→        if (m_events.size() >= m_config.max_events) {
   829→            m_events.pop_front();  // Evict oldest
   830→        }
   831→        
   832→        m_events.push_back(event);
   833→        
   834→        // Update object lifecycle tracking
   835→        updateObjectLifecycle(event);
   836→    }
   837→    
   838→    void updateObjectLifecycle(const TimelineEvent& event) {
   839→        // Get or create lifecycle entry
   840→        auto& lifecycle = m_object_lifecycles[event.object_id];
   841→        
   842→        if (lifecycle.object_id.empty()) {
   843→            lifecycle.object_id = event.object_id;
   844→            lifecycle.object_type = event.object_type;
   845→        }
   846→        
   847→        switch (event.type) {
   848→            case TimelineEventType::ObjectCreated:
   849→            case TimelineEventType::ResourceAllocated:
   850→                if (!lifecycle.created.has_value()) {
   851→                    lifecycle.created = event;
   852→                    lifecycle.creation_frame = event.frame_number;
   853→                }
   854→                break;
   855→                
   856→            case TimelineEventType::ObjectDestroyed:
   857→            case TimelineEventType::ResourceFreed:
   858→                if (!lifecycle.destroyed.has_value()) {
   859→                    lifecycle.destroyed = event;
   860→                    lifecycle.destruction_frame = event.frame_number;
   861→                }
   862→                break;
   863→                
   864→            case TimelineEventType::ObjectModified:
   865→            case TimelineEventType::StateChanged:
   866→                lifecycle.modifications.push_back(event);
   867→                lifecycle.total_modifications++;
   868→                
   869→                // Update current state
   870→                if (!event.attribute_name.empty()) {
   871→                    lifecycle.current_state[event.attribute_name] = event.new_value;
   872→                }
   873→                break;
   874→                
   875→            default:
   876→                break;
   877→        }
   878→        
   879→        // Update last access
   880→        if (event.frame_number > lifecycle.last_access_frame) {
   881→            lifecycle.last_access_frame = event.frame_number;
   882→        }
   883→    }

   887→        ss << std::this_thread::get_id();
   888→        return ss.str();
   889→    }

     1→/**
     2→ * Causal Dependency Graph - Priority 6
     3→ * 
     4→ * PROBLEM SOLVED:
     5→ * Emulator debugging is not normal application debugging.
     6→ * Architecture has multiple layers:
     7→ * 
     8→ *   Guest Application
     9→ *        ↓
    10→ *   HLE (High-Level Emulation)
    11→ *        ↓
    12→ *   Kernel
    13→ *        ↓
    14→ *   Memory
    15→ *        ↓
    16→ *   CPU
    17→ *        ↓
    18→ *   GPU
    19→ *        ↓
    20→ *   Host System
    21→ * 
    22→ * A crash in one layer may originate elsewhere.
    23→ * 
    24→ * EXAMPLE:
    25→ *   Crash: Allocator failure (Memory layer)
    26→ *   Actual cause: GPU wrote into freed memory 20 seconds earlier (GPU layer)
    27→ * 
    28→ * This module builds and queries dependency graphs to trace root causes across layers.
    29→ */
    30→

    48→enum class EmulationLayer : uint8_t {
    49→    Guest,          // Game/application code
    50→    HLE,            // High-level emulation functions
    51→    Kernel,         // OS kernel simulation
    52→    Memory,         // Memory management
    53→    CPU,            // Processor emulation
    54→    GPU,            // Graphics processing
    55→    Audio,          // Sound processing
    56→    Input,          // Input handling
    57→    FileSystem,     // File system abstraction
    58→    Network,        // Network stack
    59→    Host,           // Host operating system
    60→    Diagnostic      // Debug/diagnostic infrastructure
    61→};
    62→
    63→inline std::string emulationLayerToString(EmulationLayer layer) {
    64→    switch (layer) {
    65→        case EmulationLayer::Guest: return "Guest";
    66→        case EmulationLayer::HLE: return "HLE";
    67→        case EmulationLayer::Kernel: return "Kernel";
    68→        case EmulationLayer::Memory: return "Memory";
    69→        case EmulationLayer::CPU: return "CPU";
    70→        case EmulationLayer::GPU: return "GPU";
    71→        case EmulationLayer::Audio: return "Audio";
    72→        case EmulationLayer::Input: return "Input";
    73→        case EmulationLayer::FileSystem: return "FileSystem";
    74→        case EmulationLayer::Network: return "Network";
    75→        case EmulationLayer::Host: return "Host";
    76→        case EmulationLayer::Diagnostic: return "Diagnostic";
    77→    }
    78→    return "Unknown";
    79→}

   100→        case Subsystem::Kernel:

   105→        case Subsystem::Memory:

   186→// ============================================================================
   187→// Edge Types (Causal Relationships)
   188→// ============================================================================
   189→
   190→enum class EdgeType : uint8_t {
   191→    // Direct causation
   192→    CausedBy,           // A was directly caused by B
   193→    CorruptedBy,        // A's data was corrupted by B
   194→    TriggeredBy,        // A was triggered by B
   195→    
   196→    // Temporal relationships
   197→    HappensBefore,      // A occurs before B in time
   198→    HappensAfter,       // A occurs after B in time
   199→    ConcurrentWith,     // A and B happen at same time
   200→    
   201→    // Containment/ownership
   202→    Contains,           // A contains B (spatial or logical)
   203→    Owns,              // A owns resource B
   204→    Allocates,          // A allocated memory that became B
   205→    Frees,             // A freed memory that B used
   206→    
   207→    // Dependencies
   208→    DependsOn,          // A depends on B
   209→    ReadsFrom,          // A reads data written by B
   210→    WritesTo,          // A writes data read by B
   211→    Calls,              // A calls function B
   212→    CalledBy,           // A is called by B
   213→    
   214→    // Cross-layer
   215→    CrossLayerUp,       // From lower layer to higher layer
   216→    CrossLayerDown,     // From higher layer to lower layer
   217→    
   218→    // Abstract
   219→    RelatedTo,          // General relationship
   220→    SimilarTo,          // Pattern similarity
   221→    PossiblyCauses      // Speculative causal link
   222→};
   223→

   230→        case EdgeType::HappensAfter: return "happens_after";
   231→        case EdgeType::ConcurrentWith: return "concurrent_with";
   232→        case EdgeType::Contains: return "contains";
   233→        case EdgeType::Owns: return "owns";
   234→        case EdgeType::Allocates: return "allocates";
   235→        case EdgeType::Frees: return "frees";
   236→        case EdgeType::DependsOn: return "depends_on";
   237→        case EdgeType::ReadsFrom: return "reads_from";
   238→        case EdgeType::WritesTo: return "writes_to";
   239→        case EdgeType::Calls: return "calls";
   240→        case EdgeType::CalledBy: return "called_by";
   241→        case EdgeType::CrossLayerUp: return "cross_layer_up";
   242→        case EdgeType::CrossLayerDown: return "cross_layer_down";

   254→struct GraphNode {
   255→    std::string id;
   256→    NodeType type;
   257→    EmulationLayer layer{EmulationLayer::Diagnostic};
   258→    
   259→    // Identification
   260→    std::string name;
   261→    std::string description;
   262→    
   263→    // Location/context
   264→    SourceLocation location;
   265→    Subsystem subsystem{Subsystem::Unknown};
   266→

   268→    DebugTimestamp timestamp;
   269→    uint64_t frame_number{0};
   270→    
   271→    // Data payload
   272→    std::map<std::string, std::string> attributes;
   273→    
   274→    // State
   275→    bool is_resolved{false};
   276→    bool is_symptom{false};      // This is a symptom, not a cause
   277→    bool is_root_cause{false};   // This is identified as root cause
   278→    
   279→    static std::string generateId() {
   280→        static std::atomic<uint64_t> counter{0};
   281→        auto now = std::chrono::high_resolution_clock::now().time_since_epoch().count();
   282→        return "node_" + std::to_string(now) + "_" + std::to_string(++counter);
   283→    }
   284→    
   285→    GraphNode() {
   286→        id = generateId();
   287→        timestamp = DebugTimestamp::now();
   288→    }
   289→    
   290→    std::string toJson() const {
   291→        std::stringstream json;
   292→        json << "{\n";
   293→        json << "  \"id\": \"" << JsonUtils::escapeJson(id) << "\",\n";
   294→        json << "  \"type\": \"" << nodeTypeToString(type) << "\",\n";
   295→        json << "  \"layer\": \"" << emulationLayerToString(layer) << "\",\n";
   296→        json << "  \"name\": \"" << JsonUtils::escapeJson(name) << "\",\n";
   297→        if (!description.empty()) {
   298→            json << "  \"description\": \"" << JsonUtils::escapeJson(description) << "\",\n";
   299→        }
   300→        json << "  \"frame\": " << frame_number << "\n";

   302→        return json.str();
   303→    }

   350→// ============================================================================
   351→// Causal Path - Sequence of edges from cause to effect
   352→// ============================================================================
   353→
   354→struct CausalPath {
   355→    std::vector<std::string> node_ids;  // Nodes along the path
   356→    std::vector<GraphEdge> edges;        // Edges connecting them
   357→    
   358→    double total_confidence{1.0};
   359→    int cross_layer_transitions{0};
   360→    
   361→    // Classification
   362→    enum class Strength {
   363→        Strong,     // High confidence, direct evidence
   364→        Moderate,   // Reasonable inference
   365→        Weak,       // Speculative but plausible
   366→        VeryWeak    // Tentative, needs verification
   367→    } strength{Strength::Moderate};
   368→    
   369→    double strengthScore() const {
   370→        switch (strength) {
   371→            case Strength::Strong: return 0.9;
   372→            case Strength::Moderate: return 0.7;
   373→            case Strength::Weak: return 0.4;
   374→            case Strength::VeryWeak: return 0.2;
   375→        }
   376→        return 0.0;
   377→    }
   378→    
   379→    std::string toJson() const {
   380→        std::stringstream json;
   381→        json << "{\n";
   382→        json << "  \"nodes\": " << node_ids.size() << ",\n";
   383→        json << "  \"edges\": " << edges.size() << ",\n";
   384→        json << "  \"confidence\": " << std::fixed << std::setprecision(2) 
   385→             << total_confidence << ",\n";
   386→        
   387→        const char* str_str = "";
   388→        switch (strength) {
   389→            case Strength::Strong: str_str = "Strong"; break;
   390→            case Strength::Moderate: str_str = "Moderate"; break;
   391→            case Strength::Weak: str_str = "Weak"; break;
   392→            case Strength::VeryWeak: str_str = "VeryWeak"; break;
   393→        }
   394→        json << "  \"strength\": \"" << str_str << "\"\n";
   395→        
   396→        json << "}";
   397→        return json.str();
   398→    }
   399→};

   448→// ============================================================================
   449→// Causal Dependency Graph - Main Class
   450→// ============================================================================
   451→
   452→class CausalDependencyGraph {
   453→public:
   454→    struct Config {
   455→        size_t max_nodes{100000};
   456→        size_t max_edges{500000};
   457→        bool auto_detect_cross_layer{true};
   458→        bool infer_temporal_relations{true};
   459→        int max_path_length{10};  // Max nodes in a causal path
   460→        
   461→        static Config defaultConfig() {
   462→            return {100000, 500000, true, true, 10};
   463→        }
   464→    };
   465→    
   466→    explicit CausalDependencyGraph(const Config& config = Config())
   467→        : m_config(config), m_enabled(true) {}
   468→    
   469→    void enable() { m_enabled.store(true); }
   470→    void disable() { m_enabled.store(false); }
   471→    bool isEnabled() const { return m_enabled.load(); }
   472→    
   473→    void clear() {
   474→        std::unique_lock lock(m_mutex);
   475→        m_nodes.clear();
   476→        m_edges.clear();
   477→        m_adjacency.clear();
   478→    }
   479→    
   480→    // ========================================================================
   481→    // Node Management
   482→    // ========================================================================
   483→    
   484→    /**
   485→     * Add a node to the graph
   486→     */
   487→    std::string addNode(
   488→        NodeType type,
   489→        const std::string& name,
   490→        EmulationLayer layer,
   491→        const SourceLocation& loc = DEBUG_HERE(),
   492→        Subsystem subs = Subsystem::Unknown) {
   493→        
   494→        if (!isEnabled()) return "";
   495→        
   496→        GraphNode node;
   497→        node.type = type;
   498→        node.name = name;
   499→        node.layer = layer;
   500→        node.location = loc;
   501→        node.subsystem = subs;
   502→        
   503→        std::unique_lock lock(m_mutex);
   504→        
   505→        if (m_nodes.size() >= m_config.max_nodes) return "";
   506→        
   507→        m_nodes[node.id] = node;
   508→        m_adjacency[node.id];  // Initialize adjacency list
   509→        
   510→        return node.id;
   511→    }
   512→    
   513→    /**
   514→     * Get node by ID
   515→     */
   516→    std::optional<GraphNode> getNode(const std::string& id) const {
   517→        std::shared_lock lock(m_mutex);
   518→        
   519→        auto it = m_nodes.find(id);
   520→        if (it != m_nodes.end()) return it->second;
   521→        return std::nullopt;
   522→    }
   523→    
   524→    /**
   525→     * Find nodes matching criteria
   526→     */
   527→    std::vector<GraphNode> findNodes(
   528→        std::optional<NodeType> type_filter = std::nullopt,
   529→        std::optional<EmulationLayer> layer_filter = std::nullopt,
   530→        std::optional<Subsystem> subsystem_filter = std::nullopt) const {
   531→        
   532→        std::shared_lock lock(m_mutex);
   533→        
   534→        std::vector<GraphNode> result;
   535→        for (const auto& [id, node] : m_nodes) {
   536→            if (type_filter.has_value() && node.type != *type_filter) continue;
   537→            if (layer_filter.has_value() && node.layer != *layer_filter) continue;
   538→            if (subsystem_filter.has_value() && node.subsystem != *subsystem_filter) continue;
   539→            result.push_back(node);
   540→        }
   541→        
   542→        return result;
   543→    }
   544→    
   545→    // ========================================================================
   546→    // Edge Management (Causal Relationships)
   547→    // ========================================================================
   548→    
   549→    /**
   550→     * Add an edge (causal relationship)
   551→     */
   552→    std::string addEdge(
   553→        const std::string& source_id,
   554→        const std::string& target_id,
   555→        EdgeType type,
   556→        double confidence = 1.0,
   557→        const std::string& reasoning = "") {
   558→        
   559→        if (!isEnabled()) return "";
   560→        
   561→        GraphEdge edge;
   562→        edge.source_id = source_id;
   563→        edge.target_id = target_id;
   564→        edge.type = type;
   565→        edge.confidence = confidence;
   566→        edge.reasoning = reasoning;
   567→        
   568→        // Auto-detect cross-layer transitions
   569→        auto source_node = getNode(source_id);
   570→        auto target_node = getNode(target_id);
   571→        
   572→        if (source_node && target_node &&
   573→            m_config.auto_detect_cross_layer &&
   574→            source_node->layer != target_node->layer) {
   575→            
   576→            if (isHigherLayer(source_node->layer, target_node->layer)) {
   577→                edge.attributes["cross_layer"] = "down";
   578→            } else {
   579→                edge.attributes["cross_layer"] = "up";
   580→            }
   581→        }
   582→        
   583→        std::unique_lock lock(m_mutex);
   584→        
   585→        if (m_edges.size() >= m_config.max_edges) return "";
   586→        
   587→        m_edges[edge.id] = edge;
   588→        m_adjacency[source_id].insert(target_id);
   589→        
   590→        return edge.id;
   591→    }
   592→    
   593→    /**
   594→     * Convenience: Record that A caused B
   595→     */
   596→    std::string recordCausation(
   597→        const std::string& cause_id,
   598→        const std::string& effect_id,
   599→        double confidence = 0.8,
   600→        const std::string& evidence = "") {
   601→        
   602→        return addEdge(cause_id, effect_id, EdgeType::CausedBy, confidence, evidence);
   603→    }
   604→    
   605→    /**
   606→     * Convenience: Record that A corrupted B's data
   607→     */
   608→    std::string recordCorruption(
   609→        const std::string& corruptor_id,
   610→        const std::string& victim_id,
   611→        double confidence = 0.9,
   612→        const std::string& details = "") {
   613→        
   614→        return addEdge(corruptor_id, victim_id, EdgeType::CorruptedBy, confidence, details);
   615→    }

   617→    // ========================================================================
   618→    // Query Interface - Finding Root Causes
   619→    // ========================================================================
   620→    
   621→    /**
   622→     * Find all paths from potential causes to a symptom
   623→     */
   624→    std::vector<CausalPath> findPathsToSymptom(
   625→        const std::string& symptom_id,
   626→        int max_paths = 10,
   627→        int max_length = -1) const {
   628→        
   629→        if (max_length < 0) max_length = m_config.max_path_length;
   630→        
   631→        std::shared_lock lock(m_mutex);
   632→        
   633→        std::vector<CausalPath> all_paths;
   634→        
   635→        // BFS from symptom backwards (following reverse edges)
   636→        std::queue<std::pair<std::string, std::vector<GraphEdge>>> queue;
   637→        queue.push({symptom_id, {}});
   638→        
   639→        std::set<std::string> visited;
   640→        visited.insert(symptom_id);
   641→        
   642→        while (!queue.empty() && all_paths.size() < static_cast<size_t>(max_paths)) {
   643→            auto [current_id, path_so_far] = queue.front();
   644→            queue.pop();
   645→            
   646→            if (path_so_far.size() > static_cast<size_t>(max_length)) continue;
   647→            
   648→            // Look for nodes that have edges TO current_id (reverse traversal)
   649→            for (const auto& [edge_id, edge] : m_edges) {
   650→                if (edge.target_id == current_id && 
   651→                    visited.find(edge.source_id) == visited.end()) {
   652→                    
   653→                    visited.insert(edge.source_id);
   654→                    
   655→                    CausalPath new_path;
   656→                    new_path.edges = path_so_far;
   657→                    new_path.edges.push_back(edge);
   658→                    new_path.node_ids.push_back(edge.source_id);
   659→                    new_path.node_ids.push_back(current_id);  // Will be deduplicated
   660→                    
   661→                    // Calculate path properties
   662→                    calculatePathProperties(new_path);
   663→                    
   664→                    // Check if this could be a root cause
   665→                    auto source_node_it = m_nodes.find(edge.source_id);
   666→                    if (source_node_it != m_nodes.end()) {
   667→                        const auto& source_node = source_node_it->second;
   668→                        
   669→                        // Good root cause candidates:
   670→                        // - In a different layer than symptom
   671→                        // - Is an error/corruption itself
   672→                        // - Has no incoming edges (origin)
   673→                        
   674→                        auto symptom_node_it = m_nodes.find(symptom_id);
   675→                        if (symptom_node_it != m_nodes.end()) {
   676→                            if (source_node.layer != symptom_node_it->second.layer) {
   677→                                new_path.cross_layer_transitions++;
   678→                            }
   679→                        }
   680→                        
   681→                        if (source_node.type == NodeType::Error ||
   682→                            source_node.type == NodeType::Corruption ||
   683→                            source_node.type == NodeType::Crash) {
   684→                            all_paths.push_back(new_path);
   685→                        }
   686→                        
   687→                        // Also add paths that reach origin nodes
   688→                        bool has_incoming = false;
   689→                        for (const auto& [e_id, e] : m_edges) {
   690→                            if (e.target_id == edge.source_id) {
   691→                                has_incoming = true;
   692→                                break;
   693→                            }
   694→                        }
   695→                        
   696→                        if (!has_incoming && !path_so_far.empty()) {
   697→                            // This is an origin node - good candidate
   698→                            all_paths.push_back(new_path);
   699→                        }
   700→                    }
   701→                    
   702→                    // Continue searching from this node
   703→                    queue.push({edge.source_id, new_path.edges});
   704→                }
   705→            }
   706→        }
   707→        
   708→        // Sort by confidence/strength
   709→        std::sort(all_paths.begin(), all_paths.end(),
   710→            [](const CausalPath& a, const CausalPath& b) {
   711→                return a.total_confidence > b.total_confidence;
   712→            });
   713→        
   714→        return all_paths;
   715→    }

   717→    /**
   718→     * Perform full root cause analysis for a symptom
   719→     */
   720→    RootCauseAnalysis analyzeRootCause(const std::string& symptom_id) {
   721→        RootCauseAnalysis analysis;
   722→        
   723→        auto symptom_opt = getNode(symptom_id);
   724→        if (!symptom_opt) {
   725→            return analysis;  // Invalid symptom ID
   726→        }
   727→        
   728→        analysis.symptom_node = *symptom_opt;
   729→        
   730→        // Find candidate paths
   731→        analysis.candidate_paths = findPathsToSymptom(symptom_id, 50);
   732→        
   733→        // Score and rank each unique endpoint as a candidate
   734→        std::map<std::string, RootCauseAnalysis::Candidate> candidates;
   735→        
   736→        for (const auto& path : analysis.candidate_paths) {
   737→            if (path.node_ids.empty()) continue;
   738→            
   739→            std::string candidate_id = path.node_ids.front();  // Origin of path
   740→            
   741→            auto it = candidates.find(candidate_id);
   742→            if (it == candidates.end()) {
   743→                RootCauseAnalysis::Candidate cand;
   744→                
   745→                auto node_opt = getNode(candidate_id);
   746→                if (node_opt) {
   747→                    cand.node = *node_opt;
   748→                    cand.path = path;
   749→                    
   750→                    // Score based on multiple factors
   751→                    cand.overall_score = calculateCandidateScore(cand);
   752→                    
   753→                    candidates[candidate_id] = cand;
   754→                }
   755→            } else {
   756→                // Keep best path for this candidate
   757→                if (path.total_confidence > it->second.path.total_confidence) {
   758→                    it->second.path = path;
   759→                }
   760→            }
   761→        }
   762→        
   763→        // Move to vector and sort
   764→        for (auto& [id, cand] : candidates) {
   765→            analysis.ranked_candidates.push_back(std::move(cand));
   766→        }
   767→        
   768→        std::sort(analysis.ranked_candidates.begin(), analysis.ranked_candidates.end(),
   769→            std::greater<RootCauseAnalysis::Candidate>());
   770→        
   771→        // Set most likely cause
   772→        if (!analysis.ranked_candidates.empty()) {
   773→            analysis.most_likely_root_cause = analysis.ranked_candidates.front().node;
   774→            analysis.most_likely_path = analysis.ranked_candidates.front().path;
   775→            analysis.confidence_in_analysis = analysis.ranked_candidates.front().overall_score;
   776→        }
   777→        
   778→        // Separate weak candidates as rejected
   779→        for (const auto& cand : analysis.reanked_candidates) {
   780→            if (cand.overall_score < 0.3) {
   781→                analysis.rejected_candidates.push_back(cand);
   782→                analysis.rejection_reasons.push_back(
   783→                    "Low confidence score: " + std::to_string(cand.overall_score));
   784→            }
   785→        }
   786→        
   787→        return analysis;
   788→    }

   790→    /**
   791→     * Find what a given node could have caused (forward traversal)
   792→     */
   793→    std::vector<GraphNode> findPossibleEffects(
   794→        const std::string& cause_id,
   795→        int max_depth = 3) const {
   796→        
   797→        std::shared_lock lock(m_mutex);
   798→        
   799→        std::vector<GraphNode> effects;
   800→        std::set<std::string> visited;
   801→        
   802→        std::queue<std::pair<std::string, int>> queue;
   803→        queue.push({cause_id, 0});
   804→        visited.insert(cause_id);
   805→        
   806→        while (!queue.empty()) {
   807→            auto [current_id, depth] = queue.front();
   808→            queue.pop();
   809→            
   810→            if (depth >= max_depth) continue;
   811→            
   812→            // Follow outgoing edges
   813→            auto adj_it = m_adjacency.find(current_id);
   814→            if (adj_it != m_adjacency.end()) {
   815→                for (const auto& target_id : adj_it->second) {
   816→                    if (visited.find(target_id) == visited.end()) {
   817→                        visited.insert(target_id);
   818→                        
   819→                        auto target_node_it = m_nodes.find(target_id);
   820→                        if (target_node_it != m_nodes.end()) {
   821→                            effects.push_back(target_node_it->second);
   822→                        }
   823→                        
   824→                        queue.push({target_id, depth + 1});
   825→                    }
   826→                }
   827→            }
   828→        }
   829→        
   830→        return effects;
   831→    }

   846→    bool isHigherLayer(EmulationLayer a, EmulationLayer b) const {
   847→        // Guest is highest, Host is lowest
   848→        static const std::map<EmulationLayer, int> layer_order = {
   849→            {EmulationLayer::Guest, 10},
   850→            {EmulationLayer::HLE, 9},
   851→            {EmulationLayer::Kernel, 8},
   852→            {EmulationLayer::FileSystem, 7},
   853→            {EmulationLayer::Network, 7},
   854→            {EmulationLayer::Audio, 6},
   855→            {EmulationLayer::Input, 6},
   856→            {EmulationLayer::GPU, 5},
   857→            {EmulationLayer::Memory, 4},
   858→            {EmulationLayer::CPU, 3},
   859→            {EmulationLayer::Host, 2},
   860→            {EmulationLayer::Diagnostic, 1}
   861→        };
   862→        
   863→        auto it_a = layer_order.find(a);
   864→        auto it_b = layer_order.find(b);
   865→        
   866→        if (it_a == layer_order.end() || it_b == layer_order.end()) return false;
   867→        
   868→        return it_a->second > it_b->second;
   869→    }
   870→    
   871→    void calculatePathProperties(CausalPath& path) const {
   872→        if (path.edges.empty()) {
   873→            path.total_confidence = 0;
   874→            path.strength = CausalPath::Strength::Strong;
   875→            return;
   876→        }
   877→        
   878→        // Confidence multiplies along path
   879→        double conf = 1.0;
   880→        for (const auto& edge : path.edges) {
   881→            conf *= edge.confidence;
   882→        }
   883→        path.total_confidence = conf;
   884→        
   885→        // Count cross-layer transitions
   886→        path.cross_layer_transitions = 0;
   887→        EmulationLayer prev_layer = EmulationLayer::Diagnostic;
   888→        
   889→        for (const auto& edge : path.edges) {
   890→            auto source = m_nodes.find(edge.source_id);
   891→            if (source != m_nodes.end()) {
   892→                if (source->layer != prev_layer && prev_layer != EmulationLayer::Diagnostic) {
   893→                    path.cross_layer_transitions++;
   894→                }
   895→                prev_layer = source->layer;
   896→            }
   897→        }
   898→        
   899→        // Determine strength
   900→        if (conf >= 0.8 && path.cross_layer_transitions >= 2) {
   901→            path.strength = CausalPath::Strength::Strong;
   902→        } else if (conf >= 0.5) {
   903→            path.strength = CausalPath::Strength::Moderate;
   904→        } else if (conf >= 0.2) {
   905→            path.strength = CausalPath::Strength::Weak;
   906→        } else {
   907→            path.strength = CausalPath::Strength::VeryWeak;
   908→        }
   909→    }
   910→    
   911→    double calculateCandidateScore(const RootCauseAnalysis::Candidate& candidate) const {
   912→        double score = 0.0;
   913→        
   914→        // Base score from path confidence
   915→        score += candidate.path.total_confidence * 0.4;
   916→        
   917→        // Bonus for cross-layer origin (different layer than symptom)
   918→        if (candidate.node.layer != analysis.symptom_node.layer) {
   919→            score += 0.2;
   920→            
   921→            // Extra bonus for lower-layer origins (more fundamental)
   922→            if (isHigherLayer(analysis.symptom_node.layer, candidate.node.layer)) {
   923→                score += 0.1;
   924→            }
   925→        }
   926→        
   927→        // Bonus for being an error/corruption/crash node
   928→        if (candidate.node.type == NodeType::Error ||
   929→            candidate.node.type == NodeType::Corruption ||
   930→            candidate.node.type == NodeType::Crash) {
   931→            score += 0.15;
   932→        }
   933→        
   934→        // Bonus for path length (not too short, not too long)
   935→        size_t path_len = candidate.path.edges.size();
   936→        if (path_len >= 2 && path_len <= 5) {
   937→            score += 0.1;
   938→        }
   939→        
   940→        // Normalize to 0-1
   941→        return std::min(score, 1.0);
   942→    }

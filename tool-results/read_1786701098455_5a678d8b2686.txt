     1→/**
     2→ * Memory Provenance Tracker - Priority 1
     3→ * 
     4→ * PROBLEM SOLVED:
     5→ * The biggest missing answer in debugging: "Who wrote this bad value?"
     6→ * 
     7→ * Current debugging shows:
     8→ *   - Crash location (symptom)
     9→ *   - Invalid memory
    10→ *   - Corrupted pointer
    11→ *   
    12→ * Required answer:
    13→ *   - Which subsystem wrote to this address?
    14→ *   - When did it happen?
    15→ *   - What was the old value?
    16→ *   - What was the new value?
    17→ *   - What function caused it?
    18→ * 
    19→ * DESIGN:
    20→ * - Observer only: Does not modify memory operations
    21→ * - Optional: Must be explicitly enabled
    22→ * - Ring buffer: Bounded memory usage
    23→ * - Thread-safe: Can be called from any thread
    24→ * 
    25→ * USAGE:
    26→ *   // Enable tracking for address range
    27→ *   tracker.watchRange(0x2000000000, 0x10000);
    28→ *   
    29→ *   // On memory write (called by existing code)
    30→ *   tracker.recordWrite(0x20000000100, value, size, source);
    31→ *   
    32→ *   // Query when corruption found
    33→ *   auto writers = tracker.whoWrote(0x20000000100);
    34→ */

    49→// Memory Event Types

    52→enum class MemoryEventType : uint8_t {
    53→    Write,
    54→    Read,
    55→    Allocation,
    56→    Free,
    57→    Reallocation,
    58→    ProtectionChange,
    59→    CorruptionDetected,
    60→    AccessViolation
    61→};
    62→
    63→inline std::string memoryEventTypeToString(MemoryEventType type) {
    64→    switch (type) {
    65→        case MemoryEventType::Write: return "Write";
    66→        case MemoryEventType::Read: return "Read";
    67→        case MemoryEventType::Allocation: return "Allocation";
    68→        case MemoryEventType::Free: return "Free";
    69→        case MemoryEventType::Reallocation: return "Reallocation";
    70→        case MemoryEventType::ProtectionChange: return "ProtectionChange";
    71→        case MemoryEventType::CorruptionDetected: return "CorruptionDetected";
    72→        case MemoryEventType::AccessViolation: return "AccessViolation";
    73→    }
    74→    return "Unknown";
    75→}

    77→// ============================================================================
    78→// Memory Value Representation
    79→// ============================================================================
    80→
    81→/**
    82→ * Represents a memory value with type information
    83→ */
    84→struct MemoryValue {
    85→    std::vector<uint8_t> bytes;
    86→    size_t size{0};
    87→    
    88→    // Typed accessors
    89→    uint8_t asUint8() const { return bytes.size() >= 1 ? bytes[0] : 0; }
    90→    uint16_t asUint16() const { 
    91→        return bytes.size() >= 2 ? 
    92→            (static_cast<uint16_t>(bytes[1]) << 8) | bytes[0] : 0; 
    93→    }
    94→    uint32_t asUint32() const {
    95→        if (bytes.size() < 4) return 0;
    96→        return (static_cast<uint32_t>(bytes[3]) << 24) |
    97→               (static_cast<uint32_t>(bytes[2]) << 16) |
    98→               (static_cast<uint32_t>(bytes[1]) << 8) |
    99→               bytes[0];
   100→    }
   101→    uint64_t asUint64() const {
   102→        if (bytes.size() < 8) return asUint32();
   103→        return (static_cast<uint64_t>(asUint32()) << 32) | asUint32();
   104→    }
   105→    
   106→    /**
   107→     * Convert to hex string for display/JSON
   108→     */
   109→    std::string toHex() const {
   110→        std::stringstream ss;
   111→        ss << "0x";
   112→        // Print in reverse order (big-endian display)
   113→        for (auto it = bytes.rbegin(); it != bytes.rend(); ++it) {
   114→            ss << std::hex << std::setfill('0') << std::setw(2) << static_cast<int>(*it);
   115→        }
   116→        return ss.str();
   117→    }
   118→    
   119→    static MemoryValue fromBytes(const void* data, size_t size) {
   120→        MemoryValue val;
   121→        val.size = size;
   122→        val.bytes.assign(
   123→            static_cast<const uint8_t*>(data),
   124→            static_cast<const uint8_t*>(data) + size
   125→        );
   126→        return val;
   127→    }
   128→    
   129→    static MemoryValue fromUint64(uint64_t value, size_t size = 8) {
   130→        MemoryValue val;
   131→        val.size = size;
   132→        val.bytes.resize(size);
   133→        for (size_t i = 0; i < size; ++i) {
   134→            val.bytes[i] = (value >> (i * 8)) & 0xFF;
   135→        }
   136→        return val;
   137→    }
   138→};

   140→// ============================================================================
   141→// Memory Event - Single Memory Operation Record
   142→// ============================================================================
   143→
   144→struct MemoryEvent {
   145→    std::string id;
   146→    DebugTimestamp timestamp;
   147→    MemoryEventType type;
   148→    
   149→    // Address info
   150→    uint64_t address{0};
   151→    size_t size{0};
   152→    MemoryValue old_value;
   153→    MemoryValue new_value;
   154→    
   155→    // Provenance
   156→    Subsystem subsystem{Subsystem::Unknown};
   157→    SourceLocation source;
   158→    std::string thread_id;
   159→    
   160→    // Context
   161→    std::map<std::string, std::string> metadata;
   162→    
   163→    // For allocations
   164→    uint64_t allocation_base{0};
   165→    size_t allocation_size{0};
   166→    std::string allocation_tag;
   167→    
   168→    static std::string generateId() {
   169→        static std::atomic<uint64_t> counter{0};
   170→        auto now = std::chrono::high_resolution_clock::now().time_since_epoch().count();
   171→        return "mem_" + std::to_string(now) + "_" + std::to_string(++counter);
   172→    }
   173→    
   174→    MemoryEvent() {
   175→        id = generateId();
   176→        timestamp = DebugTimestamp::now();
   177→    }
   178→    
   179→    /**
   180→     * Convert to JSON for export
   181→     */
   182→    std::string toJson() const {
   183→        std::stringstream json;
   184→        json << "{\n";
   185→        json << "  \"id\": \"" << JsonUtils::escapeJson(id) << "\",\n";
   186→        json << "  \"timestamp\": \"" << timestamp.toISO8601() << "\",\n";
   187→        json << "  \"frame\": " << timestamp.frame_number << ",\n";
   188→        json << "  \"type\": \"" << memoryEventTypeToString(type) << "\",\n";
   189→        json << "  \"address\": \"0x" << std::hex << address << std::dec << "\",\n";
   190→        json << "  \"size\": " << size << ",\n";
   191→        json << "  \"old_value\": \"" << JsonUtils::escapeJson(old_value.toHex()) << "\",\n";
   192→        json << "  \"new_value\": \"" << JsonUtils::escapeJson(new_value.toHex()) << "\",\n";
   193→        json << "  \"subsystem\": \"" << subsystemToString(subsystem) << "\",\n";
   194→        json << "  \"source\": \"" << JsonUtils::escapeJson(source.toString()) << "\",\n";
   195→        json << "  \"thread_id\": \"" << JsonUtils::escapeJson(thread_id) << "\"\n";
   196→        json << "}";
   197→        return json.str();
   198→    }
   199→};

   201→// ============================================================================
   202→// Address Range Watch Configuration
   203→// ============================================================================
   204→
   205→struct AddressRange {
   206→    uint64_t start{0};
   207→    uint64_t end{0};  // Exclusive
   208→    std::string description;
   209→    bool enabled{true};
   210→    
   211→    bool contains(uint64_t addr, size_t size = 1) const {
   212→        return addr >= start && (addr + size) <= end;
   213→    }
   214→    
   215→    bool overlaps(uint64_t addr, size_t size) const {
   216→        uint64_t range_start = addr;
   217→        uint64_t range_end = addr + size;
   218→        return range_start < end && range_end > start;
   219→    }
   220→    
   221→    size_t length() const { return end - start; }
   222→};

   311→
   312→class MemoryProvenanceTracker {
   313→public:
   314→    struct Config {
   315→        size_t max_events_per_address{1000};      // Ring buffer per address
   316→        size_t max_total_events{1000000};         // Global limit
   317→        size_t max_watch_ranges{100};             // Max watched ranges
   318→        bool track_reads{false};                  // Read tracking (expensive)
   319→        bool auto_watch_allocations{true};        // Track all allocs/frees
   320→        bool detect_corruption_patterns{true};     // Auto-detect suspicious patterns
   321→        
   322→        static Config sensitive() {
   323→            return {10000, 10000000, 500, true, true, true};
   324→        }
   325→        
   326→        static Config minimal() {
   327→            return {100, 10000, 10, false, false, false};
   328→        }
   329→    };
   330→    
   331→    explicit MemoryProvenanceTracker(const Config& config = Config())
   332→        : m_config(config), m_enabled(false), m_total_events(0) {}
   333→    
   334→    // ========================================================================
   335→    // Control Interface
   336→    // ========================================================================
   337→    
   338→    /**
   339→     * Enable/disable tracking
   340→     * When disabled, recordXxx calls are no-ops (minimal overhead)
   341→     */
   342→    void enable() { m_enabled.store(true); }
   343→    void disable() { m_enabled.store(false); }
   344→    bool isEnabled() const { return m_enabled.load(); }
   345→    
   346→    /**
   347→     * Clear all tracked data
   348→     */
   349→    void clear() {
   350→        std::unique_lock lock(m_mutex);
   351→        m_events.clear();
   352→        m_address_index.clear();
   353→        m_watch_ranges.clear();
   354→        m_allocations.clear();
   355→        m_total_events = 0;
   356→    }
   357→    
   358→    // ========================================================================
   359→    // Range Watching
   360→    // ========================================================================
   361→    
   362→    /**
   363→     * Watch an address range for changes
   364→     */
   365→    bool watchRange(uint64_t start, uint64_t end, const std::string& description = "") {
   366→        std::unique_lock lock(m_mutex);
   367→        
   368→        if (m_watch_ranges.size() >= m_config.max_watch_ranges) {
   369→            return false;
   370→        }
   371→        
   372→        AddressRange range;
   373→        range.start = start;
   374→        range.end = end;
   375→        range.description = description;
   376→        
   377→        m_watch_ranges.push_back(range);
   378→        return true;
   379→    }
   380→    
   381→    /**
   382→     * Remove a watch range
   383→     */
   384→    bool unwatchRange(uint64_t start, uint64_t end) {
   385→        std::unique_lock lock(m_mutex);
   386→        
   387→        auto it = std::find_if(m_watch_ranges.begin(), m_watch_ranges.end(),
   388→            [start, end](const AddressRange& r) {
   389→                return r.start == start && r.end == end;
   390→            });
   391→        
   392→        if (it != m_watch_ranges.end()) {
   393→            m_watch_ranges.erase(it);
   394→            return true;
   395→        }
   396→        return false;
   397→    }
   398→    
   399→    /**
   400→     * Check if address is being watched
   401→     */
   402→    bool isWatched(uint64_t address, size_t size = 1) const {
   403→        std::shared_lock lock(m_mutex);
   404→        
   405→        for (const auto& range : m_watch_ranges) {
   406→            if (range.contains(address, size)) {
   407→                return true;
   408→            }
   409→        }
   410→        return false;
   411→    }
   412→    
   413→    // ========================================================================
   414→    // Event Recording (Call These From Existing Code)
   415→    // ========================================================================
   416→    
   417→    /**
   418→     * Record a memory write event
   419→     * Call this AFTER actual write completes, passing old and new values
   420→     */
   421→    void recordWrite(uint64_t address, const void* data, size_t size,
   422→                     Subsystem subs = Subsystem::Unknown,
   423→                     const SourceLocation& loc = SourceLocation(),
   424→                     const std::string& thread_id = "") {
   425→        
   426→        if (!isEnabled()) return;
   427→        if (!shouldTrack(address, size)) return;
   428→        
   429→        MemoryEvent event;
   430→        event.type = MemoryEventType::Write;
   431→        event.address = address;
   432→        event.size = size;
   433→        event.new_value = MemoryValue::fromBytes(data, size);
   434→        event.subsystem = subs;
   435→        event.source = loc;
   436→        event.thread_id = thread_id.empty() ? getCurrentThreadId() : thread_id;
   437→        
   438→        // Try to get old value (may not always be available)
   439→        // In real implementation, would read from actual memory
   440→        event.old_value = MemoryValue::fromUint64(0, size);  // Placeholder
   441→        
   442→        addEvent(event);
   443→    }
   444→    
   445→    /**
   446→     * Record a memory write with known old value
   447→     */
   448→    void recordWriteWithOldValue(uint64_t address, 
   449→                                  const MemoryValue& old_val,
   450→                                  const MemoryValue& new_val,
   451→                                  Subsystem subs = Subsystem::Unknown,
   452→                                  const SourceLocation& loc = SourceLocation(),
   453→                                  const std::string& thread_id = "") {
   454→        
   455→        if (!isEnabled()) return;
   456→        if (!shouldTrack(address, new_val.size)) return;
   457→        
   458→        MemoryEvent event;
   459→        event.type = MemoryEventType::Write;
   460→        event.address = address;
   461→        event.size = new_val.size;
   462→        event.old_value = old_val;
   463→        event.new_value = new_val;
   464→        event.subsystem = subs;
   465→        event.source = loc;
   466→        event.thread_id = thread_id.empty() ? getCurrentThreadId() : thread_id;
   467→        
   468→        addEvent(event);
   469→    }
   470→    
   471→    /**
   472→     * Record a memory read event (only if reads are tracked)
   473→     */
   474→    void recordRead(uint64_t address, const void* data, size_t size,
   475→                    Subsystem subs = Subsystem::Unknown,
   476→                    const SourceLocation& loc = SourceLocation(),
   477→                    const std::string& thread_id = "") {
   478→        
   479→        if (!isEnabled()) return;
   480→        if (!m_config.track_reads) return;
   481→        if (!shouldTrack(address, size)) return;
   482→        
   483→        MemoryEvent event;
   484→        event.type = MemoryEventType::Read;
   485→        event.address = address;
   486→        event.size = size;
   487→        event.new_value = MemoryValue::fromBytes(data, size);
   488→        event.subsystem = subs;
   489→        event.source = loc;
   490→        event.thread_id = thread_id.empty() ? getCurrentThreadId() : thread_id;
   491→        
   492→        addEvent(event);
   493→    }
   494→    
   495→    /**
   496→     * Record a memory allocation
   497→     */
   498→    void recordAllocation(uint64_t address, size_t size,
   499→                          const std::string& tag = "",
   500→                          Subsystem subs = Subsystem::Unknown,
   501→                          const SourceLocation& loc = SourceLocation()) {
   502→        
   503→        if (!isEnabled()) return;
   504→        if (!m_config.auto_watch_allocations) return;
   505→        
   506→        MemoryEvent event;
   507→        event.type = MemoryEventType::Allocation;
   508→        event.address = address;
   509→        event.size = size;
   510→        event.allocation_base = address;
   511→        event.allocation_size = size;
   512→        event.allocation_tag = tag;
   513→        event.subsystem = subs;
   514→        event.source = loc;
   515→        event.thread_id = getCurrentThreadId();
   516→        
   517→        addEvent(event);
   518→        
   519→        // Auto-watch this allocation
   520→        std::unique_lock lock(m_mutex);
   521→        m_allocations[address] = {address, address + size, tag, true};
   522→    }
   523→    
   524→    /**
   525→     * Record a memory free
   526→     */
   527→    void recordFree(uint64_t address,
   528→                    Subsystem subs = Subsystem::Unknown,
   529→                    const SourceLocation& loc = SourceLocation()) {
   530→        
   531→        if (!isEnabled()) return;
   532→        if (!m_config.auto_watch_allocations) return;
   533→        
   534→        MemoryEvent event;
   535→        event.type = MemoryEventType::Free;
   536→        event.address = address;
   537→        event.subsystem = subs;
   538→        event.source = loc;
   539→        event.thread_id = getCurrentThreadId();
   540→        
   541→        // Look up allocation info
   542→        std::unique_lock lock(m_mutex);
   543→        auto it = m_allocations.find(address);
   544→        if (it != m_allocations.end()) {
   545→            event.allocation_base = it->second.start;
   546→            event.allocation_size = it->second.length();
   547→            event.allocation_tag = it->second.description;
   548→            m_allocations.erase(it);
   549→        }
   550→        
   551→        addEvent(event);
   552→    }

   554→    /**
   555→     * Record detected corruption
   556→     */
   557→    void recordCorruption(uint64_t address, const MemoryValue& corrupt_value,
   558→                           const std::string& detection_method,
   559→                           Severity severity = Severity::Critical) {
   560→        
   561→        // Always record corruption, even if not in watched range
   562→        MemoryEvent event;
   563→        event.type = MemoryEventType::CorruptionDetected;
   564→        event.address = address;
   565→        event.size = corrupt_value.size;
   566→        event.new_value = corrupt_value;
   567→        event.subsystem = Subsystem::DiagnosticSystem;
   568→        event.metadata["detection_method"] = detection_method;
   569→        
   570→        addEvent(event);
   571→    }

   573→    // ========================================================================
   574→    // Query Interface - Answer "Who Wrote This?"
   575→    // ========================================================================
   576→    
   577→    /**
   578→     * Find who wrote to a specific address
   579→     */
   580→    std::vector<WriterInfo> whoWrote(uint64_t address) const {
   581→        std::shared_lock lock(m_mutex);
   582→        
   583→        std::vector<WriterInfo> writers;
   584→        auto events = getEventsForAddress(address);
   585→        
   586→        for (const auto& event : events) {
   587→            if (event.type == MemoryEventType::Write || 
   588→                event.type == MemoryEventType::Allocation) {
   589→                WriterInfo writer;
   590→                writer.event = event;
   591→                
   592→                // Analyze suspicion
   593→                analyzeSuspicion(writer);
   594→                
   595→                writers.push_back(writer);
   596→            }
   597→        }
   598→        
   599→        return writers;
   600→    }
   601→    
   602→    /**
   603→     * Get the last writer to an address
   604→     */
   605→    std::optional<WriterInfo> lastWriter(uint64_t address) const {
   606→        auto writers = whoWrote(address);
   607→        if (writers.empty()) return std::nullopt;
   608→        return writers.back();  // Most recent
   609→    }
   610→    
   611→    /**
   612→     * Get complete history for an address or range
   613→     */
   614→    MemoryHistoryQuery memoryHistory(uint64_t address, size_t range = 1) const {
   615→        MemoryHistoryQuery query;
   616→        query.target_address = address;
   617→        query.query_range = range;
   618→        
   619→        std::shared_lock lock(m_mutex);
   620→        
   621→        // Collect all events in range
   622→        for (uint64_t addr = address; addr < address + range; ++addr) {
   623→            auto events = getEventsForAddress(addr);
   624→            
   625→            for (const auto& event : events) {
   626→                WriterInfo info;
   627→                info.event = event;
   628→                
   629→                if (event.type == MemoryEventType::Write ||
   630→                    event.type == MemoryEventType::Allocation) {
   631→                    query.all_writers.push_back(info);
   632→                    query.total_writes++;
   633→                    analyzeSuspicion(info);
   634→                    if (info.is_suspicious) {
   635→                        query.suspicious_writes.push_back(info);
   636→                    }
   637→                    query.last_writer = info;
   638→                    query.current_value = event.new_value;
   639→                    
   640→                    if (query.first_access_frame == 0 || 
   641→                        event.timestamp.frame_number < query.first_access_frame) {
   642→                        query.first_access_frame = event.timestamp.frame_number;
   643→                    }
   644→                    if (event.timestamp.frame_number > query.last_access_frame) {
   645→                        query.last_access_frame = event.timestamp.frame_number;
   646→                    }
   647→                } else if (event.type == MemoryEventType::Read) {
   648→                    query.all_readers.push_back(info);
   649→                    query.total_reads++;
   650→                }
   651→            }
   652→        }
   653→        
   654→        return query;
   655→    }
   656→    
   657→    /**
   658→     * Find all addresses written by a specific subsystem
   659→     */
   660→    std::vector<uint64_t> addressesWrittenBy(Subsystem subs) const {
   661→        std::shared_lock lock(m_mutex);
   662→        
   663→        std::set<uint64_t> addresses;
   664→        for (const auto& [addr, events] : m_address_index) {
   665→            for (const auto& event : events) {
   666→                if (event.subsystem == subs && 
   667→                    (event.type == MemoryEventType::Write ||
   668→                     event.type == MemoryEventType::Allocation)) {
   669→                    addresses.insert(addr);
   670→                    break;
   671→                }
   672→            }
   673→        }
   674→        
   675→        return std::vector<uint64_t>(addresses.begin(), addresses.end());
   676→    }
   677→    
   678→    /**
   679→     * Find corruption events
   680→     */
   681→    std::vector<MemoryEvent> findCorruptions() const {
   682→        std::shared_lock lock(m_mutex);
   683→        
   684→        std::vector<MemoryEvent> corruptions;
   685→        for (const auto& [addr, events] : m_address_index) {
   686→            for (const auto& event : events) {
   687→                if (event.type == MemoryEventType::CorruptionDetected) {
   688→                    corruptions.push_back(event);
   689→                }
   690→            }
   691→        }
   692→        
   693→        return corruptions;
   694→    }

   724→                    case MemoryEventType::Read: stats.read_events++; break;
   725→                    case MemoryEventType::Allocation: 
   726→                    case MemoryEventType::Free:

   729→                    case MemoryEventType::CorruptionDetected:
   730→                        stats.corruption_events++; break;
   731→                    default: break;
   732→                }
   733→            }
   734→        }
   735→

   809→    bool shouldTrack(uint64_t address, size_t size) const {
   810→        // Always check watched ranges first
   811→        for (const auto& range : m_watch_ranges) {
   812→            if (range.overlaps(address, size)) {
   813→                return true;
   814→            }
   815→        }
   816→        
   817→        // Check if address is in a tracked allocation
   818→        auto it = m_allocations.lower_bound(address);
   819→        if (it != m_allocations.end()) {
   820→            if (it->second.contains(address, size)) {
   821→                return true;
   822→            }
   823→        }
   824→        
   825→        return false;
   826→    }

   828→    void addEvent(const MemoryEvent& event) {
   829→        std::unique_lock lock(m_mutex);
   830→        
   831→        // Check global limit
   832→        if (m_total_events >= m_config.max_total_events) {
   833→            return;  // Or could evict oldest
   834→        }
   835→        
   836→        // Add to address index
   837→        auto& events = m_address_index[event.address];
   838→        events.push_back(event);
   839→        
   840→        // Enforce per-address limit (ring buffer)
   841→        while (events.size() > m_config.max_events_per_address) {
   842→            events.pop_front();
   843→        }
   844→        
   845→        m_total_events++;
   846→        
   847→        // Auto-detect corruption patterns if enabled
   848→        if (m_config.detect_corruption_patterns &&
   849→            event.type == MemoryEventType::Write) {
   850→            detectCorruptionPatterns(event, events);
   851→        }
   852→    }

   854→    std::list<MemoryEvent> getEventsForAddress(uint64_t address) const {
   855→        auto it = m_address_index.find(address);
   856→        if (it != m_address_index.end()) {
   857→            return it->second;
   858→        }
   859→        return {};
   860→    }

   862→    void analyzeSuspicion(WriterInfo& writer) const {
   863→        // Pattern: Writing to recently freed memory
   864→        // Pattern: Writing outside allocation bounds
   865→        // Pattern: GPU writing to CPU memory (or vice versa)
   866→        // Pattern: Large value change in single write
   867→        
   868→        const auto& event = writer.event;
   869→        
   870→        // Check if writing to freed memory
   871→        auto alloc_it = m_allocations.lower_bound(event.address);
   872→        if (alloc_it != m_allocations.begin()) {
   873→            --alloc_it;
   874→            if (alloc_it != m_allocations.end()) {
   875→                const auto& alloc = alloc_it->second;
   876→                if (event.address >= alloc.end) {
   877→                    writer.is_suspicious = true;
   878→                    writer.suspicion_reason = "Write to potentially freed memory region";
   879→                }
   880→            }
   881→        }
   882→        
   883→        // Check for large value changes
   884→        if (event.old_value.size == event.new_value.size &&
   885→            event.old_value.size >= 4) {
   886→            uint64_t old_val = event.old_value.asUint64();
   887→            uint64_t new_val = event.new_value.asUint64();
   888→            
   889→            // If most bits changed, might be corruption
   890→            uint64_t xor_result = old_val ^ new_val;
   891→            int changed_bits = __builtin_popcountll(xor_result);
   892→            
   893→            if (changed_bits > (sizeof(uint64_t) * 8 * 0.7)) {  // 70%+ bits changed
   894→                writer.is_suspicious = true;
   895→                writer.suspicion_reason = "Large value change (" + 
   896→                    std::to_string(changed_bits) + " of " + 
   897→                    std::to_string(sizeof(uint64_t) * 8) + " bits changed)";
   898→            }
   899→        }
   900→        
   901→        // Cross-subsystem writes can be suspicious
   902→        if (event.subsystem == Subsystem::GPU &&
   903→            event.address < 0x100000000ULL) {  // Low memory (likely CPU)
   904→            writer.is_suspicious = true;
   905→            writer.suspicion_reason = "GPU write to low memory address space";
   906→        }
   907→    }
   908→    
   909→    void detectCorruptionPatterns(const MemoryEvent& new_event,
   910→                                   const std::list<MemoryEvent>& existing_events) {
   911→        // Look for pattern: write, write-back different, write again
   912→        // This can indicate race conditions or use-after-free
   913→        
   914→        if (existing_events.size() < 3) return;
   915→        
   916→        auto it = existing_events.rbegin();
   917→        ++it;  // Skip the event we just added
   918→        
   919→        // Check recent writes for inconsistencies
   920→        MemoryValue prev_value;
   921→        int write_count = 0;
   922→        
   923→        for (; it != existing_events.rend() && write_count < 5; ++it) {
   924→            if (it->type == MemoryEventType::Write) {
   925→                write_count++;
   926→                
   927→                if (!prev_value.bytes.empty()) {
   928→                    // Values oscillating between different states?
   929→                    if (prev_value.bytes != it->new_value.bytes &&
   930→                        it->new_value.bytes != new_event.new_value.bytes) {
   931→                        
   932→                        // Could be normal, flag for review
   933→                        // (In production, would have more sophisticated heuristics)
   934→                    }
   935→                }
   936→                
   937→                prev_value = it->new_value;
   938→            }
   939→        }
   940→    }

     8→     8→ *   - Crash location (symptom)

    11→    11→ *   
    12→    12→ * Required answer:
    13→    13→ *   - Which subsystem wrote to this address?

    17→    17→ *   - What function caused it?
    18→    18→ * 
    19→    19→ * DESIGN:
    20→    20→ * - Observer only: Does not modify memory operations
    21→    21→ * - Optional: Must be explicitly enabled
    22→    22→ * - Ring buffer: Bounded memory usage
    23→    23→ * - Thread-safe: Can be called from any thread
    24→    24→ *

    63→    63→inline std::string memoryEventTypeToString(MemoryEventType type) {
    64→    64→    switch (type) {
    65→    65→        case MemoryEventType::Write: return "Write";
    66→    66→        case MemoryEventType::Read: return "Read";
    67→    67→        case MemoryEventType::Allocation: return "Allocation";
    68→    68→        case MemoryEventType::Free: return "Free";
    69→    69→        case MemoryEventType::Reallocation: return "Reallocation";
    70→    70→        case MemoryEventType::ProtectionChange: return "ProtectionChange";
    71→    71→        case MemoryEventType::CorruptionDetected: return "CorruptionDetected";
    72→    72→        case MemoryEventType::AccessViolation: return "AccessViolation";
    73→    73→    }
    74→    74→    return "Unknown";
    75→    75→}

    81→    81→/**
    82→    82→ * Represents a memory value with type information
    83→    83→ */
    84→    84→struct MemoryValue {
    85→    85→    std::vector<uint8_t> bytes;
    86→    86→    size_t size{0};
    87→    87→    
    88→    88→    // Typed accessors
    89→    89→    uint8_t asUint8() const { return bytes.size() >= 1 ? bytes[0] : 0; }
    90→    90→    uint16_t asUint16() const { 
    91→    91→        return bytes.size() >= 2 ? 
    92→    92→            (static_cast<uint16_t>(bytes[1]) << 8) | bytes[0] : 0; 
    93→    93→    }
    94→    94→    uint32_t asUint32() const {
    95→    95→        if (bytes.size() < 4) return 0;
    96→    96→        return (static_cast<uint32_t>(bytes[3]) << 24) |
    97→    97→               (static_cast<uint32_t>(bytes[2]) << 16) |
    98→    98→               (static_cast<uint32_t>(bytes[1]) << 8) |
    99→    99→               bytes[0];
   100→   100→    }
   101→   101→    uint64_t asUint64() const {
   102→   102→        if (bytes.size() < 8) return asUint32();
   103→   103→        return (static_cast<uint64_t>(asUint32()) << 32) | asUint32();
   104→   104→    }
   105→   105→    
   106→   106→    /**
   107→   107→     * Convert to hex string for display/JSON
   108→   108→     */
   109→   109→    std::string toHex() const {
   110→   110→        std::stringstream ss;
   111→   111→        ss << "0x";
   112→   112→        // Print in reverse order (big-endian display)
   113→   113→        for (auto it = bytes.rbegin(); it != bytes.rend(); ++it) {
   114→   114→            ss << std::hex << std::setfill('0') << std::setw(2) << static_cast<int>(*it);
   115→   115→        }
   116→   116→        return ss.str();
   117→   117→    }
   118→   118→    
   119→   119→    static MemoryValue fromBytes(const void* data, size_t size) {
   120→   120→        MemoryValue val;
   121→   121→        val.size = size;
   122→   122→        val.bytes.assign(
   123→   123→            static_cast<const uint8_t*>(data),
   124→   124→            static_cast<const uint8_t*>(data) + size
   125→   125→        );
   126→   126→        return val;
   127→   127→    }
   128→   128→    
   129→   129→    static MemoryValue fromUint64(uint64_t value, size_t size = 8) {
   130→   130→        MemoryValue val;
   131→   131→        val.size = size;
   132→   132→        val.bytes.resize(size);
   133→   133→        for (size_t i = 0; i < size; ++i) {
   134→   134→            val.bytes[i] = (value >> (i * 8)) & 0xFF;
   135→   135→        }
   136→   136→        return val;
   137→   137→    }
   138→   138→};

   216→   216→        uint64_t range_start = addr;
   217→   217→        uint64_t range_end = addr + size;
   218→   218→        return range_start < end && range_end > start;
   219→   219→    }
   220→   220→    
   221→   221→    size_t length() const { return end - start; }

   312→   312→class MemoryProvenanceTracker {

   314→   314→    struct Config {

   317→   317→        size_t max_watch_ranges;

   320→   320→        bool detect_corruption_patterns;
   321→   321→

   323→   323→            : max_events_per_address(1000)
   324→   324→            , max_total_events(1000000)
   325→   325→            , max_watch_ranges(100)
   326→   326→            , track_reads(false)
   327→   327→            , auto_watch_allocations(true)
   328→   328→            , detect_corruption_patterns(true) {}
   329→   329→        
   330→   330→        static Config sensitive() {
   331→   331→            return {10000, 10000000, 500, true, true, true};
   332→   332→        }
   333→   333→        
   334→   334→        static Config minimal() {
   335→   335→            return {100, 10000, 10, false, false, false};
   336→   336→        }
   337→   337→    };
   338→   338→    
   339→   339→    explicit MemoryProvenanceTracker(const Config& config = Config())
   340→   340→        : m_config(config), m_enabled(false), m_total_events(0) {}
   341→   341→    
   342→   342→    // ========================================================================
   343→   343→    // Control Interface
   344→   344→    // ========================================================================
   345→   345→    
   346→   346→    /**
   347→   347→     * Enable/disable tracking
   348→   348→     * When disabled, recordXxx calls are no-ops (minimal overhead)
   349→   349→     */
   350→   350→    void enable() { m_enabled.store(true); }
   351→   351→    void disable() { m_enabled.store(false); }
   352→   352→    bool isEnabled() const { return m_enabled.load(); }
   353→   353→    
   354→   354→    /**
   355→   355→     * Clear all tracked data
   356→   356→     */
   357→   357→    void clear() {
   358→   358→        std::unique_lock lock(m_mutex);
   359→   359→        m_events.clear();
   360→   360→        m_address_index.clear();
   361→   361→        m_watch_ranges.clear();
   362→   362→        m_allocations.clear();
   363→   363→        m_total_events = 0;

   365→   365→    
   366→   366→    // ========================================================================
   367→   367→    // Range Watching
   368→   368→    // ========================================================================
   369→   369→    
   370→   370→    /**
   371→   371→     * Watch an address range for changes
   372→   372→     */
   373→   373→    bool watchRange(uint64_t start, uint64_t end, const std::string& description = "") {
   374→   374→        std::unique_lock lock(m_mutex);
   375→   375→        
   376→   376→        if (m_watch_ranges.size() >= m_config.max_watch_ranges) {
   377→   377→            return false;
   378→   378→        }
   379→   379→        
   380→   380→        AddressRange range;
   381→   381→        range.start = start;
   382→   382→        range.end = end;
   383→   383→        range.description = description;
   384→   384→        
   385→   385→        m_watch_ranges.push_back(range);
   386→   386→        return true;
   387→   387→    }
   388→   388→    
   389→   389→    /**
   390→   390→     * Remove a watch range
   391→   391→     */
   392→   392→    bool unwatchRange(uint64_t start, uint64_t end) {
   393→   393→        std::unique_lock lock(m_mutex);
   394→   394→        
   395→   395→        auto it = std::find_if(m_watch_ranges.begin(), m_watch_ranges.end(),
   396→   396→            [start, end](const AddressRange& r) {
   397→   397→                return r.start == start && r.end == end;
   398→   398→            });
   399→   399→        
   400→   400→        if (it != m_watch_ranges.end()) {
   401→   401→            m_watch_ranges.erase(it);
   402→   402→            return true;
   403→   403→        }
   404→   404→        return false;
   405→   405→    }
   406→   406→    
   407→   407→    /**
   408→   408→     * Check if address is being watched
   409→   409→     */
   410→   410→    bool isWatched(uint64_t address, size_t size = 1) const {
   411→   411→        std::shared_lock lock(m_mutex);
   412→   412→        
   413→   413→        for (const auto& range : m_watch_ranges) {
   414→   414→            if (range.contains(address, size)) {
   415→   415→                return true;
   416→   416→            }
   417→   417→        }
   418→   418→        return false;
   419→   419→    }

   432→   432→                     const std::string& thread_id = "") {

   438→   438→        event.type = MemoryEventType::Write;
   439→   439→        event.address = address;
   440→   440→        event.size = size;
   441→   441→        event.new_value = MemoryValue::fromBytes(data, size);
   442→   442→        event.subsystem = subs;
   443→   443→        event.source = loc;
   444→   444→        event.thread_id = thread_id.empty() ? getCurrentThreadId() : thread_id;
   445→   445→        
   446→   446→        // Try to get old value (may not always be available)
   447→   447→        // In real implementation, would read from actual memory
   448→   448→        event.old_value = MemoryValue::fromUint64(0, size);  // Placeholder
   449→   449→        
   450→   450→        addEvent(event);
   451→   451→    }

   479→   479→    /**
   480→   480→     * Record a memory read event (only if reads are tracked)
   481→   481→     */
   482→   482→    void recordRead(uint64_t address, const void* data, size_t size,
   483→   483→                    Subsystem subs = Subsystem::Unknown,
   484→   484→                    const SourceLocation& loc = SourceLocation(),
   485→   485→                    const std::string& thread_id = "") {
   486→   486→        
   487→   487→        if (!isEnabled()) return;
   488→   488→        if (!m_config.track_reads) return;
   489→   489→        if (!shouldTrack(address, size)) return;
   490→   490→        
   491→   491→        MemoryEvent event;
   492→   492→        event.type = MemoryEventType::Read;
   493→   493→        event.address = address;
   494→   494→        event.size = size;
   495→   495→        event.new_value = MemoryValue::fromBytes(data, size);
   496→   496→        event.subsystem = subs;
   497→   497→        event.source = loc;

   499→   499→

   502→   502→    
   503→   503→    /**
   504→   504→     * Record a memory allocation
   505→   505→     */
   506→   506→    void recordAllocation(uint64_t address, size_t size,
   507→   507→                          const std::string& tag = "",
   508→   508→                          Subsystem subs = Subsystem::Unknown,
   509→   509→                          const SourceLocation& loc = SourceLocation()) {
   510→   510→        
   511→   511→        if (!isEnabled()) return;
   512→   512→        if (!m_config.auto_watch_allocations) return;
   513→   513→        
   514→   514→        MemoryEvent event;
   515→   515→        event.type = MemoryEventType::Allocation;
   516→   516→        event.address = address;
   517→   517→        event.size = size;
   518→   518→        event.allocation_base = address;
   519→   519→        event.allocation_size = size;
   520→   520→        event.allocation_tag = tag;
   521→   521→        event.subsystem = subs;
   522→   522→        event.source = loc;
   523→   523→        event.thread_id = getCurrentThreadId();
   524→   524→        
   525→   525→        addEvent(event);
   526→   526→        
   527→   527→        // Auto-watch this allocation
   528→   528→        std::unique_lock lock(m_mutex);
   529→   529→        m_allocations[address] = {address, address + size, tag, true};
   530→   530→    }
   531→   531→    
   532→   532→    /**
   533→   533→     * Record a memory free
   534→   534→     */
   535→   535→    void recordFree(uint64_t address,
   536→   536→                    Subsystem subs = Subsystem::Unknown,
   537→   537→                    const SourceLocation& loc = SourceLocation()) {
   538→   538→        
   539→   539→        if (!isEnabled()) return;
   540→   540→        if (!m_config.auto_watch_allocations) return;
   541→   541→        
   542→   542→        MemoryEvent event;
   543→   543→        event.type = MemoryEventType::Free;
   544→   544→        event.address = address;
   545→   545→        event.subsystem = subs;
   546→   546→        event.source = loc;
   547→   547→        event.thread_id = getCurrentThreadId();
   548→   548→        
   549→   549→        // Look up allocation info
   550→   550→        std::unique_lock lock(m_mutex);
   551→   551→        auto it = m_allocations.find(address);
   552→   552→        if (it != m_allocations.end()) {
   553→   553→            event.allocation_base = it->second.start;
   554→   554→            event.allocation_size = it->second.length();
   555→   555→            event.allocation_tag = it->second.description;
   556→   556→            m_allocations.erase(it);
   557→   557→        }
   558→   558→        
   559→   559→        addEvent(event);
   560→   560→    }

   565→   565→    void recordCorruption(uint64_t address, const MemoryValue& corrupt_value,
   566→   566→                           const std::string& detection_method,
   567→   567→                           Severity severity = Severity::Critical) {
   568→   568→

   571→   571→        event.type = MemoryEventType::CorruptionDetected;
   572→   572→        event.address = address;
   573→   573→        event.size = corrupt_value.size;
   574→   574→        event.new_value = corrupt_value;
   575→   575→        event.subsystem = Subsystem::DiagnosticSystem;
   576→   576→        event.metadata["detection_method"] = detection_method;
   577→   577→

   579→   579→    }

   585→   585→    /**
   586→   586→     * Find who wrote to a specific address
   587→   587→     */
   588→   588→    std::vector<WriterInfo> whoWrote(uint64_t address) const {
   589→   589→        std::shared_lock lock(m_mutex);
   590→   590→        
   591→   591→        std::vector<WriterInfo> writers;
   592→   592→        auto events = getEventsForAddress(address);
   593→   593→        
   594→   594→        for (const auto& event : events) {
   595→   595→            if (event.type == MemoryEventType::Write || 
   596→   596→                event.type == MemoryEventType::Allocation) {
   597→   597→                WriterInfo writer;
   598→   598→                writer.event = event;
   599→   599→                
   600→   600→                // Analyze suspicion
   601→   601→                analyzeSuspicion(writer);
   602→   602→                
   603→   603→                writers.push_back(writer);
   604→   604→            }
   605→   605→        }
   606→   606→        
   607→   607→        return writers;
   608→   608→    }

   613→   613→    std::optional<WriterInfo> lastWriter(uint64_t address) const {
   614→   614→        auto writers = whoWrote(address);
   615→   615→        if (writers.empty()) return std::nullopt;
   616→   616→        return writers.back();  // Most recent
   617→   617→    }

   619→   619→    /**
   620→   620→     * Get complete history for an address or range
   621→   621→     */
   622→   622→    MemoryHistoryQuery memoryHistory(uint64_t address, size_t range = 1) const {
   623→   623→        MemoryHistoryQuery query;
   624→   624→        query.target_address = address;
   625→   625→        query.query_range = range;
   626→   626→        
   627→   627→        std::shared_lock lock(m_mutex);
   628→   628→        
   629→   629→        // Collect all events in range
   630→   630→        for (uint64_t addr = address; addr < address + range; ++addr) {
   631→   631→            auto events = getEventsForAddress(addr);
   632→   632→            
   633→   633→            for (const auto& event : events) {
   634→   634→                WriterInfo info;
   635→   635→                info.event = event;
   636→   636→                
   637→   637→                if (event.type == MemoryEventType::Write ||
   638→   638→                    event.type == MemoryEventType::Allocation) {
   639→   639→                    query.all_writers.push_back(info);
   640→   640→                    query.total_writes++;
   641→   641→                    analyzeSuspicion(info);
   642→   642→                    if (info.is_suspicious) {
   643→   643→                        query.suspicious_writes.push_back(info);
   644→   644→                    }
   645→   645→                    query.last_writer = info;
   646→   646→                    query.current_value = event.new_value;
   647→   647→                    
   648→   648→                    if (query.first_access_frame == 0 || 
   649→   649→                        event.timestamp.frame_number < query.first_access_frame) {
   650→   650→                        query.first_access_frame = event.timestamp.frame_number;
   651→   651→                    }
   652→   652→                    if (event.timestamp.frame_number > query.last_access_frame) {
   653→   653→                        query.last_access_frame = event.timestamp.frame_number;
   654→   654→                    }
   655→   655→                } else if (event.type == MemoryEventType::Read) {
   656→   656→                    query.all_readers.push_back(info);
   657→   657→                    query.total_reads++;
   658→   658→                }
   659→   659→            }
   660→   660→        }
   661→   661→        
   662→   662→        return query;
   663→   663→    }

   665→   665→    /**
   666→   666→     * Find all addresses written by a specific subsystem
   667→   667→     */
   668→   668→    std::vector<uint64_t> addressesWrittenBy(Subsystem subs) const {
   669→   669→        std::shared_lock lock(m_mutex);
   670→   670→        
   671→   671→        std::set<uint64_t> addresses;
   672→   672→        for (const auto& [addr, events] : m_address_index) {
   673→   673→            for (const auto& event : events) {
   674→   674→                if (event.subsystem == subs && 
   675→   675→                    (event.type == MemoryEventType::Write ||
   676→   676→                     event.type == MemoryEventType::Allocation)) {
   677→   677→                    addresses.insert(addr);
   678→   678→                    break;
   679→   679→                }
   680→   680→            }
   681→   681→        }
   682→   682→        
   683→   683→        return std::vector<uint64_t>(addresses.begin(), addresses.end());
   684→   684→    }

   737→   737→                    case MemoryEventType::CorruptionDetected:
   738→   738→                        stats.corruption_events++; break;
   739→   739→                    default: break;
   740→   740→                }
   741→   741→            }
   742→   742→        }
   743→   743→

   799→   799→    mutable std::shared_mutex m_mutex;

   801→   801→    // Storage: address -> list of events (ring buffer behavior via size limit)
   802→   802→    std::map<uint64_t, std::list<MemoryEvent>> m_address_index;
   803→   803→    
   804→   804→    // Watch ranges
   805→   805→    std::vector<AddressRange> m_watch_ranges;
   806→   806→    
   807→   807→    // Active allocations (for auto-unwatch on free)
   808→   808→    std::map<uint64_t, AddressRange> m_allocations;

   813→   813→    // ========================================================================
   814→   814→    // Internal Methods

   816→   816→    
   817→   817→    bool shouldTrack(uint64_t address, size_t size) const {
   818→   818→        // Always check watched ranges first
   819→   819→        for (const auto& range : m_watch_ranges) {
   820→   820→            if (range.overlaps(address, size)) {
   821→   821→                return true;
   822→   822→            }
   823→   823→        }
   824→   824→        
   825→   825→        // Check if address is in a tracked allocation
   826→   826→        auto it = m_allocations.lower_bound(address);
   827→   827→        if (it != m_allocations.end()) {
   828→   828→            if (it->second.contains(address, size)) {
   829→   829→                return true;
   830→   830→            }
   831→   831→        }
   832→   832→        
   833→   833→        return false;
   834→   834→    }
   835→   835→    
   836→   836→    void addEvent(const MemoryEvent& event) {
   837→   837→        std::unique_lock lock(m_mutex);
   838→   838→        
   839→   839→        // Check global limit
   840→   840→        if (m_total_events >= m_config.max_total_events) {
   841→   841→            return;  // Or could evict oldest
   842→   842→        }
   843→   843→        
   844→   844→        // Add to address index
   845→   845→        auto& events = m_address_index[event.address];
   846→   846→        events.push_back(event);
   847→   847→        
   848→   848→        // Enforce per-address limit (ring buffer)
   849→   849→        while (events.size() > m_config.max_events_per_address) {
   850→   850→            events.pop_front();
   851→   851→        }
   852→   852→        
   853→   853→        m_total_events++;
   854→   854→        
   855→   855→        // Auto-detect corruption patterns if enabled
   856→   856→        if (m_config.detect_corruption_patterns &&
   857→   857→            event.type == MemoryEventType::Write) {
   858→   858→            detectCorruptionPatterns(event, events);
   859→   859→        }
   860→   860→    }
   861→   861→    
   862→   862→    std::list<MemoryEvent> getEventsForAddress(uint64_t address) const {
   863→   863→        auto it = m_address_index.find(address);
   864→   864→        if (it != m_address_index.end()) {
   865→   865→            return it->second;
   866→   866→        }
   867→   867→        return {};
   868→   868→    }
   869→   869→    
   870→   870→    void analyzeSuspicion(WriterInfo& writer) const {
   871→   871→        // Pattern: Writing to recently freed memory
   872→   872→        // Pattern: Writing outside allocation bounds
   873→   873→        // Pattern: GPU writing to CPU memory (or vice versa)
   874→   874→        // Pattern: Large value change in single write
   875→   875→        
   876→   876→        const auto& event = writer.event;
   877→   877→        
   878→   878→        // Check if writing to freed memory
   879→   879→        auto alloc_it = m_allocations.lower_bound(event.address);
   880→   880→        if (alloc_it != m_allocations.begin()) {
   881→   881→            --alloc_it;
   882→   882→            if (alloc_it != m_allocations.end()) {
   883→   883→                const auto& alloc = alloc_it->second;
   884→   884→                if (event.address >= alloc.end) {
   885→   885→                    writer.is_suspicious = true;
   886→   886→                    writer.suspicion_reason = "Write to potentially freed memory region";
   887→   887→                }
   888→   888→            }
   889→   889→        }
   890→   890→        
   891→   891→        // Check for large value changes
   892→   892→        if (event.old_value.size == event.new_value.size &&
   893→   893→            event.old_value.size >= 4) {
   894→   894→            uint64_t old_val = event.old_value.asUint64();
   895→   895→            uint64_t new_val = event.new_value.asUint64();
   896→   896→            
   897→   897→            // If most bits changed, might be corruption
   898→   898→            uint64_t xor_result = old_val ^ new_val;
   899→   899→            int changed_bits = __builtin_popcountll(xor_result);
   900→   900→            
   901→   901→            if (changed_bits > (sizeof(uint64_t) * 8 * 0.7)) {  // 70%+ bits changed
   902→   902→                writer.is_suspicious = true;
   903→   903→                writer.suspicion_reason = "Large value change (" + 
   904→   904→                    std::to_string(changed_bits) + " of " + 
   905→   905→                    std::to_string(sizeof(uint64_t) * 8) + " bits changed)";
   906→   906→            }
   907→   907→        }
   908→   908→        
   909→   909→        // Cross-subsystem writes can be suspicious
   910→   910→        if (event.subsystem == Subsystem::GPU &&
   911→   911→            event.address < 0x100000000ULL) {  // Low memory (likely CPU)
   912→   912→            writer.is_suspicious = true;
   913→   913→            writer.suspicion_reason = "GPU write to low memory address space";
   914→   914→        }
   915→   915→    }
   916→   916→    
   917→   917→    void detectCorruptionPatterns(const MemoryEvent& new_event,
   918→   918→                                   const std::list<MemoryEvent>& existing_events) {
   919→   919→        // Look for pattern: write, write-back different, write again
   920→   920→        // This can indicate race conditions or use-after-free
   921→   921→        
   922→   922→        if (existing_events.size() < 3) return;
   923→   923→        
   924→   924→        auto it = existing_events.rbegin();
   925→   925→        ++it;  // Skip the event we just added
   926→   926→        
   927→   927→        // Check recent writes for inconsistencies
   928→   928→        MemoryValue prev_value;
   929→   929→        int write_count = 0;
   930→   930→        
   931→   931→        for (; it != existing_events.rend() && write_count < 5; ++it) {
   932→   932→            if (it->type == MemoryEventType::Write) {
   933→   933→                write_count++;
   934→   934→                
   935→   935→                if (!prev_value.bytes.empty()) {
   936→   936→                    // Values oscillating between different states?
   937→   937→                    if (prev_value.bytes != it->new_value.bytes &&
   938→   938→                        it->new_value.bytes != new_event.new_value.bytes) {
   939→   939→                        
   940→   940→                        // Could be normal, flag for review
   941→   941→                        // (In production, would have more sophisticated heuristics)
   942→   942→                    }
   943→   943→                }
   944→   944→                
   945→   945→                prev_value = it->new_value;
   946→   946→            }
   947→   947→        }
   948→   948→    }

   951→   951→        std::stringstream ss;
   952→   952→        ss << std::this_thread::get_id();
   953→   953→        return ss.str();
   954→   954→    }
